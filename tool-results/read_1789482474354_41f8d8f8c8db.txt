    44→type Row = Record<string, unknown>;
    45→
    46→function pick(row: Row, keys: string[]): unknown {
    47→  for (const k of keys) {
    48→    const v = row[k];
    49→    if (v !== undefined && v !== null && v !== "") return v;
    50→  }
    51→  return undefined;
    52→}
    53→
    54→function pickStr(row: Row, keys: string[]): string {
    55→  const v = pick(row, keys);
    56→  if (v === undefined) return "";
    57→  // مقادیر شیء (مثل realtor: {...}) از نگاشت مستقیم مستثنی‌اند؛
    58→  // نگاشت تو در تو در فراخواننده انجام می‌شود.
    59→  if (typeof v === "object") return "";
    60→  return String(v).trim();
    61→}
    62→
    63→function pickNum(row: Row, keys: string[]): number {
    64→  const v = pick(row, keys);
    65→  if (v === undefined) return 0;
    66→  if (typeof v === "number") return Number.isFinite(v) ? v : 0;
    67→  const n = Number(String(v).replace(/[٬,،\s]/g, ""));
    68→  return Number.isFinite(n) ? n : 0;
    69→}
    70→
    71→function pickId(row: Row, fallback: string): string {
    72→  const v = pick(row, ["_id", "id", "code"]);
    73→  if (v === undefined) return fallback;
    74→  if (typeof v === "object") return String((v as Row).$oid ?? (v as Row).toString?.() ?? fallback);
    75→  return String(v);
    76→}
    77→
    78→/** خواندن نام از شیء تو در تو: businessId.title، role.name، creator.fullName و… */
    79→function pickNestedStr(row: Row, paths: string[]): string {
    80→  for (const path of paths) {
    81→    const segs = path.split(".");
    82→    let cur: unknown = row;
    83→    for (const s of segs) {
    84→      if (cur === null || cur === undefined || typeof cur !== "object") {
    85→        cur = undefined;
    86→        break;
    87→      }
    88→      cur = (cur as Row)[s];
    89→    }
    90→    if (cur !== null && cur !== undefined && cur !== "" && typeof cur !== "object") {
    91→      return String(cur).trim();
    92→    }
    93→  }
    94→  return "";
    95→}

   107→export function mapArmTradedRow(row: Row, index: number, page: number, limit: number): CrmTransaction | null {
   108→  if (!row || typeof row !== "object") return null;
   109→

   111→    pickStr(row, ["propertyCode", "property", "listingCode", "adCode"]) ||
   112→    pickNestedStr(row, ["property.code", "advertisement.code"]);

   114→  const propertyCustomer =
   115→    pickStr(row, ["propertyCustomer", "propertyCustomerName", "propertyOwner", "ownerName"]) ||
   116→    pickNestedStr(row, ["property.customer.fullName", "property.customer"]) ||
   117→    null;
   118→
   119→  const requestCode =
   120→    pickStr(row, ["requestCode", "request", "demandCode"]) ||
   121→    pickNestedStr(row, ["request.code"]) ||

   124→  const requestCustomer =
   125→    pickStr(row, ["requestCustomer", "requestCustomerName", "requesterName", "demanderName"]) ||

   127→    null;
   128→
   129→  // تاریخ معامله — ستون contractDate خود سامانه (جلالی یا ISO)
   130→  const contractDateDate =
   131→    anyDateStringToDate(pick(row, ["contractDate", "tradedAt", "dealDate", "createdAt", "date"])) ??

   135→    pickStr(row, ["contractHolo", "holo", "hologram", "holoCode"]) || null;

   139→
   140→  return {
   141→    id: pickId(row, `arm-traded-${page}-${limit}-${index}-${Date.now()}`),
   142→    contractDate: contractDateDate.toISOString(),
   143→    propertyCode: propertyCode || "—",
   144→    propertyCustomer: propertyCustomer || null,

   359→/**
   360→ * استخراج «نام و نام خانوادگی + شماره تماس» از ساختار کاربر سامانه ARM —
   361→ * این ساختار در پاسخ‌های properties/owner و requests/applicant مشترک است:
   362→ *   { _id, firstName?, lastName?, fullName?, mobileNumber, phoneNumber?,
   363→ *     extraUserInfo: [{ infos: [{ firstName, lastName, mobileNumber, userId }] }] }
   364→ */
   365→export function extractPersonInfo(payload: unknown): PersonInfo | null {
   366→  if (!payload || typeof payload !== "object") return null;
   367→  const p = payload as Row;
   368→
   369→  // ۱) نام — مسیرهای مستقیم
   370→  let firstName = pickStr(p, ["firstName", "first_name"]);
   371→  let lastName = pickStr(p, ["lastName", "last_name"]);
   372→  let fullName = pickStr(p, ["fullName", "full_name", "name"]);
   373→
   374→  // ۲) نام — مسیر تو در تو extraUserInfo[].infos[0]
   375→  if (!fullName && !firstName && !lastName) {
   376→    const extra = Array.isArray(p.extraUserInfo) ? (p.extraUserInfo as unknown[]) : [];
   377→    for (const e of extra) {
   378→      if (!e || typeof e !== "object") continue;
   379→      const infos = (e as Row).infos;
   380→      if (!Array.isArray(infos) || infos.length === 0) continue;
   381→      const info = infos[0];
   382→      if (!info || typeof info !== "object") continue;
   383→      const infoRow = info as Row;
   384→      firstName = pickStr(infoRow, ["firstName", "first_name"]);
   385→      lastName = pickStr(infoRow, ["lastName", "last_name"]);
   386→      if (firstName || lastName) {
   387→        if (!fullName) fullName = pickStr(infoRow, ["fullName", "name"]);
   388→        break;
   389→      }
   390→    }
   391→  }
   392→
   393→  // ۳) شماره تماس — مستقیم یا از extraUserInfo
   394→  let mobile = pickStr(p, ["mobileNumber", "mobile", "phoneNumber", "phone"]);
   395→  if (!mobile) {
   396→    const extra = Array.isArray(p.extraUserInfo) ? (p.extraUserInfo as unknown[]) : [];
   397→    for (const e of extra) {
   398→      if (!e || typeof e !== "object") continue;
   399→      const infos = (e as Row).infos;
   400→      if (!Array.isArray(infos) || infos.length === 0) continue;
   401→      const infoRow = infos[0] as Row;
   402→      mobile = pickStr(infoRow, ["mobileNumber", "mobile"]);
   403→      if (mobile) break;
   404→    }
   405→  }
   406→
   407→  const phone = pickStr(p, ["phoneNumber", "phone", "tel"]);
   408→  const isRegisteredUser = Boolean(pick(p, ["_id", "id", "userId"]));
   409→
   410→  const resolvedName =
   411→    fullName ||
   412→    [firstName, lastName].filter(Boolean).join(" ").trim();
   413→
   414→  if (!resolvedName && !mobile) return null;
   415→
   416→  // نرمال‌سازی شماره موبایل (ارقام فارسی/عربی → لاتین)
   417→  const normalizedMobile = normalizeMobile(mobile);
   418→  const normalizedPhone = phone ? String(phone).replace(/[۰-۹]/g, (d) => String("۰۱۲۳۴۵۶۷۸۹".indexOf(d))) : "";
   419→
   420→  return {
   421→    fullName: resolvedName || "نامشخص",
   422→    mobile: normalizedMobile,
   423→    phone: normalizedPhone,
   424→    isRegisteredUser,
   425→  };
   426→}

   428→/** نرمال‌سازی شماره موبایل: ارقام فارسی→لاتین، حذف جداکننده‌ها، پیشوند ۰ */
   429→export function normalizeMobile(input: string): string {
   430→  if (!input) return "";
   431→  const latin = String(input).replace(/[۰-۹]/g, (d) => String("۰۱۲۳۴۵۶۷۸۹".indexOf(d)));
   432→  const digits = latin.replace(/[^0-9+]/g, "");
   433→  if (digits.startsWith("+98")) return `0${digits.slice(3)}`;
   434→  if (digits.startsWith("98") && digits.length === 12) return `0${digits.slice(2)}`;
   435→  return digits;
   436→}

   438→// ==========================================================================
   439→// نگاشت جزئیات ساختاری (details) از پاسخ properties/details یا requests/details

   442→/**
   443→ * جزئیات ساختاری آگهی/درخواست را از پاسخ details استخراج می‌کند:
   444→ * ساختار واقعی: { details: [{ label, value, unit }], address, ... }
   445→ */
   446→export function extractDetailItems(payload: unknown): { label: string; value: string; unit: string }[] {
   447→  if (!payload || typeof payload !== "object") return [];
   448→  const p = payload as Row;
   449→  const list = Array.isArray(p.details) ? (p.details as unknown[]) : [];
   450→  const items: { label: string; value: string; unit: string }[] = [];
   451→  for (const d of list) {
   452→    if (!d || typeof d !== "object") continue;
   453→    const dr = d as Row;
   454→    const label = pickStr(dr, ["label", "title", "name", "key"]);
   455→    const value = pickStr(dr, ["value", "val"]);
   456→    const unit = pickStr(dr, ["unit", "unitTitle"]);
   457→    if (label && value) items.push({ label, value, unit });
   458→  }
   459→  return items;
   460→}

   462→/** ساخت آدرس خوانا از پاسخ details + فیلدهای مکانی آگهی/درخواست */
   463→export function extractAddress(payload: unknown, rowFallback: Row | null): string {
   464→  const parts: string[] = [];
   465→  if (payload && typeof payload === "object") {
   466→    const p = payload as Row;
   467→    const direct = pickStr(p, ["address", "fullAddress"]);
   468→    if (direct) parts.push(direct);
   469→    // عناوین مکانی تو در تو: state.name، city.name، area.name، line.name
   470→    for (const key of ["state", "city", "area", "line"]) {
   471→      const nested = pickNestedStr(p, [`${key}.name`, `${key}.title`]);
   472→      if (nested) parts.push(nested);
   473→    }
   474→  }
   475→  if (rowFallback && parts.length === 0) {
   476→    const title = pickStr(rowFallback, ["title"]);
   477→    if (title) parts.push(title);
   478→    for (const key of ["state", "city", "area", "line"]) {
   479→      const nested = pickNestedStr(rowFallback, [`${key}.name`, `${key}.title`]);
   480→      if (nested) parts.push(nested);
   481→    }
   482→  }
   483→  // حذف تکراری‌ها با حفظ ترتیب
   484→  const unique = [...new Set(parts.map((s) => s.trim()).filter(Boolean))];
   485→  return unique.join("، ");
   486→}

   492→/**
   493→ * پاسخ‌های صفحه‌بندی سامانه در قالب‌های مختلفی برمی‌گردند:
   494→ * {data: [...]} | {data: {docs|rows|list|items|data: [...], count, totalPages}}
   495→ * این تابع همه را به {rows, total, totalPages} نرمال می‌کند.
   496→ */
   497→export function extractPagedList<T>(payload: unknown, mapRow: (row: Row, index: number) => T | null): { rows: T[]; total: number; totalPages?: number } {
   498→  let list: unknown = payload;
   499→  let total = 0;
   500→  let totalPages: number | undefined;
   501→
   502→  if (payload && typeof payload === "object") {
   503→    const p = payload as Row;
   504→    // data.docs / data.rows / data.data / data.list / data.items
   505→    const inner = (p.data ?? p.docs ?? p.rows ?? p.list ?? p.items ?? p.result) as Row | undefined;
   506→    if (inner && typeof inner === "object" && !Array.isArray(inner)) {
   507→      list = (inner.docs ?? inner.rows ?? inner.data ?? inner.list ?? inner.items ?? inner.result ?? []) as unknown;
   508→      total = Number(pick(inner as Row, ["count", "total", "totalDocs", "totalCount"]) ?? 0) || 0;
   509→      totalPages = Number(pick(inner as Row, ["totalPages", "pages", "pageCount"]) ?? 0) || undefined;
   510→    } else if (Array.isArray(inner)) {
   511→      list = inner;
   512→    }
   513→    if (!total && Array.isArray(list)) total = list.length;
   514→  }
   515→
   516→  if (!Array.isArray(list)) list = [];
   517→  const rows = (list as Row[])
   518→    .map((row, i) => mapRow(row ?? {}, i))
   519→    .filter((r): r is T => r !== null);
   520→  if (!total) total = rows.length;
   521→  return { rows, total, totalPages };
   522→}

   572→/**
   573→ * ادغام جزئیات «trade-overviews/info/{id}» در قرارداد —
   574→ * تاریخ سند/قرارداد/تحویل، عنوان و آدرس ملک، خریداران، فروشندگان و
   575→ * مشاوران با حق‌السهم (اصلی و فرعی).

   577→export function mergeArmContractInfo(contract: CrmContract, infoPayload: unknown): CrmContract {
   578→  const info = (infoPayload && typeof infoPayload === "object" ? (infoPayload as Row) : {}) as Row;
   579→  const toDate = (keys: string[]): string | null => {
   580→    const d = anyDateStringToDate(pickStr(info, keys));
   581→    return d ? d.toISOString() : null;
   582→  };
   583→
   584→  const buyersRaw = Array.isArray(info.buyerList) ? (info.buyerList as Row[]) : [];
   585→  const sellersRaw = Array.isArray(info.sellerList) ? (info.sellerList as Row[]) : [];
   586→  const primaryRaw = Array.isArray(info.primaryRealtorList) ? (info.primaryRealtorList as Row[]) : [];
   587→  const secondaryRaw = Array.isArray(info.secondaryRealtorList) ? (info.secondaryRealtorList as Row[]) : [];
   588→
   589→  return {
   590→    ...contract,
   591→    contractDate: toDate(["contractDate"]) ?? contract.createdAt,
   592→    documentDate: toDate(["documentDate"]),
   593→    deliveryDate: toDate(["contractPropertyDate"]),
   594→    propertyTitle: pickNestedStr(info, ["property.title", "propertyTitle"]) || null,
   595→    propertyAddress: pickNestedStr(info, ["property.address", "propertyAddress"]) || null,
   596→    contractType: pickNestedStr(info, ["contractType.label"]) || contract.contractType,
   597→    buyers: buyersRaw
   598→      .map((b) => ({ fullName: pickStr(b, ["fullName"]), mobileNumber: pickStr(b, ["mobileNumber"]) }))
   599→      .filter((b) => b.fullName),
   600→    sellers: sellersRaw
   601→      .map((s) => ({ fullName: pickStr(s, ["fullName"]), mobileNumber: pickStr(s, ["mobileNumber"]) }))
   602→      .filter((s) => s.fullName),
   603→    realtorShares: [
   604→      ...primaryRaw.map((r) => mapRealtorShare(r, "primary")).filter((r): r is ContractRealtorShare => r !== null),
   605→      ...secondaryRaw.map((r) => mapRealtorShare(r, "secondary")).filter((r): r is ContractRealtorShare => r !== null),
   606→    ],
   607→  };
   608→}

   644→      detail: detailRaw.map((d) => {

   647→          contractType: pickStr(d, ["contractType"]),

   655→      }),
   656→    });
   657→  }
