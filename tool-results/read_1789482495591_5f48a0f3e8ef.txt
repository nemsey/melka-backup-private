   110→/** نرمال‌سازی جست‌وجوی فارسی (ارقام فارسی/عربی، آ ← ا، حذف فاصله/ZWNJ) */
   111→function normFa(s: string): string {
   112→  return s
   113→    .replace(/[۰-۹]/g, (d) => String("۰۱۲۳۴۵۶۷۸۹".indexOf(d)))
   114→    .replace(/[٠-٩]/g, (d) => String("٠١٢٣٤٥٦٧٨٩".indexOf(d)))
   115→    .replace(/[آأإ]/g, "ا")
   116→    .replace(/[\u200c\s]/g, "")
   117→    .trim();
   118→}
