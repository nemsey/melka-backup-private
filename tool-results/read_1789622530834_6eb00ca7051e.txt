   125→/** همگام‌سازی با توکن ARM — با قفل همزمانی: اجرای موازی = پیوستن به اجرای جاری */
   126→export function syncFromArm(armToken: string): Promise<SyncResult> {
   127→  if (__armSyncInFlight) return __armSyncInFlight; // در جریان است → صبر تا اتمام (نه شروع مجدد)
   128→  __armSyncExecutionCount += 1;
   129→  __armSyncInFlight = doSyncFromArm(armToken).finally(() => {
   130→    __armSyncInFlight = null;
   131→  });
   132→  return __armSyncInFlight;
   133→}

   135→async function readSyncState() {
   136→  let state = await db.syncState.findUnique({ where: { id: SYNC_STATE_ID } });
   137→  if (!state) {
   138→    state = await db.syncState.create({ data: { id: SYNC_STATE_ID } });
   139→  }
   140→  return state;
   141→}

   143→function parseSummary(json: string | null | undefined): TradedSummary | null {
   144→  if (!json) return null;
   145→  try {
   146→    const s = JSON.parse(json) as TradedSummary;
   147→    if (typeof s.totalCount === "number") return s;
   148→    return null;
   149→  } catch {
   150→    return null;
   151→  }
   152→}

   154→/** خواندن وضعیت همگام‌سازی برای نمایش (بدون خطا پرتاب نمی‌شود) */
   155→export async function getSyncState(): Promise<SyncStateInfo> {
   156→  const state = await readSyncState();
   157→  const [txCount, exCount, inCount, viCount, sdCount] = await Promise.all([
   158→    db.transactionRecord.count(),
   159→    db.expertRecord.count(),
   160→    db.interactionRecord.count(),
   161→    db.visitRecord.count(),
   162→    db.supplyDemandRecord.count(),
   163→  ]);
   164→  return {
   165→    lastSyncedAt: state.lastSyncedAt?.toISOString() ?? null,
   166→    lastSuccessAt: state.lastSuccessAt?.toISOString() ?? null,
   167→    lastError: state.lastError,
   168→    lastErrorCode: state.lastErrorCode,
   169→    transactionCount: txCount,
   170→    expertCount: exCount,
   171→    interactionCount: inCount,
   172→    visitCount: viCount,
   173→    supplyDemandCount: sdCount,
   174→    summary: parseSummary(state.summaryJson),
   175→    attempts: state.attempts,
   176→  };

   204→/** خواندن یک گزارش صفحه‌بندی‌شده تا انتها (با سقف ایمن صفحات) */
   205→async function fetchAllPages<T>(
   206→  path: string,
   207→  armToken: string,
   208→  maxPages: number,
   209→  pageLimit: number,
   210→  mapRow: (row: Record<string, unknown>, index: number, page: number, limit: number) => T | null
   211→): Promise<{ rows: T[]; rawSummary: unknown }> {
   212→  const cfg = getMelkApiConfig();
   213→  const rows: T[] = [];
   214→  let rawSummary: unknown = null;
   215→  for (let page = 1; page <= maxPages; page++) {
   216→    const res = await armCall<unknown>({
   217→      path,
   218→      baseUrl: cfg.armApiBase,
   219→      token: armToken,
   220→      query: { page, limit: pageLimit },
   221→      timeoutMs: cfg.timeoutMs,
   222→      maxRetries: 0, // در حلقه صفحه‌بندی تلاش مجدد نمی‌کنیم (صفحه بعدی خودش تازه می‌شود)
   223→    });
   224→    // خلاصه چهارگانه گزارش تطابق — خود پاسخ «data» است
   225→    if (path === MELK_ENDPOINTS.arm.tradedTotal && res.data && typeof res.data === "object") {
   226→      const d = res.data as Record<string, unknown>;
   227→      if (typeof d.totalCount === "number" || typeof d.hasBoth === "number") {
   228→        rawSummary = d;
   229→      }
   230→    }
   231→    const { rows: pageRows, total, totalPages } = extractPagedList(res.data, (row, i) => mapRow(row, i, page, pageLimit));
   232→    rows.push(...pageRows);
   233→    const pages = totalPages ?? (total > 0 ? Math.ceil(total / pageLimit) : 1);
   234→    if (page >= pages || pageRows.length === 0) break;
   235→  }
   236→  return { rows, rawSummary };
   237→}

   239→/**
   240→ * استخراج کامل معاملات، کارشناسان، تعاملات، بازدیدها و عرضه/تقاضا از سامانه ARM
   241→ * با توکن ARM. صفحه‌بندی به‌طور خودکار طی می‌شود (تا سقف ایمن صفحات).
   242→ * (پیاده‌سازی داخلی — همیشه از syncFromArm استفاده کنید تا قفل همزمانی اعمال شود)
   243→ */
   244→async function doSyncFromArm(armToken: string): Promise<SyncResult> {
   245→  const now = new Date();
   246→  const incremental = syncMode() === "incremental";
   247→  const runId = randomUUID();
   248→  const syncStartedAt = Date.now();
   249→  await writeSyncState({
   250→    lastSyncedAt: now,
   251→    // شناسه اجرا فقط در حالت افزایشی معنا دارد (legacy به آن دست نمی‌زند)
   252→    ...(incremental ? { syncRunId: runId } : {}),
   253→  });
   254→  // P2-05 — رویداد الزامی همگام‌سازی: آغاز
   255→  logSync("sync-run", { status: "start", mode: incremental ? "incremental" : "legacy", detail: { runId } });
   256→
   257→  try {
   258→    // ---------------- ۱) گزارش تطابق معاملات + خلاصه ----------------
   259→    const { rows: transactions, rawSummary } = await fetchAllPages(
   260→      MELK_ENDPOINTS.arm.tradedTotal,
   261→      armToken,
   262→      50,
   263→      100,
   264→      mapArmTradedRow
   265→    );
   266→    const summary = mapArmTradedSummary(rawSummary) ?? computeSummaryFromRows(transactions);
   267→
   268→    // ---------------- ۲) فهرست کاربران/مشاوران ----------------
   269→    const { rows: experts } = await fetchAllPages(MELK_ENDPOINTS.arm.realtorsList, armToken, 30, 100, mapArmRealtorRow);
   270→
   271→    // ---------------- ۳) تعاملات مشاوران با یکدیگر ----------------
   272→    let interactions: CrmInteraction[] = [];
   273→    let interactionsFetched = false;
   274→    try {
   275→      const { rows: inRows } = await fetchAllPages(MELK_ENDPOINTS.arm.realtorInteractions, armToken, 20, 200, mapArmInteractionRow);
   276→      interactions = inRows;
   277→      interactionsFetched = true;
   278→    } catch {
   279→      // تعاملات گزارش مکمل است — نبودش نباید کل همگام‌سازی را شکست بدهد
   280→      interactions = [];
   281→    }
   282→
   283→    // ---------------- ۴) بازدیدها و پیگیری‌ها (غیربحرانی) ----------------
   284→    let visits: CrmVisit[] = [];
   285→    let visitsFetched = false;
   286→    try {
   287→      const { rows: viRows } = await fetchAllPages(MELK_ENDPOINTS.arm.visitsList, armToken, 20, 100, mapArmVisitRow);
   288→      visits = viRows;
   289→      visitsFetched = true;
   290→    } catch {
   291→      visits = [];
   292→    }
   293→
   294→    // ---------------- ۵) گزارش عرضه و تقاضا (غیربحرانی) ----------------
   295→    let supplyDemand: CrmSupplyDemand[] = [];
   296→    let supplyDemandFetched = false;
   297→    try {
   298→      const { rows: sdRows } = await fetchAllPages(MELK_ENDPOINTS.armReports.supplyDemand, armToken, 10, 100, mapArmSupplyDemandRow);
   299→      supplyDemand = sdRows;
   300→      supplyDemandFetched = true;
   301→    } catch {
   302→      supplyDemand = [];
   303→    }
   304→
   305→    // ---------------- ۶) قراردادها و اسناد (صورت وضعیت‌ها) (غیربحرانی) ----------------
   306→    // لیست + جزئیات هر قرارداد (خریداران، فروشندگان، حق‌السهم مشاوران)
   307→    let contracts: CrmContract[] = [];
   308→    let contractsFetched = false;
   309→    try {
   310→      contracts = await fetchContracts(armToken);
   311→      contractsFetched = true;
   312→    } catch {
   313→      contracts = [];
   314→    }
   315→
   316→    // ---------------- ذخیره در پایگاه محلی ----------------
   317→    if (incremental) {
   318→      // P2-04 — مسیر افزایشی: upsert بر کلید طبیعی + نقطه ازسرگیری + نامه مرده
   319→      await applyIncrementalSync(runId, {
   320→        transactions, experts, interactions, visits, supplyDemand, contracts,
   321→        interactionsFetched, visitsFetched, supplyDemandFetched, contractsFetched,
   322→      });
   323→    } else {
   324→      // مسیر legacy — عین کد قبلی: پاک‌سازی کامل + درج اتمیک
   325→      await db.$transaction([
   326→        db.transactionRecord.deleteMany({}),
   327→        db.expertRecord.deleteMany({}),
   328→        db.interactionRecord.deleteMany({}),
   329→        db.visitRecord.deleteMany({}),
   330→        db.supplyDemandRecord.deleteMany({}),
   331→        db.contractRecord.deleteMany({}),
   332→        ...toTxCreates(transactions),
   333→        ...toExCreates(experts),
   334→        ...toInCreates(interactions),
   335→        ...toViCreates(visits),
   336→        ...toSdCreates(supplyDemand),
   337→        ...toContractCreates(contracts),
   338→      ]);
   339→    }
   340→
   341→    const successAt = new Date();
   342→    await writeSyncState({
   343→      lastSuccessAt: successAt,
   344→      lastError: null,
   345→      lastErrorCode: null,
   346→      attempts: 0,
   347→      summaryJson: summary ? JSON.stringify(summary) : null,
   348→      ...(incremental ? { lastWatermark: successAt } : {}),
   349→    });
   350→    // P2-05 — رویداد الزامی همگام‌سازی: پایان موفق
   351→    logSync("sync-run", {
   352→      status: "ok",
   353→      mode: incremental ? "incremental" : "legacy",
   354→      duration: Date.now() - syncStartedAt,
   355→      detail: {
   356→        runId,
   357→        transactions: transactions.length,
   358→        experts: experts.length,
   359→        interactions: interactions.length,
   360→        visits: visits.length,
   361→        supplyDemand: supplyDemand.length,
   362→        contracts: contracts.length,
   363→      },
   364→    });
   365→
   366→    return {
   367→      ok: true,
   368→      transactions: transactions.length,
   369→      experts: experts.length,
   370→      interactions: interactions.length,
   371→      visits: visits.length,
   372→      supplyDemand: supplyDemand.length,
   373→      contracts: contracts.length,
   374→      syncedAt: successAt.toISOString(),
   375→    };
   376→  } catch (err) {
   377→    const e = err instanceof MelkApiError ? err : new MelkApiError("UNKNOWN", (err as Error)?.message ?? "خطای نامشخص", 500, false);
   378→    const state = await readSyncState();
   379→    await writeSyncState({
   380→      lastError: e.message,
   381→      lastErrorCode: e.code,
   382→      attempts: state.attempts + 1,
   383→    });
   384→    // P2-05 — رویداد الزامی همگام‌سازی: شکست
   385→    logSync("sync-run", {
   386→      status: "fail",
   387→      mode: incremental ? "incremental" : "legacy",
   388→      duration: Date.now() - syncStartedAt,
   389→      errorCode: e.code,
   390→      detail: { runId },
   391→    });
   392→    return {
   393→      ok: false,
   394→      transactions: 0,
   395→      experts: 0,
   396→      interactions: 0,
   397→      visits: 0,
   398→      supplyDemand: 0,
   399→      contracts: 0,
   400→      syncedAt: now.toISOString(),
   401→      error: e.message,
   402→      errorCode: e.code,
   403→    };
   404→  }
   405→}

   407→/** اگر سامانه خلاصه چهارگانه را نفرستاد، از ردیف‌ها محاسبه می‌شود */
   408→function computeSummaryFromRows(rows: CrmTransaction[]): TradedSummary {
   409→  return {
   410→    totalCount: rows.length,
   411→    hasRequestEvent: rows.filter((r) => r.eventsCount > 0).length,
   412→    hasSuggestedProperty: rows.filter((r) => r.suggestedPropertyCount > 0).length,
   413→    hasBoth: rows.filter((r) => r.eventsCount > 0 && r.suggestedPropertyCount > 0).length,
   414→  };
   415→}

   487→  sellerListJson: c.sellers.length > 0 ? JSON.stringify(c.sellers) : null,
   488→  realtorListJson: c.realtorShares.length > 0 ? JSON.stringify(c.realtorShares) : null,
   489→  canAddRealtor: c.canAddRealtor,
   490→});

   492→/** ثبت/افزایش نامه مرده برای یک ردیف ناموفق — attempts پیاپی؛ ۳+ یعنی «مرده» */
   493→async function recordDeadLetter(runId: string, entity: string, naturalKey: string, err: unknown): Promise<void> {
   494→  const message = ((err as Error)?.message ?? "خطای نامشخص").slice(0, 300);
   495→  await db.syncDeadLetter
   496→    .upsert({
   497→      where: { entity_naturalKey: { entity, naturalKey } },
   498→      create: { runId, entity, naturalKey, attempts: 1, lastError: message },
   499→      update: { runId, attempts: { increment: 1 }, lastError: message, lastAttemptAt: new Date(), resolvedAt: null },
   500→    })
   501→    .catch(() => undefined); // خطای خود نامه مرده هرگز مسیر افزایشی را نمی‌شکند
   502→}
   503→
   504→/** ردیفی که این بار موفق شد → نامه مرده‌اش «حل‌شده» می‌شود */
   505→async function resolveDeadLetters(entity: string, naturalKeys: string[]): Promise<void> {
   506→  if (naturalKeys.length === 0) return;
   507→  await db.syncDeadLetter
   508→    .updateMany({ where: { entity, naturalKey: { in: naturalKeys }, resolvedAt: null }, data: { resolvedAt: new Date() } })
   509→    .catch(() => undefined);
   510→}

   512→/** کلیدهای باز (حل‌نشده) یک خانواده — از جاروی سنگ‌قبرها محافظت می‌شوند */
   513→async function openDeadLetterKeys(entity: string): Promise<Set<string>> {
   514→  const rows = await db.syncDeadLetter.findMany({ where: { entity, resolvedAt: null }, select: { naturalKey: true } }).catch(() => []);
   515→  return new Set(rows.map((r) => r.naturalKey));
   516→}

   518→/** مشخصات یک خانواده داده برای مسیر افزایشی */
   519→interface IncrementalFamily<T> {
   520→  entity: "transactions" | "experts" | "interactions" | "visits" | "supplyDemand" | "contracts";
   521→  rows: T[];
   522→  /** واکشی این خانواده موفق بوده؟ (شکست واکشی ⇒ خانواده دست‌نخورده می‌ماند — تفاوت عمدی با legacy) */
   523→  fetched: boolean;
   524→  keyOf: (row: T) => string;
   525→  upsert: (rows: T[]) => Promise<unknown>;
   526→  /** کلیدهای موجود در پایگاه (برای جاروی سنگ‌قبرها) */
   527→  listExistingKeys: () => Promise<string[]>;
   528→  deleteByKeys: (keys: string[]) => Promise<unknown>;
   529→}
   530→
   531→/**
   532→ * اعمال افزایشی: به‌جای پاک‌سازی کامل، upsert بر کلید طبیعی + جاروی
   533→ * ردیف‌های محو‌شده (با محافظت نامه‌های مرده باز) + نقطه ازسرگیری پس از
   534→ * هر خانواده. شکست ردیف منفرد کل اجرا را نمی‌شکند (نامه مرده می‌شود).
   535→ */
   536→/** صادرات برای تست (tests/phase2-sync.test.ts) — در کد محصول از syncFromArm استفاده کنید */
   537→export async function applyIncrementalSync(
   538→  runId: string,
   539→  data: {
   540→    transactions: CrmTransaction[];
   541→    experts: CrmExpert[];
   542→    interactions: CrmInteraction[];
   543→    visits: CrmVisit[];
   544→    supplyDemand: CrmSupplyDemand[];
   545→    contracts: CrmContract[];
   546→    interactionsFetched: boolean;
   547→    visitsFetched: boolean;
   548→    supplyDemandFetched: boolean;
   549→    contractsFetched: boolean;
   550→  }
   551→): Promise<void> {
   552→  const txRepo = getTransactionRepository();
   553→  const exRepo = getExpertRepository();
   554→  const mkRepo = getMarketRepository();
   555→  const ctRepo = getContractRepository();
   556→  const viRepo = getVisitRepository();
   557→
   558→  const families: IncrementalFamily<never>[] = [
   559→    {
   560→      entity: "transactions",
   561→      rows: data.transactions.map(toTxUpsert) as never[],
   562→      fetched: true,
   563→      keyOf: (t) => (t as unknown as TransactionUpsertInput).id,
   564→      upsert: (rows) => txRepo.upsertBatch(rows as unknown as TransactionUpsertInput[]),
   565→      listExistingKeys: async () => (await db.transactionRecord.findMany({ select: { id: true } })).map((r) => r.id),
   566→      deleteByKeys: (keys) => db.transactionRecord.deleteMany({ where: { id: { in: keys } } }),
   567→    },
   568→    {
   569→      entity: "experts",
   570→      rows: data.experts.map(toExUpsert) as never[],
   571→      fetched: true,
   572→      keyOf: (e) => (e as unknown as ExpertUpsertInput).personnelCode,
   573→      upsert: (rows) => exRepo.upsertBatch(rows as unknown as ExpertUpsertInput[]),
   574→      listExistingKeys: async () => (await db.expertRecord.findMany({ select: { personnelCode: true } })).map((r) => r.personnelCode),
   575→      deleteByKeys: (keys) => db.expertRecord.deleteMany({ where: { personnelCode: { in: keys } } }),
   576→    },
   577→    {
   578→      entity: "interactions",
   579→      rows: data.interactions.map(toInUpsert) as never[],
   580→      fetched: data.interactionsFetched,
   581→      keyOf: (i) => (i as unknown as InteractionUpsertInput).id,
   582→      upsert: (rows) => exRepo.upsertInteractionsBatch(rows as unknown as InteractionUpsertInput[]),
   583→      listExistingKeys: async () => (await db.interactionRecord.findMany({ select: { id: true } })).map((r) => r.id),
   584→      deleteByKeys: (keys) => db.interactionRecord.deleteMany({ where: { id: { in: keys } } }),
   585→    },
   586→    {
   587→      entity: "visits",
   588→      rows: data.visits.map(toViUpsert) as never[],
   589→      fetched: data.visitsFetched,
   590→      keyOf: (v) => (v as unknown as VisitUpsertInput).id,
   591→      upsert: (rows) => viRepo.upsertBatch(rows as unknown as VisitUpsertInput[]),
   592→      listExistingKeys: async () => (await db.visitRecord.findMany({ select: { id: true } })).map((r) => r.id),
   593→      deleteByKeys: (keys) => db.visitRecord.deleteMany({ where: { id: { in: keys } } }),
   594→    },
   595→    {
   596→      entity: "supplyDemand",
   597→      rows: data.supplyDemand.map(toSdUpsert) as never[],
   598→      fetched: data.supplyDemandFetched,
   599→      keyOf: (s) => (s as unknown as SupplyDemandUpsertInput).id,
   600→      upsert: (rows) => mkRepo.upsertSupplyDemandBatch(rows as unknown as SupplyDemandUpsertInput[]),
   601→      listExistingKeys: async () => (await db.supplyDemandRecord.findMany({ select: { id: true } })).map((r) => r.id),
   602→      deleteByKeys: (keys) => db.supplyDemandRecord.deleteMany({ where: { id: { in: keys } } }),
   603→    },
   604→    {
   605→      entity: "contracts",
   606→      rows: data.contracts.map(toContractUpsert) as never[],
   607→      fetched: data.contractsFetched,
   608→      keyOf: (c) => (c as unknown as ContractUpsertInput).id,
   609→      upsert: (rows) => ctRepo.upsertBatch(rows as unknown as ContractUpsertInput[]),
   610→      listExistingKeys: async () => (await db.contractRecord.findMany({ select: { id: true } })).map((r) => r.id),
   611→      deleteByKeys: (keys) => db.contractRecord.deleteMany({ where: { id: { in: keys } } }),
   612→    },
   613→  ];
   614→
   615→  const done: Array<{ entity: string; count: number; failed: number }> = [];
   616→
   617→  // شناسه اجرا از همان ابتدا ثبت می‌شود (خودکفا بودن applier برای تست/فراخوانی مستقیم)
   618→  await writeSyncState({ syncRunId: runId });
   619→
   620→  for (const family of families) {
   621→    // واکشی ناموفق ⇒ این خانواده در این اجرا دست‌نخورده می‌ماند (از دست رفتن داده نداریم)
   622→    if (!family.fetched) {
   623→      done.push({ entity: family.entity, count: 0, failed: -1 }); // -1 = skipped
   624→      continue;
   625→    }
   626→
   627→    const keys = family.rows.map(family.keyOf);
   628→    let failedKeys: string[] = [];
   629→
   630→    try {
   631→      // تلاش دسته‌ای (مسیر سریع — اکثر اجراها موفق‌اند)
   632→      await family.upsert(family.rows);
   633→    } catch {
   634→      // شکست دسته‌ای ⇒ تک‌تک برای ایزوله کردن ردیف‌های خراب (upsertها idempotentند)
   635→      for (let i = 0; i < family.rows.length; i++) {
   636→        try {
   637→          await family.upsert([family.rows[i]]);
   638→        } catch (err) {
   639→          failedKeys.push(keys[i]);
   640→          await recordDeadLetter(runId, family.entity, keys[i], err);
   641→        }
   642→      }
   643→    }
   644→
   645→    // ردیف‌های موفق این خانواده که قبلاً نامه مرده بودند → حل‌شده
   646→    await resolveDeadLetters(family.entity, keys.filter((k) => !failedKeys.includes(k)));
   647→
   648→    // جاروی سنگ‌قبرها: ردیف‌هایی که دیگر در سامانه نیستند (به‌جز نامه‌های مرده باز)
   649→    try {
   650→      const existing = await family.listExistingKeys();
   651→      const fetchedSet = new Set(keys);
   652→      const openDead = await openDeadLetterKeys(family.entity);
   653→      const stale = existing.filter((k) => !fetchedSet.has(k) && !openDead.has(k));
   654→      if (stale.length > 0) await family.deleteByKeys(stale);
   655→    } catch {
   656→      // جارو شکست خورد ⇒ اجرا همچنان موفق است؛ دفع بعد تمیز می‌کند
   657→    }
   658→
   659→    // نقطه ازسرگیری — پس از هر خانواده به‌روزرسانی می‌شود
   660→    done.push({ entity: family.entity, count: family.rows.length, failed: failedKeys.length });
   661→    await writeSyncState({
   662→      lastSyncCursor: JSON.stringify({ runId, done: done.map((d) => d.entity), detail: done }),
   663→    });
   664→  }
   665→}

   667→/**
   668→ * همگام‌سازی با توکن ARM نشست جاری — اگر توکن منقضی شده باشد
   669→ * ابتدا با refreshToken تمدید می‌شود.
   670→ */
   671→export async function syncWithRefresh(
   672→  armToken: string,
   673→  armRefreshToken: string
   674→): Promise<{ sync: SyncResult; newArmToken?: string }> {
   675→  const provider = getMelkProvider();
   676→  let token = armToken;
   677→  let newArmToken: string | undefined;
   678→
   679→  // تلاش اول؛ در صورت انقضا، تمدید و تلاش مجدد
   680→  let sync = await syncFromArm(token);
   681→  if (!sync.ok && (sync.errorCode === "UNAUTHORIZED" || sync.errorCode === "OTP_INVALID") && armRefreshToken) {
   682→    try {
   683→      const refreshed = await provider.refreshArmToken(armRefreshToken);
   684→      token = refreshed.accessToken;
   685→      newArmToken = refreshed.accessToken;
   686→      sync = await syncFromArm(token);
   687→    } catch (refreshErr) {
   688→      const e = refreshErr instanceof MelkApiError ? refreshErr : new MelkApiError("UNKNOWN", "تمدید توکن ناموفق", 401, false);
   689→      sync = { ok: false, transactions: 0, experts: 0, interactions: 0, visits: 0, supplyDemand: 0, contracts: 0, syncedAt: new Date().toISOString(), error: e.message, errorCode: e.code };
   690→    }
   691→  }
   692→  return { sync, newArmToken };
   693→}

   696→export async function ensureSeeded(armToken?: string): Promise<{ transactions: number; experts: number; interactions: number; visits: number; supplyDemand: number; liveError?: string }> {
   697→  const [txCount, exCount, inCount, viCount, sdCount] = await Promise.all([
   698→    db.transactionRecord.count(),
   699→    db.expertRecord.count(),
   700→    db.interactionRecord.count(),
   701→    db.visitRecord.count(),
   702→    db.supplyDemandRecord.count(),
   703→  ]);
   704→
   705→  if ((txCount === 0 || exCount === 0) && armToken) {
   706→    const { sync } = await syncWithRefresh(armToken, "");
   707→    if (sync.ok) return { transactions: sync.transactions, experts: sync.experts, interactions: sync.interactions, visits: sync.visits, supplyDemand: sync.supplyDemand };
   708→    return { transactions: txCount, experts: exCount, interactions: inCount, visits: viCount, supplyDemand: sdCount, liveError: sync.error };
   709→  }
   710→  return { transactions: txCount, experts: exCount, interactions: inCount, visits: viCount, supplyDemand: sdCount };
   711→}

   887→/** خواندن کش فهرست عمومی — null اگر نیست یا کهنه است */
   888→async function readRefCache<T>(key: string): Promise<T | null> {
   889→  const row = await db.refListCache.findUnique({ where: { key } });
   890→  if (!row) return null;
   891→  const age = Date.now() - row.fetchedAt.getTime();
   892→  if (age > REF_TTL_MS) return null;
   893→  try {
   894→    return JSON.parse(row.dataJson) as T;
   895→  } catch {
   896→    return null;
   897→  }
   898→}

   906→  });
   907→}

   968→      db.refListCache.findUnique({ where: { key: REF_KEYS.seriousness } }),

   976→      jobs: parse(jobs),
   977→      roles: parse(roles),
