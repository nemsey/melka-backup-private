   186→   186→/** تأخیر پیش از تلاش retry شماره n (1-based) — null یعنی «دیگر retry نکن» */
   187→   187→export function retryDelayFor(retryNumber: number, maxRetries: number): number | null {
   188→   188→  if (retryNumber < 1) return null;
   189→   189→  if (retryNumber > maxRetries) return null;
   190→   190→  const idx = Math.min(retryNumber - 1, DEFAULT_RETRY_DELAYS_MS.length - 1);
   191→   191→  return DEFAULT_RETRY_DELAYS_MS[idx];
   192→   192→}

   208→   208→/** §۱.۷ — قابل‌retry: 5xx/429/شبکه (پرچم retryable)؛ 4xx غیر 429 = فوری؛ ناشناخته = فوری */
   209→   209→export function classifyFetchError(err: unknown): { retryable: boolean; retryAfterMs: number | null } {
   210→   210→  if (err instanceof BulkFetchError) {
   211→   211→    return { retryable: err.retryable, retryAfterMs: err.retryAfterMs ?? null };
   212→   212→  }
   213→   213→  if (err instanceof MelkApiError) {
   214→   214→    const retryable = err.retryable === true || err.httpStatus >= 500 || err.httpStatus === 429;
   215→   215→    return { retryable, retryAfterMs: null };
   216→   216→  }
   217→   217→  return { retryable: false, retryAfterMs: null };
   218→   218→}

   222→   222→  const n = Number.isFinite(raw) ? Math.floor(raw as number) : 100;

   237→   237→// ==========================================================================
   238→   238→
   239→   239→export class RateLimiter {
   240→   240→  private lastCompletedAt = 0;
   241→   241→
   242→   242→  constructor(private readonly minIntervalMs: number) {}
   243→   243→
   244→   244→  /** پیش از هر درخواست — نخستین درخواست آزاد است */
   245→   245→  async wait(): Promise<void> {
   246→   246→    if (this.lastCompletedAt === 0) return;
   247→   247→    const elapsed = Date.now() - this.lastCompletedAt;
   248→   248→    const remaining = this.minIntervalMs - elapsed;
   249→   249→    if (remaining > 0) await defaultSleep(remaining);
   250→   250→  }
   251→   251→
   252→   252→  /** پس از پایان هر پاسخ (موفق یا ناموفق) */
   253→   253→  complete(): void {
   254→   254→    this.lastCompletedAt = Date.now();
   255→   255→  }
   256→   256→}

   262→   262→/** سقف ایمنی صفحات (حلقه بی‌نهایت منبع خراب را نمی‌پذیرد) — additive option */
   263→   263→const DEFAULT_MAX_PAGES = 10_000;
   264→   264→
   265→   265→class BulkImportRunner {
   266→   266→  private readonly opts: BulkImportOptions;
   267→   267→  private readonly pageSize: number;
   268→   268→  private readonly batchSize: number;
   269→   269→  private readonly maxRetries: number;
   270→   270→  private readonly deps: BulkImportDeps;
   271→   271→  private readonly limiter: RateLimiter;
   272→   272→  private readonly sleep: (ms: number) => Promise<void>;
   273→   273→
   274→   274→  constructor(options: BulkImportOptions, deps: BulkImportDeps) {
   275→   275→    this.opts = options;
   276→   276→    this.pageSize = clampPageSize(options.pageSize);
   277→   277→    this.batchSize = clampBatchSize(options.batchSize);
   278→   278→    this.maxRetries = options.maxRetries ?? 5;
   279→   279→    this.deps = deps;
   280→   280→    this.limiter = new RateLimiter(Math.max(1, options.rateLimitMs ?? 200));
   281→   281→    this.sleep = deps.sleep ?? defaultSleep;
   282→   282→  }
   283→   283→
   284→   284→  private now(): Date {
   285→   285→    return this.deps.now ? this.deps.now() : new Date();
   286→   286→  }
   287→   287→
   288→   288→  async run(): Promise<BulkImportResult> {
   289→   289→    const t0 = this.now().getTime();
   290→   290→    const entity = this.opts.entity;
   291→   291→    const range =
   292→   292→      this.opts.startDate || this.opts.endDate
   293→   293→        ? {
   294→   294→            from: (this.opts.startDate ?? this.opts.endDate)!.toISOString(),
   295→   295→            to: (this.opts.endDate ?? this.opts.startDate)!.toISOString(),
   296→   296→          }
   297→   297→        : null;
   298→   298→
   299→   299→    // ---- ۱) بارگذاری checkpoint (تطبیق entity + بازه + نسخه) ----
   300→   300→    const saved = await this.deps.checkpointStore.load(entity);
   301→   301→    const resumable =
   302→   302→      saved !== null &&
   303→   303→      saved.version === BULK_CHECKPOINT_VERSION &&
   304→   304→      sameDateRange(saved.dateRange ?? null, range);
   305→   305→
   306→   306→    const cp: BulkImportCheckpoint = resumable
   307→   307→      ? { ...saved, runId: saved.runId ?? this.deps.newRunId?.() ?? randomUUID() }
   308→   308→      : {
   309→   309→          entity,
   310→   310→          lastPage: 0,
   311→   311→          lastRecordId: "",
   312→   312→          totalProcessed: 0,
   313→   313→          startedAt: this.now().toISOString(),
   314→   314→          version: BULK_CHECKPOINT_VERSION,
   315→   315→          runId: this.deps.newRunId?.() ?? randomUUID(),
   316→   316→          updatedAt: this.now().toISOString(),
   317→   317→          dateRange: range,
   318→   318→          totals: { fetched: 0, saved: 0, duplicates: 0, errors: 0 },
   319→   319→        };
   320→   320→
   321→   321→    const progress: BulkImportProgress = {
   322→   322→      entity,
   323→   323→      totalFetched: cp.totals?.fetched ?? 0,
   324→   324→      totalSaved: cp.totals?.saved ?? 0,
   325→   325→      totalDuplicates: cp.totals?.duplicates ?? 0,
   326→   326→      totalErrors: cp.totals?.errors ?? 0,
   327→   327→      currentPage: cp.lastPage,
   328→   328→      currentBatch: 0,
   329→   329→      checkpoint: cp,
   330→   330→      startedAt: new Date(cp.startedAt),
   331→   331→      updatedAt: this.now(),
   332→   332→    };
   333→   333→
   334→   334→    let status: BulkImportResult["status"] = "COMPLETED";
   335→   335→    let error: string | undefined;
   336→   336→
   337→   337→    // ---- ۲) حلقه صفحات ----
   338→   338→    let page = cp.lastPage + 1; // نقطه ادامه = آخرین صفحه کامل + ۱
   339→   339→    let pagesRan = 0;
   340→   340→
   341→   341→    while (true) {
   342→   342→      if (pagesRan >= DEFAULT_MAX_PAGES) {
   343→   343→        // سقف ایمنی — از سر رفتن حلقه منبع خراب؛ داده‌های تا اینجا سالم‌اند
   344→   344→        status = "PARTIAL";
   345→   345→        error = `سقف ایمنی صفحات (${DEFAULT_MAX_PAGES}) رسید؛ داده تا صفحه ${cp.lastPage} سالم ذخیره شد.`;
   346→   346→        break;
   347→   347→      }
   348→   348→
   349→   349→      let fetched: BulkFetchPage;
   350→   350→      try {
   351→   351→        fetched = await this.fetchPageWithRetry(page);
   352→   352→      } catch (err) {
   353→   353→        // §۱.۴-۳ — checkpoint (آخرین صفحه کامل) ذخیره + توقف با خطای صریح فارسی
   354→   354→        await this.deps.checkpointStore.save(cp);
   355→   355→        status = "FAILED";
   356→   356→        error = fetchFailureMessage(err, entity, page, cp.lastRecordId);
   357→   357→        break;
   358→   358→      }
   359→   359→
   360→   360→      progress.totalFetched += fetched.rows.length;
   361→   361→      pagesRan += 1;
   362→   362→
   363→   363→      if (fetched.rows.length === 0) break; // پایان داده
   364→   364→
   365→   365→      // ---- ذخیره: برش‌های batchSize + فلاش اجباری پیش از پیشروی checkpoint ----
   366→   366→      for (let offset = 0; offset < fetched.rows.length; offset += this.batchSize) {
   367→   367→        const chunk = fetched.rows.slice(offset, offset + this.batchSize);
   368→   368→        const stats = await this.deps.sink.saveBatch(chunk, { sourceRunId: cp.runId!, entity });
   369→   369→        progress.totalSaved += stats.created + stats.updated;
   370→   370→        progress.totalDuplicates += stats.duplicates;
   371→   371→        progress.totalErrors += stats.errors;
   372→   372→        progress.currentBatch += 1;
   373→   373→      }
   374→   374→
   375→   375→      // نا-تکرارپذیری فلاش/چک‌پوینت: صفحه کامل شد → داده نوشته شد → حالا checkpoint جلو می‌رود
   376→   376→      const lastRow = fetched.rows[fetched.rows.length - 1] as Record<string, unknown> | undefined;
   377→   377→      cp.lastRecordId = String(lastRow?.id ?? lastRow?.propertyCode ?? lastRow?.requestCode ?? lastRow?.personnelCode ?? "");
   378→   378→      cp.totalProcessed = progress.totalFetched;
   379→   379→      cp.lastPage = page;
   380→   380→      cp.updatedAt = this.now().toISOString();
   381→   381→      cp.totals = {
   382→   382→        fetched: progress.totalFetched,
   383→   383→        saved: progress.totalSaved,
   384→   384→        duplicates: progress.totalDuplicates,
   385→   385→        errors: progress.totalErrors,
   386→   386→      };
   387→   387→      await this.deps.checkpointStore.save(cp);
   388→   388→
   389→   389→      progress.currentPage = page;
   390→   390→      progress.checkpoint = { ...cp };
   391→   391→      progress.updatedAt = this.now();
   392→   392→      this.opts.onProgress?.({ ...progress, checkpoint: { ...cp, totals: cp.totals ? { ...cp.totals } : undefined } });
   393→   393→
   394→   394→      if (!fetched.hasMore) break;
   395→   395→      page += 1;
   396→   396→    }
   397→   397→
   398→   398→    // خطای رکورد-منفرد درون صفحات موفق ⇒ PARTIAL (نه FAILED)
   399→   399→    if (status === "COMPLETED" && progress.totalErrors > 0) {
   400→   400→      status = "PARTIAL";
   401→   401→      error = `${progress.totalErrors} رکورد در حین ذخیره خطا خورد (صفحات طی شد؛ از اجرای مجدد idempotent استفاده کنید).`;
   402→   402→    }
   403→   403→
   404→   404→    return {
   405→   405→      entity,
   406→   406→      status,
   407→   407→      progress,
   408→   408→      ...(error !== undefined ? { error } : {}),
   409→   409→      durationMs: this.now().getTime() - t0,
   410→   410→    };
   411→   411→  }
   412→   412→
   413→   413→  /** واکشی یک صفحه با retry طبقه‌بندی‌شده + احترام کامل به rate limit و Retry-After */
   414→   414→  private async fetchPageWithRetry(page: number): Promise<BulkFetchPage> {
   415→   415→    let retryNumber = 0;
   416→   416→    while (true) {
   417→   417→      await this.limiter.wait();
   418→   418→      try {
   419→   419→        const res = await this.deps.fetcher.fetchPage({
   420→   420→          entity: this.opts.entity,
   421→   421→          page,
   422→   422→          pageSize: this.pageSize,
   423→   423→          ...(this.opts.startDate ? { startDate: this.opts.startDate } : {}),
   424→   424→          ...(this.opts.endDate ? { endDate: this.opts.endDate } : {}),
   425→   425→        });
   426→   426→        this.limiter.complete();
   427→   427→        return res;
   428→   428→      } catch (err) {
   429→   429→        this.limiter.complete(); // درخواست (ناموفق) هم «تمام شد» — فاصله از پایان آن
   430→   430→        const { retryable, retryAfterMs } = classifyFetchError(err);
   431→   431→        const policyDelay = retryDelayFor(retryNumber + 1, this.maxRetries);
   432→   432→        if (!retryable || policyDelay === null) throw err;
   433→   433→        // 429: احترام به Retry-After مبدأ — هرگز کمتر از سیاست، هرچه بیشتر خواست شکیبایی
   434→   434→        await this.sleep(Math.max(policyDelay, retryAfterMs ?? 0));
   435→   435→        retryNumber += 1;
   436→   436→      }
   437→   437→    }
   438→   438→  }
   439→   439→}

   455→   455→  options: BulkImportOptions,
   456→   456→  deps: Partial<BulkImportDeps> & { armToken?: string; repoClient?: PrismaClient } = {}
   457→   457→): Promise<BulkImportResult> {

   459→   459→    ...options,
   460→   460→    pageSize: clampPageSize(options.pageSize),
   461→   461→    batchSize: clampBatchSize(options.batchSize),
   462→   462→    rateLimitMs: Math.max(1, options.rateLimitMs ?? 200),
   463→   463→    maxRetries: options.maxRetries ?? 5,
   464→   464→  };
   465→   465→  const fetcher =
   466→   466→    deps.fetcher ??
   467→   467→    createArmBulkFetcher({ entity: options.entity, armToken: deps.armToken ?? "" });

   476→   476→    ...(deps.newRunId ? { newRunId: deps.newRunId } : {}),
   477→   477→  }).run();
   478→   478→}

   484→   484→/** نگاشت ردیف خام → ورودی مخزن برای هر موجودیت (listing/request = stub تا P6.5-04) */
   485→   485→type RawRow = Record<string, unknown>;
   486→   486→
   487→   487→function pickRaw(row: RawRow, keys: string[]): unknown {
   488→   488→  for (const k of keys) {
   489→   489→    const v = row[k];
   490→   490→    if (v !== undefined && v !== null && v !== "") return v;
   491→   491→  }
   492→   492→  return null;
   493→   493→}
   494→   494→
   495→   495→function pickRawStr(row: RawRow, keys: string[]): string | null {
   496→   496→  const v = pickRaw(row, keys);
   497→   497→  return typeof v === "string" ? v : v === undefined || v === null ? null : String(v);
   498→   498→}

   500→   500→/** STUB P6.5-04 — آگهی ARM → PropertyBulkInput (نگاشت حداقلی؛ تکمیل در normalize.ts) */
   501→   501→function stubMapPropertyRow(row: RawRow): PropertyBulkInput | null {
   502→   502→  const id = pickRawStr(row, ["id", "_id", "objectId"]);
   503→   503→  const propertyCode = pickRawStr(row, ["code", "propertyCode"]);
   504→   504→  if (!id && !propertyCode) return null;
   505→   505→  return {
   506→   506→    id: id ?? propertyCode!,
   507→   507→    propertyCode: propertyCode ?? id!,
   508→   508→    title: pickRawStr(row, ["title"]),
   509→   509→    status: pickRawStr(row, ["status"]),
   510→   510→    ownerName: pickRawStr(row, ["relatedRealtorName", "realtorName"]),
   511→   511→  };

   527→   527→export function createArmBulkFetcher(init: { entity: BulkEntity; armToken: string }): BulkPageFetcher {
   528→   528→  return {
   529→   529→    fetchPage: async (req) => {
   530→   530→      if (!init.armToken) {
   531→   531→        throw new BulkFetchError("توکن ARM برای Bulk Import لازم است — نشست کاربر واردشده را تمدید کنید.", { status: 401, retryable: false });
   532→   532→      }
   533→   533→      const cfg = getMelkApiConfig();
   534→   534→      const path = armEntityPath(req.entity);
   535→   535→      const res = await armCall<unknown>({
   536→   536→        path,
   537→   537→        baseUrl: cfg.armApiBase,
   538→   538→        token: init.armToken,
   539→   539→        query: {
   540→   540→          page: req.page,
   541→   541→          limit: req.pageSize,
   542→   542→          // نام پارامترهای بازه تاریخ در P6.5-04/D0 با تست زنده نهایی می‌شود (نگاشت صادقانه)
   543→   543→          ...(req.startDate ? { from: req.startDate.toISOString().slice(0, 10) } : {}),
   544→   544→          ...(req.endDate ? { to: req.endDate.toISOString().slice(0, 10) } : {}),
   545→   545→        },
   546→   546→        timeoutMs: cfg.timeoutMs,
   547→   547→        maxRetries: 0, // مدیریت retry با موتور خود خط لوله (نه client)
   548→   548→      });
   549→   549→
   550→   550→      const mapped = extractPagedList(res.data, (row, i) =>
   551→   551→        mapArmRowToInput(req.entity, (row ?? {}) as RawRow, i, req.page, req.pageSize)
   552→   552→      );
   553→   553→      const totalPages =
   554→   554→        mapped.totalPages ?? (mapped.total > 0 ? Math.ceil(mapped.total / req.pageSize) : 1);
   555→   555→      return {
   556→   556→        rows: mapped.rows,
   557→   557→        hasMore: mapped.rows.length > 0 && req.page < totalPages,
   558→   558→      };
   559→   559→    },
   560→   560→  };
   561→   561→}

   575→   575→function mapArmRowToInput(entity: BulkEntity, row: RawRow, index: number, page: number, pageSize: number): unknown | null {
   576→   576→  switch (entity) {
   577→   577→    case "transaction": {
   578→   578→      const crm = mapArmTradedRow(row, index, page, pageSize);
   579→   579→      return crm ? toTxUpsert(crm) : null;
   580→   580→    }
   581→   581→    case "expert": {
   582→   582→      const crm = mapArmRealtorRow(row, index);
   583→   583→      return crm ? toExUpsert(crm) : null;
   584→   584→    }
   585→   585→    case "interaction": {
   586→   586→      const crm = mapArmInteractionRow(row, index);
   587→   587→      return crm ? toInUpsert(crm) : null;
   588→   588→    }
   589→   589→    case "visit": {
   590→   590→      const crm = mapArmVisitRow(row, index);
   591→   591→      return crm ? toViUpsert(crm) : null;
   592→   592→    }
   593→   593→    case "contract": {
   594→   594→      // فهرست صورت‌وضعیت‌ها — غنی‌سازی مرحله دوم (info/{id}) در P6.5-04
   595→   595→      const crm = mapArmContractRow(row);
   596→   596→      return crm ? toContractUpsert(crm) : null;
   597→   597→    }
   598→   598→    case "listing": return stubMapPropertyRow(row);
   599→   599→    case "request": return stubMapRequestRow(row);
   600→   600→  }

   607→   607→export function createRepositorySink(entity: BulkEntity, opts: { client?: PrismaClient } = {}): BulkRecordSink {

   611→   611→  const viRepo = opts.client ? new PrismaVisitRepository({ client: opts.client }) : getVisitRepository();
   612→   612→  const ctRepo = opts.client ? new PrismaContractRepository({ client: opts.client }) : getContractRepository();
   613→   613→  const rqRepo = opts.client ? new PrismaRequestRepository({ client: opts.client }) : getRequestRepository();
   614→   614→  const prRepo = opts.client ? new PrismaPropertyRecordRepository({ client: opts.client }) : getPropertyRecordRepository();
   615→   615→
   616→   616→  return {
   617→   617→    saveBatch: async (rows, meta) => {
   618→   618→      const up: BulkUpsertOptions = { sourceRunId: meta.sourceRunId };
   619→   619→      switch (entity) {
   620→   620→        case "transaction":
   621→   621→          return txRepo.upsertBulk(rows as TransactionUpsertInput[], up);
   622→   622→        case "expert":
   623→   623→          return exRepo.upsertBulk(rows as ExpertUpsertInput[], up);
   624→   624→        case "interaction":
   625→   625→          return exRepo.upsertInteractionsBulk(rows as InteractionUpsertInput[], up);
   626→   626→        case "visit":
   627→   627→          return viRepo.upsertBulk(rows as VisitUpsertInput[], up);
   628→   628→        case "contract":
   629→   629→          return ctRepo.upsertBulk(rows as ContractUpsertInput[], up);
   630→   630→        case "request":
   631→   631→          return rqRepo.upsertArmBulk(rows as ArmRequestBulkRow[], up);
   632→   632→        case "listing":
   633→   633→          return prRepo.upsertBulk(rows as PropertyBulkInput[], up);
   634→   634→      }
   635→   635→    },
   636→   636→  };
   637→   637→}
