    29→const FA_TO_EN_DIGITS: Record<string, string> = {
    30→  "۰": "0", "۱": "1", "۲": "2", "۳": "3", "۴": "4", "۵": "5", "۶": "6", "۷": "7", "۸": "8", "۹": "9",
    31→  "٠": "0", "١": "1", "٢": "2", "٣": "3", "٤": "4", "٥": "5", "٦": "6", "٧": "7", "٨": "8", "٩": "9",
    32→};

    34→/** ارقام فارسی/عربی → لاتین (تعیین‌کننده — عین enDigits در jalali.ts) */
    35→export function latinDigits(input: string): string {
    36→  return input.replace(/[۰-۹٠-٩]/g, (d) => FA_TO_EN_DIGITS[d] ?? d);
    37→}
    38→
    39→/** یکسان‌سازی حروف: ي→ی، ك→ک، أ/إ/آ→ا، ة→ه + حذف اعراب/کشیده */
    40→export function unifyPersianLetters(input: string): string {
    41→  return input
    42→    .replace(/[ي]/g, "ی")
    43→    .replace(/[ك]/g, "ک")
    44→    .replace(/[آأإ]/g, "ا")
    45→    .replace(/[ة]/g, "ه")
    46→    .replace(/[\u064B-\u0652\u0670\u0640]/g, ""); // اعراب + تنوین + کشیده ( Tateel )
    47→}

    49→/** نویسه‌های کنترلی و جهت‌متن */
    50→const CTRL_RE = /[\u0000-\u0008\u000B\u000C\u000E-\u001F\u200E\u200F\u202A-\u202E]/g;

    52→// ==========================================================================
    53→// ۲) normalizeName — نام اشخاص/مالکان (عین B.1 دستور)

    56→/** نشان‌های نگارشی که از نام حذف می‌شوند (عین دستور: نقطه، ویرگول، پرانتز و…) */
    57→const NAME_PUNCT_RE = /[.,()\[\]{}«»"'\u066A\u066B\u066C،؛:!?؟…\-_ـ*#+@^~`|/\\]/g;
    58→
    59→/**
    60→ * نرمال‌سازی نام (اشخاص، مالکان) — عین قواعد B.1-۱ دستور:
    61→ *  · حذف ZWNJ (U+200C)
    62→ *  · ی/ک عربی → فارسی · أ/إ/آ → ا · ة → ه
    63→ *  · حذف فاصله اضافی (→ یک فاصله) · حذف نشان‌های نگارشی و اعراب
    64→ * نمونه شاهد دستور: «محمّد  احمدی» → «محمد احمدی»
    65→ */
    66→export function normalizeName(raw: string): string {
    67→  if (typeof raw !== "string" || raw.length === 0) return "";
    68→  let s = raw.replace(CTRL_RE, "");
    69→  s = latinDigits(s);
    70→  s = unifyPersianLetters(s);
    71→  s = s.replace(/\u200C/g, ""); // ZWNJ حذف (عین دستور)
    72→  s = s.replace(NAME_PUNCT_RE, " ");
    73→  s = s.replace(/\s+/g, " ").trim();
    74→  return s;
    75→}

    78→// ۳) normalizePropertyCode / normalizeRequestCode — کدهای ARM (عین B.1-۳)

    81→/** نرمال‌سازی مشترک کد آرم: trim · ارقام→لاتین · حذف فاصله/جداکننده · حرف اول ثابت · حذف صفر ابتدا */
    82→function normalizeArmCode(raw: string): string {
    83→  if (typeof raw !== "string") return "";
    84→  let s = latinDigits(raw).replace(CTRL_RE, "");
    85→  s = s.replace(/[\s\u200C\u066C,،_\-]/g, ""); // حذف فاصله و جداکننده‌های رایج کد
    86→  if (s.length === 0) return "";
    87→  // حرف اول (P یا R یا هر پیشوند لاتین) → بزرگ؛ بقیه فقط حرف/رقم نگه داشته می‌شود
    88→  const first = s[0];
    89→  const isLatinLetter = /[a-zA-Z]/.test(first);
    90→  if (isLatinLetter) s = first.toUpperCase() + s.slice(1);
    91→  // حذف صفرهای ابتدای بخش عددی (پس از پیشوند حرفی)
    92→  const body = isLatinLetter ? s.slice(1) : s;
    93→  const bodyNoLeadingZeros = body.replace(/^0+(?=\d)/, "");
    94→  const cleanedBody = bodyNoLeadingZeros.replace(/[^0-9A-Za-z]/g, "");
    95→  if (!isLatinLetter && cleanedBody.length === 0) return "";
    96→  if (isLatinLetter && cleanedBody.length === 0) return ""; // فقط حرف بدون بدنه → نامعتبر
    97→  return isLatinLetter ? first.toUpperCase() + cleanedBody : cleanedBody;
    98→}

   120→/**
   121→ * نرمال‌سازی موبایل → قالب استاندارد ۰۹xxxxxxxxx
   122→ *  · ارقام فارسی/عربی → لاتین · حذف نویسه‌های غیررقمی
   123→ *  · حذف +۹۸/۰۰۹۸ → ۰ · افزودن ۰ به ۱۰رقمی‌های آغازشونده با ۹
   124→ *  · اعتبارسنجی نهایی ^09\d{9}$ — نامعتبر → رشته خالی (شمارش در گزارش)
   125→ */
   126→export function normalizeMobile(raw: string): string {
   127→  if (typeof raw !== "string") return "";
   128→  let d = latinDigits(raw).replace(/[^0-9]/g, "");
   129→  if (d.length === 0) return "";
   130→  if (d.startsWith("0098")) d = "0" + d.slice(4);
   131→  else if (d.startsWith("98") && d.length === 12) d = "0" + d.slice(2);
   132→  if (d.length === 10 && d.startsWith("9")) d = "0" + d;
   133→  return /^09\d{9}$/.test(d) ? d : "";
   134→}

   136→// ==========================================================================
   137→// ۵) normalizeAmount — مبلغ → ریال پایه (عین B.1-۵ و §۳.۵ طراحی)
   138→// ==========================================================================
   139→
   140→/**
   141→ * نرمال‌سازی مبلغ → عدد پایه به «ریال».
   142→ *  · ارقام فارسی/عربی → لاتین · حذف جداکننده هزارگان (٬ ، , ' و فاصله)
   143→ *  · مقیاس‌واژه‌ها: میلیون ×۱۰۰۰ · میلیارد ×۱۰⁶ · تریلیون ×۱۰⁹ (نسبت به مبنای تومان)
   144→ *  · واحد: «ریال» ×۱ · «تومان» ×۱۰ · بدون واحد → پیش‌فرض تومان ×۱۰ (نمونه دستور: «۱٬۲۰۰٬۰۰۰» → ۱۲٬۰۰۰٬۰۰۰ ریال)
   145→ *  · كسر با «/» یا «.» یا «٫» بین ارقام (۲/۵ میلیارد = ۲٫۵)
   146→ *  · ورودی number → عیناً (مقدار پایه ساخت‌یافته — بدون تفسیر واحد متنی)
   147→ *  · ورودی متنی غیرقابل‌تجزیه → NaN (پاسخ صادقانه؛ فراخواننده باید فیلد را کنار بگذارد)
   148→ */
   149→export function normalizeAmount(raw: string | number): number {
   150→  if (typeof raw === "number") return Number.isFinite(raw) ? raw : NaN;
   151→  if (typeof raw !== "string") return NaN;
   152→  let s = latinDigits(raw).replace(CTRL_RE, "").trim();
   153→  if (s.length === 0) return NaN;
   154→
   155→  // واحد پول
   156→  const hasToman = s.includes("تومان");
   157→  const hasRial = !hasToman && s.includes("ریال");
   158→  const unitFactor = hasRial ? 1 : 10; // پیش‌فرض (بدون واحد) = تومان، عین نمونه دستور
   159→
   160→  // مقیاس‌واژه‌ها
   161→  let scale = 1;
   162→  if (s.includes("تریلیون")) scale = 1e12;
   163→  else if (s.includes("میلیارد")) scale = 1e9;
   164→  else if (s.includes("میلیون")) scale = 1e6;
   165→
   166→  // استخراج بخش عددی: نگه‌داشتن ارقام + جداکننده اعشار / و .
   167→  s = s.replace(/[\u066C\u060C,'\u00A0\s]/g, ""); // جداکننده هزارگان
   168→  s = s.replace(/\u066B/g, "."); // ممیز عربی اعشار
   169→  const decimalMatch = s.match(/(\d+)([/.])(\d+)/);
   170→  if (decimalMatch) {
   171→    s = decimalMatch[1] + "." + decimalMatch[3];
   172→  } else {
   173→    s = s.replace(/[/.]/g, ""); // نقطه/ممیل بدون جفت رقم = نویسه زائد
   174→  }
   175→  const numeric = s.match(/^-?\d+(\.\d+)?/);
   176→  if (!numeric) return NaN;
   177→  const value = parseFloat(numeric[0]);
   178→  if (!Number.isFinite(value)) return NaN;
   179→  return value * scale * unitFactor;
   180→}

   183→// ۶) normalizeJalaliDate — تاریخ جلالی → ISO 8601 نیمه‌شب UTC (عین B.1-۶ و §۳.۴)
   184→// ==========================================================================
   185→
   186→/**
   187→ * نرمال‌سازی تاریخ جلالی → «YYYY-MM-DDT00:00:00.000Z» (تاریخ تجاری بدون ساعت).
   188→ *  · فرم‌ها: 1400/01/15 · 1400-1-5 · ۱۴۰۰/۰۱/۱۵ (تبدیل با jalaliToGregorian موجود)
   189→ *  · سال ۱۳۰۰–۱۶۰۰ = جلالی؛ سال ≥ ۱۶۰۱ = میلادی (تحمل ورودی ISO)
   190→ *  · تاریخ نامعتبر (مثل 1400/13/01) → رشته خالی «» — هرگز مقدار ساختگی (§۳.۴ طراحی)
   191→ */
   192→export function normalizeJalaliDate(raw: string): string {
   193→  if (typeof raw !== "string") return "";
   194→  let s = latinDigits(raw).replace(CTRL_RE, "").trim();
   195→  if (s.length === 0) return "";
   196→
   197→  // ورودی ISO با ساعت → فقط بخش تاریخ
   198→  const iso = s.match(/^(\d{4})-(\d{2})-(\d{2})T/);
   199→  if (iso) s = `${iso[1]}/${iso[2]}/${iso[3]}`;
   200→
   201→  s = s.replace(/[-.٫]/g, "/");
   202→  const m = s.match(/^(\d{4})\/(\d{1,2})\/(\d{1,2})$/);
   203→  if (!m) return "";
   204→  const y = parseInt(m[1], 10);
   205→  const mo = parseInt(m[2], 10);
   206→  const d = parseInt(m[3], 10);
   207→
   208→  if (y >= 1300 && y <= 1600) {
   209→    // جلالی
   210→    if (mo < 1 || mo > 12) return "";
   211→    if (d < 1 || d > jalaliMonthLength(y, mo)) return "";
   212→    const [gy, gm, gd] = jalaliToGregorian(y, mo, d);
   213→    return new Date(Date.UTC(gy, gm - 1, gd)).toISOString();
   214→  }
   215→  if (y > 1600) {
   216→    // میلادی (تحمل ورودی)
   217→    if (mo < 1 || mo > 12) return "";
   218→    const dt = new Date(Date.UTC(y, mo - 1, d));
   219→    if (dt.getUTCMonth() !== mo - 1 || dt.getUTCDate() !== d) return ""; // ۳۰ فوریه و…
   220→    return dt.toISOString();
   221→  }
   222→  return ""; // سال خارج از بازه → قابل‌تبدیل نیست
   223→}

   229→/** توکن‌سازی واژه‌ای — جداکننده‌ها: فاصله/ZWNJ/نقطه‌گذاری؛ واژه‌های تک‌نویسه فقط اگر رقم باشند */
   230→export function tokenizeWords(text: string): string[] {
   231→  return text
   232→    .split(/[\s\u200C،؛:().!؟?()[\]{}«»"'،_\-…/\\|]+/)
   233→    .filter((t) => t.length > 1 || (t.length === 1 && /[0-9]/.test(t)));

   236→/** بردار فراوانی واژه (Term Frequency) — الگوریتم عین termVector در base-agent؛ رقم‌های تکی حفظ می‌شوند («ولیعصر 2») */
   237→export function wordTermVector(text: string): Map<string, number> {
   238→  const vec = new Map<string, number>();
   239→  for (const tok of tokenizeWords(text)) vec.set(tok, (vec.get(tok) ?? 0) + 1);
   240→  return vec;
   241→}

   243→/** بردار نویسه‌نگرام ۱+۲+۳-گرامی از رشته بدون فاصله — مقاوم‌سازی املای محاوره‌ای */
   244→export function charGramVector(flat: string): Map<string, number> {
   245→  const vec = new Map<string, number>();
   246→  const bump = (k: string) => vec.set(k, (vec.get(k) ?? 0) + 1);
   247→  for (let i = 0; i < flat.length; i++) bump(flat[i]);
   248→  for (let i = 0; i + 1 < flat.length; i++) bump(flat.slice(i, i + 2));
   249→  for (let i = 0; i + 2 < flat.length; i++) bump(flat.slice(i, i + 3));
   250→  return vec;
   251→}

   253→/** شباهت کسینوسی دو بردار فراوانی — الگوریتم عین cosineSimilarity در base-agent */
   254→export function cosineSimilarity(a: Map<string, number>, b: Map<string, number>): number {
   255→  if (a.size === 0 || b.size === 0) return 0;
   256→  let dot = 0;
   257→  let normA = 0;
   258→  let normB = 0;
   259→  for (const v of a.values()) normA += v * v;
   260→  for (const v of b.values()) normB += v * v;
   261→  if (normA === 0 || normB === 0) return 0;
   262→  for (const [k, v] of a) {
   263→    const w = b.get(k);
   264→    if (w) dot += v * w;
   265→  }
   266→  return dot / (Math.sqrt(normA) * Math.sqrt(normB));
   267→}

   332→/** کلید تطبیق دقیق: ارقام→لاتین، حروف یکسان، حذف فاصله/ZWNJ/نقطه‌گذاری، کوچک */

   336→    .replace(/\u200C/g, "")
   337→    .replace(/[\s«»"'()\[\]{}،؛:,.!?؟…\-_ـ*#+@^~`|/\\· ٬٬]+/g, "");
   338→  return out.toLowerCase();
   339→}
   340→
   341→/** نرمال‌سازی سبک برای توکن‌سازی لایه سوم */
   342→function lightNormalize(s: string): string {
   343→  return latinDigits(unifyPersianLetters(s)).replace(CTRL_RE, "").replace(/[\u200C]+/g, " ").trim();
   344→}
   345→
   346→/** ساخت رجیستری نمایه‌دار از مدخل‌های خام — خالص و deterministic؛
   347→ *  نمایه‌ها: کلید کامل + کلید هسته (بدون پرانتز) + توکن‌ها؛ بردارهای لایه دوم تنبل ساخته می‌شوند */
   348→export function buildLocationRegistry(rawEntries: readonly LocationEntry[]): LocationRegistry {
   349→  const indexed: IndexedEntry[] = rawEntries.map((e) => ({
   350→    ...e,
   351→    fullKey: locationKey(e.name),
   352→    coreKey: locationKey(stripParens(e.name) || e.name),
   353→    tokens: new Set(tokenizeWords(lightNormalize(e.name)).filter((t) => t.length >= 3)),
   354→  }));
   355→
   356→  const byKey = new Map<string, LocationEntry[]>();
   357→  const byIdMap = new Map<string, LocationEntry>();
   358→  const tokenIndex = new Map<string, Set<string>>();
   359→
   360→  const pushKey = (key: string, e: LocationEntry) => {
   361→    if (!key) return;
   362→    const arr = byKey.get(key);
   363→    if (arr) arr.push(e);
   364→    else byKey.set(key, [e]);
   365→  };
   366→  for (const e of indexed) {
   367→    pushKey(e.fullKey, e);
   368→    if (e.coreKey !== e.fullKey) pushKey(e.coreKey, e);
   369→    byIdMap.set(e.id, e);
   370→    for (const t of e.tokens) {
   371→      const set = tokenIndex.get(t);
   372→      if (set) set.add(e.id);
   373→      else tokenIndex.set(t, new Set([e.id]));
   374→    }
   375→  }
   376→
   377→  return {
   378→    entries: rawEntries,
   379→    findByKey: (key: string): readonly LocationEntry[] => byKey.get(key) ?? [],
   380→    findIdsByToken: (token: string): readonly string[] => [...(tokenIndex.get(token) ?? [])],
   381→    byId: (id: string): LocationEntry | undefined => byIdMap.get(id),
   382→  };
   383→}
   384→
   385→// بردارهای لایه دوم — ساخته‌شده تنبل (یک‌بار برای هر رجیستری، کش WeakMap)
   386→interface EntryVector {
   387→  words: Map<string, number>;
   388→  grams: Map<string, number>;
   389→  core: string;
   390→  kind: "area" | "line";
   391→}
   392→const REGISTRY_VECS = new WeakMap<LocationRegistry, Map<string, EntryVector>>();
   393→
   394→/** بردارهای مدخل‌های رجیستری — تنبل + کش‌شده؛ برای رجیستری‌های ساخته‌شده با buildLocationRegistry و رجیستری‌های تزریقی تست */
   395→function registryVectors(registry: LocationRegistry): Map<string, EntryVector> {
   396→  let m = REGISTRY_VECS.get(registry);
   397→  if (m) return m;
   398→  m = new Map<string, EntryVector>();
   399→  for (const e of registry.entries) {
   400→    const core = stripGenericWords(lightNormalize(stripParens(e.name) || e.name));
   401→    m.set(e.id, { words: wordTermVector(core), grams: charGramVector(locationKey(core)), core, kind: e.kind });
   402→  }
   403→  REGISTRY_VECS.set(registry, m);
   404→  return m;
   405→}

   407→// ==========================================================================
   408→// ۹) findLocationInRegistry — تطبیق سه‌لایه (عین B.1-۲ دستور)
   409→// ==========================================================================
   410→
   411→/** آستانه لایه برداری — عین دستور: score ≥ ۰٫۷ */
   412→export const AREA_MATCH_THRESHOLD = 0.7;
   413→
   414→function toMatch(e: LocationEntry, score: number, method: LocationMatch["method"]): LocationMatch {
   415→  return {
   416→    matchedId: e.id,
   417→    matchedName: e.name,
   418→    kind: e.kind,
   419→    areaId: e.areaId,
   420→    areaName: e.areaName,
   421→    score,
   422→    method,
   423→  };
   424→}
   425→
   426→function pickUniqueByKey(cands: readonly LocationEntry[]): LocationEntry | null {
   427→  const areas = cands.filter((c) => c.kind === "area");
   428→  if (areas.length === 1) return areas[0];
   429→  const lines = cands.filter((c) => c.kind === "line");
   430→  if (areas.length === 0 && lines.length === 1) return lines[0];
   431→  return null; // چند ناحیه یا چند خط هم‌نام → مبهم → عدم تطبیق (fail-closed)
   432→}
   433→
   434→/**
   435→ * تطبیق نام محله/منطقه در رجیستری — سه لایه عین دستور:
   436→ *  ۱) تطبیق دقیق (کلید کامل، سپس کلید هسته بدون پرانتز) — برخورد هم‌نام: ترجیح ناحیه
   437→ *  ۲) تطبیق برداری: max(کسینوس واژه‌ای، کسینوس نویسه‌نگرام) روی هسته بدون واژه عمومی — آستانه ۰٫۷
   438→ *     تساوی هم‌نوع (ناحیه-ناحیه / خط-خط) → مبهم؛ تساوی ناحیه-خط → ترجیح ناحیه
   439→ *  ۳) توکن متمایز مشترک (posting یکتا یا دوگانه که به یک مدخل می‌رسد)
   440→ * بدون تطبیق → null (صفر تطبیق کاذب — نمونه‌های ناموجود راستی‌آزمایی شده)
   441→ */
   442→export function findLocationInRegistry(raw: string, registry: LocationRegistry): LocationMatch | null {
   443→  if (typeof raw !== "string" || raw.trim().length === 0) return null;
   444→  const fullKey = locationKey(raw);
   445→  if (!fullKey) return null;
   446→
   447→  // ---- لایه ۱: تطبیق دقیق ----
   448→  const coreKey = locationKey(stripParens(raw) || raw);
   449→  for (const key of fullKey === coreKey ? [fullKey] : [fullKey, coreKey]) {
   450→    const cands = registry.findByKey(key);
   451→    if (cands.length === 1) return toMatch(cands[0], 1, "exact");
   452→    if (cands.length > 1) {
   453→      const picked = pickUniqueByKey(cands);
   454→      if (picked) return toMatch(picked, 1, "exact+pref");
   455→    }
   456→  }
   457→
   458→  // ---- لایه ۲: تطبیق برداری ----
   459→  const inputCore = stripGenericWords(lightNormalize(stripParens(raw) || raw));
   460→  const wv = wordTermVector(inputCore);
   461→  const gv = charGramVector(locationKey(inputCore));
   462→  const vecs = registryVectors(registry);
   463→  let bestEntry: LocationEntry | null = null;
   464→  let bestScore = 0;
   465→  let bestKind: "area" | "line" | null = null;
   466→  let secondScore = -1;
   467→  let secondKind: "area" | "line" | null = null;
   468→  for (const e of registry.entries) {
   469→    const v = vecs.get(e.id);
   470→    if (!v) continue;
   471→    const score = Math.max(cosineSimilarity(wv, v.words), cosineSimilarity(gv, v.grams));
   472→    if (bestEntry === null) {
   473→      bestEntry = e;
   474→      bestScore = score;
   475→      bestKind = v.kind;
   476→      continue;
   477→    }
   478→    if (score > bestScore + 1e-9) {
   479→      // برترین جدید — برترین قبلی به دومین تبدیل می‌شود
   480→      secondScore = bestScore;
   481→      secondKind = bestKind;
   482→      bestEntry = e;
   483→      bestScore = score;
   484→      bestKind = v.kind;
   485→    } else if (Math.abs(score - bestScore) <= 1e-9) {
   486→      if (v.kind === "area" && bestKind === "line") {
   487→        // تساوی ناحیه-خط → ناحیه جایگزین؛ خط به دومین تبدیل می‌شود
   488→        secondScore = bestScore;
   489→        secondKind = bestKind;
   490→        bestEntry = e;
   491→        bestKind = v.kind;
   492→      } else {
   493→        // هم‌امتیاز هم‌نوع (یا خط هم‌امتیاز با ناحیه برتر) → کاندید دوم
   494→        secondScore = score;
   495→        secondKind = v.kind;
   496→      }
   497→    } else if (score > secondScore) {
   498→      secondScore = score;
   499→      secondKind = v.kind;
   500→    }
   501→  }
   502→  if (bestEntry && bestKind && bestScore >= AREA_MATCH_THRESHOLD) {
   503→    // تساوی دقیق هم‌نوع (ناحیه-ناحیه / خط-خط) → مبهم (fail-closed)؛ ناحیه-خط هم‌امتیاز → ناحیه (بالا اعمال شد)
   504→    if (secondKind !== null && secondKind === bestKind && Math.abs(secondScore - bestScore) <= 1e-9) {
   505→      return null;
   506→    }
   507→    return toMatch(bestEntry, bestScore, "vector");
   508→  }
   509→
   510→  // ---- لایه ۳: توکن متمایز مشترک ----
   511→  const inputTokens = tokenizeWords(lightNormalize(raw)).filter((t) => t.length >= 3);
   512→  const candIds = new Set<string>();
   513→  for (const t of inputTokens) {
   514→    const posting = registry.findIdsByToken(t);
   515→    if (posting.length >= 1 && posting.length <= 2) {
   516→      for (const id of posting) candIds.add(id);
   517→    }
   518→  }
   519→  if (candIds.size === 1) {
   520→    const e = registry.byId([...candIds][0]);
   521→    if (e) return toMatch(e, 0.5, "token");
   522→  }
   523→  return null;
   524→}

   531→export interface NormalizedAreaName {
   532→  /** نام نرمال‌شده (در صورت تطبیق: نام رسمی رجیستری؛ وگرنه نام پاک‌شده ورودی) */
   533→  normalized: string;

   535→  matchedName?: string;
   536→  // ---- افزوده‌های گزارش‌دهی (مجاز طبق دستور) ----

   546→/**
   547→ * نرمال‌سازی نام محله/منطقه — طبق قواعد normalizeName، سپس تطبیق با رجیستری ۹۲۶ محله.
   548→ * اگر تطبیق یافت نشد: matchedId = undefined (لاگ/شمارش در گزارش خط لوله).

   550→export function normalizeAreaName(raw: string, registry: LocationRegistry): NormalizedAreaName {
   551→  const normalized = normalizeName(raw);
   552→  const match = findLocationInRegistry(raw, registry);
   553→  if (!match) {
   554→    return { normalized, method: "none", score: 0 };
   555→  }

   557→    normalized: match.matchedName,
   558→    matchedId: match.matchedId,
   559→    matchedName: match.matchedName,
   560→    method: match.method,
   561→    score: match.score,
   562→    matchedKind: match.kind,
   563→    ...(match.kind === "area"
   564→      ? { areaId: match.areaId, areaName: match.areaName }
   565→      : { areaId: match.areaId, areaName: match.areaName, lineId: match.matchedId, lineName: match.matchedName }),
   566→  };
   567→}

   570→// ۱۱) dedupKey — کلید یکتای مفهومی هر موجودیت (عین B.1-۷ + §۲.۱/§۳.۷ طراحی)
   571→// ==========================================================================
   572→
   573→function str(v: unknown): string {
   574→  return typeof v === "string" ? v : v === null || v === undefined ? "" : String(v);
   575→}
   576→
   577→function isoDateOrNull(v: unknown): string {
   578→  if (v instanceof Date) return v.toISOString();
   579→  const s = str(v);
   580→  if (!s) return "";
   581→  const d = new Date(s);
   582→  return Number.isNaN(d.getTime()) ? "" : d.toISOString();
   583→}

   585→/**
   586→ * کلید یکتای رکورد — عین قواعد B.1-۷ دستور:
   587→ *  transaction: contractHolo اگر موجود، وگرنه propertyCode + contractDate
   588→ *  expert: personnelCode · listing: propertyCode · request: requestCode
   589→ *  visit: visitCode · contract: contractHolo
   590→ *  interaction: requestCode + requestRealtorName + propertyRealtorName
   591→ *
   592→ * نقشه صادقانه (عین کشف §۱.۱۱ P6.5-03): «visitCode» در داده واقعی ARM وجود ندارد —
   593→ * معادل پایدار آن شناسه ردیف (id) است؛ «requestCode» تعاملات هم در ساختار واقعی
   594→ * realtor-interactions نیست — کلید واقعی جفت نام مشاوران است (requestCode در صورت
   595→ * وجود در رکورد، پیشوند می‌شود). این کلید «مفهومی» فقط برای گزارش/شمارش/تست است؛
   596→ * خود upsert روی کلید فیزیکی Prisma عمل می‌کند (§۳.۷ طراحی).
   597→ */
   598→export function dedupKey(entity: BulkEntity, record: Record<string, unknown>): string {
   599→  switch (entity) {
   600→    case "transaction": {
   601→      const holo = normalizeDigitsOnly(str(record.contractHolo));
   602→      if (holo) return `transaction:${holo}`;
   603→      const code = normalizePropertyCode(str(record.propertyCode));
   604→      const date = isoDateOrNull(record.contractDate);
   605→      return `transaction:${code}:${date}`;
   606→    }
   607→    case "expert":
   608→      return `expert:${normalizeDigitsOnly(str(record.personnelCode))}`;
   609→    case "listing":
   610→      return `listing:${normalizePropertyCode(str(record.propertyCode))}`;
   611→    case "request":
   612→      return `request:${normalizeRequestCode(str(record.requestCode))}`;
   613→    case "visit":
   614→      return `visit:${normalizeDigitsOnly(str(record.visitCode ?? record.id))}`;
   615→    case "contract": {
   616→      const holo = normalizeDigitsOnly(str(record.contractHolo));
   617→      return holo ? `contract:${holo}` : `contract:${normalizeDigitsOnly(str(record.id))}`;
   618→    }
   619→    case "interaction": {
   620→      const rc = normalizeRequestCode(str(record.requestCode));
   621→      const a = normalizeName(str(record.requestRealtorName));
   622→      const b = normalizeName(str(record.propertyRealtorName));
   623→      return `interaction:${rc}:${a}:${b}`;
   624→    }
   625→  }
   626→}

   631→
   632→/**
   633→ * merge رکورد جدید با رکورد موجود — فیلدهای غیر-null جدید جایگزین می‌شوند؛
   634→ * فیلدهای null جدید، مقادیر قدیمی را پاک نمی‌کنند («جدیدترین داده غیرتهی برنده»).
   635→ * هم‌ارز خالصِ mergeNonNullUpdate در لایه Repository (منبع مرجع تست‌ها و خط لوله اعتبارسنجی).
   636→ */
   637→export function mergeRecords<T extends Record<string, unknown>>(existing: T, incoming: Partial<T>): T {
   638→  const out: Record<string, unknown> = { ...existing };
   639→  for (const [k, v] of Object.entries(incoming)) {
   640→    if (v !== null && v !== undefined) out[k] = v;
   641→  }
   642→  return out as T;
   643→}

   704→/** آمار نرمال‌سازی یک ردیف */
   705→export interface RowNormStats {
   706→  /** فیلدهایی که نرمال‌سازی مقدارشان را عوض کرد */
   707→  changedFields: string[];
   708→  /** فیلدهایی که ورودی معنادار داشت اما خروجی نامعتبر شد (رد صادقانه) */
   709→  invalidFields: string[];
   710→  /** تطبیق محله موفق (فقط listing) */
   711→  areaMatched: boolean;
   712→  /** نام محله/خط موجود ولی بدون تطبیق (فقط listing) */
   713→  areaUnmatched: boolean;
   714→  /** کلید یکتای مفهومی */
   715→  key: string;
   716→}
   717→
   718→/**
   719→ * نرمال‌سازی یک ردیف نگاشت‌شده خط لوله — خالص و idempotent
   720→ * (نرمال‌سازی مجدد ردیف نرمال‌شده = همان ردیف؛ اثبات‌شده در تست deterministic).
   721→ * فیلدهای متناسب هر موجودیت + تطبیق رجیستری محله برای listing (areaName→areaId، lineName→lineId).
   722→ */
   723→export function normalizeBulkRow<T extends Record<string, unknown>>(
   724→  entity: BulkEntity,
   725→  row: T,
   726→  registry: LocationRegistry
   727→): { row: T; stats: RowNormStats } {
   728→  const out: Record<string, unknown> = { ...row };
   729→  const changedFields: string[] = [];
   730→  const invalidFields: string[] = [];
   731→  let areaMatched = false;
   732→  let areaUnmatched = false;
   733→
   734→  for (const rule of FIELD_RULES[entity]) {
   735→    const raw = out[rule.field];
   736→    if (typeof raw !== "string" || raw.length === 0) continue; // فقط رشته‌های غیرخالی
   737→    let normalized: string;
   738→
   739→    switch (rule.kind) {
   740→      case "name":
   741→        normalized = normalizeName(raw);
   742→        break;
   743→      case "propertyCode":
   744→        normalized = normalizePropertyCode(raw);
   745→        break;
   746→      case "requestCode":
   747→        normalized = normalizeRequestCode(raw);
   748→        break;
   749→      case "mobile":
   750→        normalized = normalizeMobile(raw);
   751→        break;
   752→      case "holo":
   753→        normalized = normalizeDigitsOnly(raw);
   754→        break;
   755→      case "areaName":
   756→      case "lineName": {
   757→        const match = findLocationInRegistry(raw, registry);
   758→        if (match) {
   759→          areaMatched = true;
   760→          if (rule.kind === "areaName") {
   761→            // فیلد منطقه → هویت منطقه (چه مدخل خط باشد چه ناحیه، areaId/areaName والد معتبر است)
   762→            if (out.areaId !== match.areaId) changedFields.push("areaId");
   763→            out.areaId = match.areaId;
   764→            if (out.areaName !== match.areaName) changedFields.push("areaName");
   765→            out.areaName = match.areaName;
   766→            normalized = match.areaName;
   767→          } else if (match.kind === "line") {
   768→            // فیلد محله → هویت محله (خط)
   769→            if (out.lineId !== match.matchedId) changedFields.push("lineId");
   770→            out.lineId = match.matchedId;
   771→            normalized = match.matchedName;
   772→          } else {
   773→            // نام «منطقه» در فیلد محله → اطلاعات منطقه پر می‌شود؛ lineId نمی‌گیرد و نام فیلد پاک‌شده می‌ماند
   774→            if (!out.areaId) {
   775→              out.areaId = match.areaId;
   776→              out.areaName = match.areaName;
   777→              changedFields.push("areaId", "areaName");
   778→            }
   779→            normalized = normalizeName(raw);
   780→          }
   781→        } else {
   782→          if (raw.trim().length > 0) areaUnmatched = true;
   783→          normalized = normalizeName(raw);
   784→        }
   785→        if (out[rule.field] !== normalized) {
   786→          if (!changedFields.includes(rule.field)) changedFields.push(rule.field);
   787→          out[rule.field] = normalized;
   788→        }
   789→        if (normalized.length === 0 && raw.trim().length > 0) {
   790→          invalidFields.push(rule.field);
   791→        }
   792→        continue;
   793→      }
   794→    }
   795→
   796→    if (normalized !== raw) {
   797→      changedFields.push(rule.field);
   798→      out[rule.field] = normalized;
   799→    }
   800→    // رد صادقانه: ورودی غیرخالی → خروجی خالی (موبایل نامعتبر و…)
   801→    if (normalized.length === 0 && raw.trim().length > 0) {
   802→      invalidFields.push(rule.field);
   803→    }
   804→  }
   805→
   806→  return {
   807→    row: out as T,
   808→    stats: {
   809→      changedFields,
   810→      invalidFields,
   811→      areaMatched,
   812→      areaUnmatched,
   813→      key: dedupKey(entity, out),
   814→    },
   815→  };
   816→}
