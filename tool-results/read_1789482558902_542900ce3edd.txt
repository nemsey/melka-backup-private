   102→      hit,
   103→      evidence: hit ? "در متن آگهی یافت شد" : "یافت نشد",

   175→export interface UnderpriceResult {
   176→  score: number; // 0..100
   177→  percentBelow: number | null; // درصد زیر بودن قیمت هر متر نسبت به مرجع
   178→  referencePerMeter: number | null; // میانگین مرجع (تومان بر متر)
   179→  sampleCount: number;
   180→  confidence: "high" | "medium" | "low";
   181→  method: "divar+arm" | "divar" | "arm" | "none";
   182→  reason: string;
   183→}
   184→
   185→export interface MarketRef {
   186→  medianPerMeter: number | null;
   187→  sampleCount: number;
   188→  source: "divar" | "arm" | "blend";
   189→}
   190→
   191→/**
   192→ * امتیاز زیرقیمت — شاهکار ایجنت:
   193→ * درصد زیر بودن «قیمت هر متر» آگهی نسبت به میانگین آگهی‌های مشابه همان محله
   194→ * (ترکیب وزن‌دار با قیمت‌های سامانه آرم در صورت وجود).
   195→ * نرمال‌سازی: ~۰٪ زیر = امتیاز ۵۰؛ ۳۰٪+ زیر = امتیاز ۱۰۰؛ ۱۵٪+ بالاتر = ۰
   196→ */
   197→export function scoreUnderprice(input: {
   198→  pricePerMeter: number | null;
   199→  ref: MarketRef | null;
   200→  armRefPerMeter?: number | null;
   201→  armWeight?: number;
   202→  minSample?: number;
   203→}): UnderpriceResult {
   204→  const minSample = input.minSample ?? 4;
   205→  const armWeight = Math.min(Math.max(input.armWeight ?? 0.35, 0), 0.9);
   206→  if (!input.pricePerMeter || !(input.pricePerMeter > 0)) {
   207→    return { score: 0, percentBelow: null, referencePerMeter: null, sampleCount: 0, confidence: "low", method: "none", reason: "قیمت هر متر برای این آگهی قابل استخراج نیست." };
   208→  }
   209→  const divarRef = (input.ref?.sampleCount ?? 0) >= minSample ? input.ref?.medianPerMeter ?? null : null;
   210→  const armRef = input.armRefPerMeter && input.armRefPerMeter > 0 ? input.armRefPerMeter : null;
   211→
   212→  let reference: number | null = null;
   213→  let method: UnderpriceResult["method"] = "none";
   214→  if (divarRef && armRef) {
   215→    reference = divarRef * (1 - armWeight) + armRef * armWeight;
   216→    method = "divar+arm";
   217→  } else if (divarRef) {
   218→    reference = divarRef;
   219→    method = "divar";
   220→  } else if (armRef) {
   221→    reference = armRef;
   222→    method = "arm";
   223→  }
   224→  if (!reference || reference <= 0) {
   225→    return {
   226→      score: 0,
   227→      percentBelow: null,
   228→      referencePerMeter: null,
   229→      sampleCount: input.ref?.sampleCount ?? 0,
   230→      confidence: "low",
   231→      method: "none",
   232→      reason: `آمار کافی برای محله این آگهی در دسترس نیست (حداقل ${minSample} آگهی مشابه لازم است).`,
   233→    };
   234→  }
   235→
   236→  const percentBelow = ((reference - input.pricePerMeter) / reference) * 100;
   237→  const sampleCount = input.ref?.sampleCount ?? 0;
   238→  // نرمال‌سازی درصد به امتیاز
   239→  let score: number;
   240→  if (percentBelow <= -15) score = 0;
   241→  else if (percentBelow >= 30) score = 100;
   242→  else score = 50 + (percentBelow / 30) * 50 * (percentBelow > 0 ? 1.2 : 0.9);
   243→  score = Math.max(0, Math.min(100, score));
   244→
   245→  const confidence: "high" | "medium" | "low" =
   246→    method === "divar+arm" ? "high" : sampleCount >= minSample * 2 || method === "arm" ? "medium" : "low";
   247→
   248→  const fa = (n: number) => Math.round(n).toLocaleString("en-US").replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[Number(d)]);
   249→  const reason =
   250→    percentBelow >= 0
   251→      ? `قیمت هر متر این آگهی حدود ${fa(percentBelow)}٪ زیر میانگین مرجع (${fa(reference)} تومان بر متر، ${fa(sampleCount)} نمونه، روش ${method === "divar+arm" ? "ترکیبی دیوار+آرم" : method === "divar" ? "آگهی‌های مشابه دیوار" : "سامانه آرم"}) است.`
   252→      : `قیمت هر متر این آگهی حدود ${fa(-percentBelow)}٪ بالاتر از میانگین مرجع است — فایل گران محسوب می‌شود.`;
   253→
   254→  return {
   255→    score: Math.round(score),
   256→    percentBelow: Math.round(percentBelow * 10) / 10,
   257→    referencePerMeter: Math.round(reference),
   258→    sampleCount,
   259→    confidence,
   260→    method,
   261→    reason,
   262→  };
   263→}

   270→export function scoreLocation(input: {

   346→  /** زمان واقعی انتشار آگهی — برای فاکتور تازگی */
   347→  publishedAt?: Date | null;
   348→  now?: Date;
   349→  /** نتیجه تطبیق مکان با رجیستری آرم (برای فاکتور دقت مکان) */
   350→  location?: { score: number; reason?: string } | null;
   351→}): OtherFactorsResult {
   352→  const text = `${input.title} ${input.description ?? ""}`.replace(/\u200c/g, " ");
   353→  const has = (w: string) => text.includes(w.replace(/\u200c/g, " "));
   354→  const factors: OtherFactor[] = [];
   355→
   356→  // ---------- ۱) فوریت و انگیزه فروش‌دهنده ----------
   357→  {
   358→    let s = 35; // پایه خنثی: فروش‌نده معمولی
   359→    const strong = URGENCY_STRONG.filter(has);
   360→    const medium = URGENCY_MEDIUM.filter(has);
   361→    const soft = URGENCY_SOFT.filter(has);
   362→    s += strong.length * 22 + medium.length * 12 + soft.length * 6;
   363→    s = Math.min(100, s);
   364→    const ev = strong.length || medium.length || soft.length
   365→      ? [...strong.slice(0, 2), ...medium.slice(0, 2), ...soft.slice(0, 1)].map((w) => `«${w}»`).join("، ")
   366→      : "هیچ نشانه فوریت/انگیزه در متن نیست — فروش‌نده آرام و بدون فشار زمانی";
   367→    factors.push({ key: "urgency", label: "فوریت و انگیزه فروش‌دهنده", score: s, weight: 25, evidence: ev });
   368→  }
   369→
   370→  // ---------- ۲) تازگی انتشار ----------
   371→  {
   372→    const now = input.now ?? new Date();
   373→    let s: number;
   374→    let ev: string;
   375→    if (!input.publishedAt) {
   376→      s = 0;
   377→      ev = "زمان انتشار واقعی در دسترس نیست.";
   378→    } else {
   379→      const hours = Math.max(0, (now.getTime() - input.publishedAt.getTime()) / 3600_000);
   380→      if (hours <= 2) { s = 100; ev = "تازه‌ترین وضعیت — کمتر از ۲ ساعت از انتشار گذشته (فرصت نخستین تماس)"; }
   381→      else if (hours <= 6) { s = 88; ev = `حدود ${Math.round(hours)} ساعت از انتشار گذشته — هنوز زود است.`; }
   382→      else if (hours <= 12) { s = 75; ev = `حدود ${Math.round(hours)} ساعت از انتشار گذشته.`; }
   383→      else if (hours <= 24) { s = 60; ev = "حدود یک روز از انتشار گذشته — رقبا احتمالاً دیده‌اند."; }
   384→      else if (hours <= 48) { s = 40; ev = `دو روز از انتشار گذشته (${Math.round(hours)} ساعت).`; }
   385→      else if (hours <= 72) { s = 22; ev = "سه روز از انتشار گذشته — فرصت شکار رو به کاهش."; }
   386→      else { s = 10; ev = `بیش از ${Math.floor(hours / 24)} روز از انتشار گذشته — فایل کهنه است.`; }
   387→    }
   388→    factors.push({ key: "freshness", label: "تازگی انتشار", score: s, weight: 15, evidence: ev });
   389→  }
   390→
   391→  // ---------- ۳) تکمیل و شفافیت داده‌ها ----------
   392→  {
   393→    // ساختار داده‌ها: قیمت، هرمتر، متراژ، اتاق، طبقه، سال ساخت، تصاویر
   394→    const parts: { ok: boolean; w: number; name: string }[] = [
   395→      { ok: input.priceTotal != null, w: 15, name: "قیمت کل" },
   396→      { ok: input.pricePerMeter != null, w: 10, name: "قیمت هر متر" },
   397→      { ok: input.meterage != null, w: 15, name: "متراژ" },
   398→      { ok: input.rooms != null, w: 10, name: "اتاق" },
   399→      { ok: input.floor != null, w: 10, name: "طبقه" },
   400→      { ok: input.buildYear != null, w: 15, name: "سال ساخت" },
   401→    ];
   402→    const pc = input.photoCount ?? 0;
   403→    const photoScore = pc >= 8 ? 25 : pc >= 3 ? 20 : pc >= 1 ? 10 : 0;
   404→    let s = parts.reduce((acc, p) => acc + (p.ok ? p.w : 0), 0) + photoScore;
   405→    // نکته‌سنجی: قیمت «توافقی» شفاف است اما ارزش‌گذاری را ناممکن می‌کند
   406→    if (input.priceTotal == null && (has("توافقی") || has("تمدید"))) s = Math.min(s, 55);
   407→    s = Math.min(100, s);
   408→    const missing = parts.filter((p) => !p.ok).map((p) => p.name);
   409→    const ev = `${pc >= 3 ? `${pc} تصویر واقعی` : pc >= 1 ? "فقط " + pc + " تصویر" : "بدون تصویر"}${missing.length ? "؛ داده غایب: " + missing.join("، ") : "؛ داده‌ها کامل است"}`;
   410→    factors.push({ key: "completeness", label: "تکمیل و شفافیت داده‌ها", score: s, weight: 20, evidence: ev });
   411→  }
   412→
   413→  // ---------- ۴) کیفیت و اصالت محتوا ----------
   414→  {
   415→    const desc = (input.description ?? "").replace(/\u200c/g, " ").trim();
   416→    const len = desc.length;
   417→    let s: number;
   418→    if (len >= 500) s = 85;
   419→    else if (len >= 150) s = 70;
   420→    else if (len >= 40) s = 45;
   421→    else s = 15;
   422→    // نکته‌سنجی: توضیحات دارای عدد (متراژ/قیمت/طبقه واقعی) = مستند، نه شعار
   423→    if (/\d/.test(desc)) s += 6;
   424→    // نکته‌سنجی: صداقت — اعلام عیب، نشانه مالک واقعی و قیمت واقعی
   425→    const honest = HONESTY_WORDS.filter(has);
   426→    s += Math.min(14, honest.length * 7);
   427→    // نکته‌سنجی: ادبیات تبلیغاتی توخالی = کپی‌پیس، اعتبار محتوا را می‌خورد
   428→    const noise = MARKETING_NOISE.filter(has);
   429→    s -= Math.min(15, noise.length * 6);
   430→    s = Math.max(0, Math.min(100, s));
   431→    const evBits: string[] = [];
   432→    evBits.push(len >= 150 ? `توضیحات مستند (${len} نویسه)` : len >= 40 ? "توضیحات کوتاه" : "توضیحات تقریباً خالی");
   433→    if (honest.length) evBits.push(`صداقت در عیوب: ${honest.slice(0, 2).map((w) => `«${w}»`).join("، ")}`);
   434→    if (noise.length) evBits.push(`ادبیات تبلیغاتی: ${noise.slice(0, 2).map((w) => `«${w}»`).join("، ")}`);
   435→    factors.push({ key: "content", label: "کیفیت و اصالت محتوا", score: s, weight: 15, evidence: evBits.join("؛ ") });
   436→  }
   437→
   438→  // ---------- ۵) آمادگی معامله ----------
   439→  {
   440→    let s = 30; // پایه: معامله‌پذیری معمولی
   441→    const deed = READINESS_DEED.filter(has);
   442→    const empty = READINESS_EMPTY.filter(has);
   443→    const flexible = URGENCY_SOFT.filter(has);
   444→    s += Math.min(40, deed.length * 22);
   445→    s += Math.min(25, empty.length * 15);
   446→    s += Math.min(15, flexible.length * 8);
   447→    if (has("معاوضه")) s -= 10; // معاوضه: پیچیدگی معامله نقدی
   448→    if (input.priceTotal != null) s += 5; // قیمت قطعی = شفافیت
   449→    s = Math.max(0, Math.min(100, s));
   450→    const evBits: string[] = [];
   451→    if (deed.length) evBits.push(`اسناد: ${deed.slice(0, 2).map((w) => `«${w}»`).join("، ")}`);
   452→    if (empty.length) evBits.push("ملک خالی/قابل تحویل");
   453→    if (flexible.length) evBits.push("قیمت منعطف");
   454→    if (!evBits.length) evBits.push("نشانه خاصی از آمادگی معامله در متن نیست");
   455→    factors.push({ key: "readiness", label: "آمادگی معامله", score: s, weight: 15, evidence: evBits.join("؛ ") });
   456→  }
   457→
   458→  // ---------- ۶) دقت مکان ----------
   459→  {
   460→    const loc = input.location ?? null;
   461→    const s = loc ? Math.max(0, Math.min(100, loc.score)) : 0;
   462→    const ev = loc
   463→      ? loc.score >= 70 ? "نگاشت مطمئن به محله آرم (متنی + مختصات) — قیاس بازار قابل‌اعتماد" : "نگاشت مکانی با اطمینان متوسط"
   464→      : "بدون نشانه مکانی قابل‌اتکا";
   465→    factors.push({ key: "location", label: "دقت مکان", score: s, weight: 10, evidence: ev });
   466→  }
   467→
   468→  // ---------- ترکیب نهایی ----------
   469→  const wsum = factors.reduce((acc, f) => acc + f.weight, 0) || 100;
   470→  const score = Math.round(factors.reduce((acc, f) => acc + f.score * f.weight, 0) / wsum);
   471→  const top = [...factors].sort((a, b) => b.score * b.weight - a.score * a.weight)[0];
   472→  const reason =
   473→    top && top.score >= 65
   474→      ? `قوی‌ترین فاکتور: «${top.label}» (${top.evidence})`
   475→      : "هیچ فاکتور بهینه‌سازی برجسته‌ای در این فایل دیده نمی‌شود — فایل متعارف است.";
   476→  return { score: Math.max(0, Math.min(100, score)), factors, reason };
   477→}
   478→
   479→// ===================== قانون مالک: رد کامل فایل‌های غیرشخصی =====================
   480→
   481→/**
   482→ * آیا این آگهی «غیرشخصی» است و باید کاملاً رد شود؟ (قانون ۱ و ۲ مالک)
   483→ * نکته‌سنجی: برچسب رسمی دیوار زیر نقشه قطعی‌ترین منبع است؛
   484→ * برچسب «شخصی» + سیگنال متنی قوی شخصی، بر نوفه متنی ضعیف غلبه می‌کند.
   485→ */
   486→export function shouldAutoReject(personal: PersonalScoreResult, publisherType?: string | null): boolean {
   487→  const pt = publisherType ?? null;
   488→  if (pt === "agent" || pt === "department" || pt === "business") return true;
   489→  if (pt === "personal" && personal.score >= 65) return false; // برچسب رسمی شخصی + متن قوی
   490→  return personal.isBroker;
   491→}

   493→// ===================== امتیاز نهایی و دسته‌بندی =====================
   494→
   495→export interface TotalScoreInput {
   496→  personal: number;
   497→  underprice: number;
   498→  other: number;
   499→  location: number;
   500→  settings: Pick<DivarSettings, "weightPersonal" | "weightUnderprice" | "weightOther" | "weightLocation" | "huntThreshold" | "followThreshold">;
   501→}
   502→
   503→export type AdBucket = "hunt" | "follow" | "review";
   504→
   505→export interface TotalScoreResult {
   506→  total: number;
   507→  bucket: AdBucket;
   508→  breakdown: { personal: number; underprice: number; other: number; location: number };
   509→}
   510→
   511→/**
   512→ * امتیاز نهایی ۱ تا ۱۰۰ — فقط فایل‌های شخصی به اینجا می‌رسند (غیرشخصی‌ها کامل رد شده‌اند).
   513→ * پیش‌فرض مالک (نسخه ۳): ۵۰ شخصی‌بودن + ۲۰ زیرقیمت + ۲۰ سایر فاکتورهای بهینه + ۱۰ مکان (قابل تنظیم).
   514→ */
   515→export function computeTotalScore(input: TotalScoreInput): TotalScoreResult {
   516→  const wp = Math.max(0, input.settings.weightPersonal);
   517→  const wu = Math.max(0, input.settings.weightUnderprice);
   518→  const wo = Math.max(0, input.settings.weightOther);
   519→  const wl = Math.max(0, input.settings.weightLocation);
   520→  const sum = wp + wu + wo + wl;
   521→  const wsum = sum > 0 ? sum : 100;
   522→  const total = Math.round(
   523→    (input.personal * wp + input.underprice * wu + input.other * wo + input.location * wl) / wsum
   524→  );
   525→  const hunt = Math.max(1, Math.min(99, input.settings.huntThreshold));
   526→  const follow = Math.max(1, Math.min(hunt - 1, input.settings.followThreshold));
   527→  const bucket: AdBucket = total > hunt ? "hunt" : total >= follow ? "follow" : "review";
   528→  return {
   529→    total: Math.max(1, Math.min(100, total)),
   530→    bucket,
   531→    breakdown: { personal: input.personal, underprice: input.underprice, other: input.other, location: input.location },
   532→  };
   533→}
   534→
   535→/** پیش‌بینی هوشمند دلایل احتمالی رد آگهی توسط کاربر (کلید جزییات) — فقط فایل‌های شخصی */
   536→export function predictRejectReasons(input: {
   537→  personal: PersonalScoreResult;
   538→  underprice: UnderpriceResult;
   539→  location: LocationMatchResult;
   540→  /** تحلیل سایر فاکتورهای بهینه (شاهکار ایجنت) */
   541→  other?: OtherFactorsResult | null;
   542→  pricePerMeter: number | null;
   543→}): { key: string; title: string; likely: number; hint: string }[] {
   544→  const out: { key: string; title: string; likely: number; hint: string }[] = [];
   545→  // غیرشخصی‌ها هرگز به فهرست نمی‌رسند؛ این بررسی صرفاً حاشیه‌ای است (فایل مرزی)
   546→  if (input.personal.isBroker || input.personal.score < 35) {
   547→    out.push({ key: "broker", title: "شخصی‌بودن فایل قطعی نیست", likely: 70, hint: "امتیاز شخصی‌بودن پایین است" });
   548→  }
   549→  if (input.personal.score >= 35 && input.personal.score < 60) {
   550→    out.push({ key: "semi-personal", title: "شخصی‌بودن آگهی قطعی نیست", likely: 55, hint: "نشانه‌های متناقص در متن آگهی" });
   551→  }
   552→  if (input.pricePerMeter == null) {
   553→    out.push({ key: "no-price", title: "قیمت آگهی درج نشده / قابل استخراج نیست", likely: 70, hint: "قیمت هر متر نامشخص" });
   554→  } else if (input.underprice.percentBelow != null && input.underprice.percentBelow < 0) {
   555→    out.push({ key: "overpriced", title: "قیمت بالاتر از بازار است", likely: 75, hint: "زیرقیمت نیست" });
   556→  }
   557→  if (input.underprice.percentBelow != null && input.underprice.percentBelow < 5 && input.pricePerMeter != null) {
   558→    out.push({ key: "not-underpriced", title: "قیمت معمولی است — شکار نمی‌ارزد", likely: 45, hint: "تخفیف معنادار ندارد" });
   559→  }
   560→  if (input.location.score < 40) {
   561→    out.push({ key: "location", title: "موقعیت مکانی نامشخص/خارج از محدوده", likely: 60, hint: input.location.reason });
   562→  }
   563→  // شاهکار ایجنت — پیش‌بینی از روی فاکتورهای بهینه
   564→  const other = input.other ?? null;
   565→  if (other) {
   566→    const byKey = new Map(other.factors.map((f) => [f.key, f]));
   567→    const freshness = byKey.get("freshness");
   568→    if (freshness && freshness.score <= 22) {
   569→      out.push({ key: "stale", title: "فایل کهنه است — فرصت شکار گذشته", likely: 60, hint: freshness.evidence });
   570→    }
   571→    const completeness = byKey.get("completeness");
   572→    if (completeness && completeness.score <= 40) {
   573→      out.push({ key: "thin-data", title: "داده‌های فایل ناقص است (ارزیابی سخت)", likely: 55, hint: completeness.evidence });
   574→    }
   575→    const urgency = byKey.get("urgency");
   576→    if (urgency && urgency.score <= 35) {
   577→      out.push({ key: "no-motivation", title: "فروش‌نده انگیزه/فوریت ندارد", likely: 40, hint: "مذاکره تخفیف سخت می‌شود" });
   578→    }
   579→    const content = byKey.get("content");
   580→    if (content && content.score <= 30) {
   581→      out.push({ key: "weak-content", title: "توضیحات آگهی ضعیف/توخالی است", likely: 40, hint: content.evidence });
   582→    }
   583→  }
   584→  out.push({ key: "already-sold", title: "ملک قبلاً فروخته/تماس بی‌پاسخ", likely: 25, hint: "ایجنت از روی الگوی ردهای قبلی شما" });
   585→  out.push({ key: "condition", title: "وضعیت ساختمان/سند مشکل‌دار", likely: 25, hint: "نیازمند بازدید حضوری" });
   586→  out.push({ key: "other", title: "دلیل دیگر (در توضیحات بنویسید)", likely: 10, hint: "خودتان شرح دهید" });
   587→  return out;
   588→}
