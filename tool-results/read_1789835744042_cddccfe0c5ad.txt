    36→const FA_TO_EN_DIGITS: Record<string, string> = {
    37→  "۰": "0", "۱": "1", "۲": "2", "۳": "3", "۴": "4", "۵": "5", "۶": "6", "۷": "7", "۸": "8", "۹": "9",
    38→  "٠": "0", "١": "1", "٢": "2", "٣": "3", "٤": "4", "٥": "5", "٦": "6", "٧": "7", "٨": "8", "٩": "9",

    41→/** ارقام فارسی/عربی → لاتین (تعیین‌کننده — عین enDigits در jalali.ts) */
    42→export function latinDigits(input: string): string {
    43→  return input.replace(/[۰-۹٠-٩]/g, (d) => FA_TO_EN_DIGITS[d] ?? d);
    44→}

    46→/** یکسان‌سازی حروف: ي→ی، ك→ک، أ/إ/آ→ا، ة→ه + حذف اعراب/کشیده */
    47→export function unifyPersianLetters(input: string): string {
    48→  return input
    49→    .replace(/[ي]/g, "ی")
    50→    .replace(/[ك]/g, "ک")
    51→    .replace(/[آأإ]/g, "ا")
    52→    .replace(/[ة]/g, "ه")
    53→    .replace(/[\u064B-\u0652\u0670\u0640]/g, ""); // اعراب + تنوین + کشیده ( Tateel )
    54→}

    60→// ۲) normalizeName — نام اشخاص/مالکان (عین B.1 دستور)

    63→/** نشان‌های نگارشی که از نام حذف می‌شوند (عین دستور: نقطه، ویرگول، پرانتز و…) */
    64→const NAME_PUNCT_RE = /[.,()\[\]{}«»"'\u066A\u066B\u066C،؛:!?؟…\-_ـ*#+@^~`|/\\]/g;
    65→
    66→/**
    67→ * نرمال‌سازی نام (اشخاص، مالکان) — عین قواعد B.1-۱ دستور:
    68→ *  · حذف ZWNJ (U+200C)
    69→ *  · ی/ک عربی → فارسی · أ/إ/آ → ا · ة → ه
    70→ *  · حذف فاصله اضافی (→ یک فاصله) · حذف نشان‌های نگارشی و اعراب
    71→ * نمونه شاهد دستور: «محمّد  احمدی» → «محمد احمدی»
    72→ */
    73→export function normalizeName(raw: string): string {
    74→  if (typeof raw !== "string" || raw.length === 0) return "";
    75→  let s = raw.replace(CTRL_RE, "");
    76→  s = latinDigits(s);
    77→  s = unifyPersianLetters(s);
    78→  s = s.replace(/\u200C/g, ""); // ZWNJ حذف (عین دستور)
    79→  s = s.replace(NAME_PUNCT_RE, " ");
    80→  // املا محاوره‌ای پسوند پایانی در نام‌ها (شاهد §۳.۲: رضائي → رضایی):
    81→  // «ئی/ئي» پایانی = «یی» (ئ→ی)؛ ئ تنها-پایانی هم «ی» — ئ میانی (مسئول/مسئله) دست‌نخورده
    82→  s = s.replace(/ئ(?=ی(?=\s|$))/g, "ی");
    83→  s = s.replace(/ئ(?=\s|$)/g, "ی");
    84→  s = s.replace(/\s+/g, " ").trim();
    85→  return s;
    86→}

    92→/** نرمال‌سازی مشترک کد آرم: trim · ارقام→لاتین · حذف فاصله/جداکننده · حرف اول ثابت · حذف صفر ابتدا */
    93→function normalizeArmCode(raw: string): string {
    94→  if (typeof raw !== "string") return "";
    95→  let s = latinDigits(raw).replace(CTRL_RE, "");
    96→  s = s.replace(/[\s\u200C\u066C,،_\-]/g, ""); // حذف فاصله و جداکننده‌های رایج کد
    97→  if (s.length === 0) return "";
    98→  // حرف اول (P یا R یا هر پیشوند لاتین) → بزرگ؛ بقیه فقط حرف/رقم نگه داشته می‌شود
    99→  const first = s[0];
   100→  const isLatinLetter = /[a-zA-Z]/.test(first);
   101→  if (isLatinLetter) s = first.toUpperCase() + s.slice(1);
   102→  // حذف صفرهای ابتدای بخش عددی (پس از پیشوند حرفی)
   103→  const body = isLatinLetter ? s.slice(1) : s;
   104→  const bodyNoLeadingZeros = body.replace(/^0+(?=\d)/, "");
   105→  const cleanedBody = bodyNoLeadingZeros.replace(/[^0-9A-Za-z]/g, "");
   106→  if (!isLatinLetter && cleanedBody.length === 0) return "";
   107→  if (isLatinLetter && cleanedBody.length === 0) return ""; // فقط حرف بدون بدنه → نامعتبر
   108→  return isLatinLetter ? first.toUpperCase() + cleanedBody : cleanedBody;
   109→}

   131→/**
   132→ * نرمال‌سازی موبایل → قالب استاندارد ۰۹xxxxxxxxx
   133→ *  · ارقام فارسی/عربی → لاتین · حذف نویسه‌های غیررقمی
   134→ *  · حذف +۹۸/۰۰۹۸ → ۰ · افزودن ۰ به ۱۰رقمی‌های آغازشونده با ۹
   135→ *  · اعتبارسنجی نهایی ^09\d{9}$ — نامعتبر → رشته خالی (شمارش در گزارش)
   136→ */
   137→export function normalizeMobile(raw: string): string {
   138→  if (typeof raw !== "string") return "";
   139→  let d = latinDigits(raw).replace(/[^0-9]/g, "");
   140→  if (d.length === 0) return "";
   141→  if (d.startsWith("0098")) d = "0" + d.slice(4);
   142→  else if (d.startsWith("98") && d.length === 12) d = "0" + d.slice(2);
   143→  if (d.length === 10 && d.startsWith("9")) d = "0" + d;
   144→  return /^09\d{9}$/.test(d) ? d : "";
   145→}

   148→// ۵) normalizeAmount — مبلغ → ریال پایه (عین B.1-۵ و §۳.۵ طراحی)
   149→// ==========================================================================
   150→
   151→/**
   152→ * نرمال‌سازی مبلغ → عدد پایه به «ریال».
   153→ *  · ارقام فارسی/عربی → لاتین · حذف جداکننده هزارگان (٬ ، , ' و فاصله)
   154→ *  · مقیاس‌واژه‌ها: میلیون ×۱۰۰۰ · میلیارد ×۱۰⁶ · تریلیون ×۱۰⁹ (نسبت به مبنای تومان)
   155→ *  · واحد: «ریال» ×۱ · «تومان» ×۱۰ · بدون واحد → پیش‌فرض تومان ×۱۰ (نمونه دستور: «۱٬۲۰۰٬۰۰۰» → ۱۲٬۰۰۰٬۰۰۰ ریال)
   156→ *  · كسر با «/» یا «.» یا «٫» بین ارقام (۲/۵ میلیارد = ۲٫۵)
   157→ *  · ورودی number → عیناً (مقدار پایه ساخت‌یافته — بدون تفسیر واحد متنی)
   158→ *  · ورودی متنی غیرقابل‌تجزیه → NaN (پاسخ صادقانه؛ فراخواننده باید فیلد را کنار بگذارد)
   159→ */
   160→export function normalizeAmount(raw: string | number): number {
   161→  if (typeof raw === "number") return Number.isFinite(raw) ? raw : NaN;
   162→  if (typeof raw !== "string") return NaN;
   163→  let s = latinDigits(raw).replace(CTRL_RE, "").trim();
   164→  if (s.length === 0) return NaN;
   165→
   166→  // واحد پول
   167→  const hasToman = s.includes("تومان");
   168→  const hasRial = !hasToman && s.includes("ریال");
   169→  const unitFactor = hasRial ? 1 : 10; // پیش‌فرض (بدون واحد) = تومان، عین نمونه دستور
   170→
   171→  // مقیاس‌واژه‌ها
   172→  let scale = 1;
   173→  if (s.includes("تریلیون")) scale = 1e12;
   174→  else if (s.includes("میلیارد")) scale = 1e9;
   175→  else if (s.includes("میلیون")) scale = 1e6;
   176→
   177→  // استخراج بخش عددی: نگه‌داشتن ارقام + جداکننده اعشار / و .
   178→  s = s.replace(/[\u066C\u060C,'\u00A0\s]/g, ""); // جداکننده هزارگان
   179→  s = s.replace(/\u066B/g, "."); // ممیز عربی اعشار
   180→  const decimalMatch = s.match(/(\d+)([/.])(\d+)/);
   181→  if (decimalMatch) {
   182→    s = decimalMatch[1] + "." + decimalMatch[3];
   183→  } else {
   184→    s = s.replace(/[/.]/g, ""); // نقطه/ممیل بدون جفت رقم = نویسه زائد
   185→  }
   186→  const numeric = s.match(/^-?\d+(\.\d+)?/);
   187→  if (!numeric) return NaN;
   188→  const value = parseFloat(numeric[0]);
   189→  if (!Number.isFinite(value)) return NaN;
   190→  return value * scale * unitFactor;
   191→}

   193→/**
   194→ * P6.5-04-e-v2 — تجزیه عدد خام بدون مقیاس واحد (برای متراژ و اعداد ساده):
   195→ * ارقام فارسی/عربی → لاتین · حذف جداکننده هزارگان · ممیز / و ٫ و . → اعشار.
   196→ * برخلاف normalizeAmount هیچ واژه واحد/مقیاسی تفسیر نمی‌شود («۱۲۰ متر» → 120).
   197→ * غیرقابل‌تجزیه → NaN (پاسخ صادقانه).
   198→ */
   199→export function parsePersianNumber(raw: string | number): number {
   200→  if (typeof raw === "number") return Number.isFinite(raw) ? raw : NaN;
   201→  if (typeof raw !== "string") return NaN;
   202→  let s = latinDigits(raw).replace(CTRL_RE, "").trim();
   203→  if (s.length === 0) return NaN;
   204→  s = s.replace(/[\u066C\u060C,'\u00A0\s]/g, "");
   205→  s = s.replace(/\u066B/g, ".");
   206→  const decimalMatch = s.match(/(\d+)([/.])(\d+)/);
   207→  if (decimalMatch) s = decimalMatch[1] + "." + decimalMatch[3];
   208→  else s = s.replace(/[/.]/g, "");
   209→  const m = s.match(/^-?\d+(\.\d+)?/);
   210→  if (!m) return NaN;
   211→  const v = parseFloat(m[0]);
   212→  return Number.isFinite(v) ? v : NaN;
   213→}

   216→// ۶) normalizeJalaliDate — تاریخ جلالی → ISO 8601 نیمه‌شب UTC (عین B.1-۶ و §۳.۴)
   217→// ==========================================================================
   218→
   219→/**
   220→ * نرمال‌سازی تاریخ جلالی → «YYYY-MM-DDT00:00:00.000Z» (تاریخ تجاری بدون ساعت).
   221→ *  · فرم‌ها: 1400/01/15 · 1400-1-5 · ۱۴۰۰/۰۱/۱۵ (تبدیل با jalaliToGregorian موجود)
   222→ *  · سال ۱۳۰۰–۱۶۰۰ = جلالی؛ سال ≥ ۱۶۰۱ = میلادی (تحمل ورودی ISO)
   223→ *  · تاریخ نامعتبر (مثل 1400/13/01) → رشته خالی «» — هرگز مقدار ساختگی (§۳.۴ طراحی)
   224→ */
   225→export function normalizeJalaliDate(raw: string): string {
   226→  if (typeof raw !== "string") return "";
   227→  let s = latinDigits(raw).replace(CTRL_RE, "").trim();
   228→  if (s.length === 0) return "";
   229→
   230→  // ورودی ISO با ساعت → فقط بخش تاریخ
   231→  const iso = s.match(/^(\d{4})-(\d{2})-(\d{2})T/);
   232→  if (iso) s = `${iso[1]}/${iso[2]}/${iso[3]}`;
   233→
   234→  s = s.replace(/[-.٫]/g, "/");
   235→  const m = s.match(/^(\d{4})\/(\d{1,2})\/(\d{1,2})$/);
   236→  if (!m) return "";
   237→  const y = parseInt(m[1], 10);
   238→  const mo = parseInt(m[2], 10);
   239→  const d = parseInt(m[3], 10);
   240→
   241→  if (y >= 1300 && y <= 1600) {
   242→    // جلالی
   243→    if (mo < 1 || mo > 12) return "";
   244→    if (d < 1 || d > jalaliMonthLength(y, mo)) return "";
   245→    const [gy, gm, gd] = jalaliToGregorian(y, mo, d);
   246→    return new Date(Date.UTC(gy, gm - 1, gd)).toISOString();
   247→  }
   248→  if (y > 1600) {
   249→    // میلادی (تحمل ورودی)
   250→    if (mo < 1 || mo > 12) return "";
   251→    const dt = new Date(Date.UTC(y, mo - 1, d));
   252→    if (dt.getUTCMonth() !== mo - 1 || dt.getUTCDate() !== d) return ""; // ۳۰ فوریه و…
   253→    return dt.toISOString();
   254→  }
   255→  return ""; // سال خارج از بازه → قابل‌تبدیل نیست
   256→}

   263→export function tokenizeWords(text: string): string[] {
   264→  return text
   265→    .split(/[\s\u200C،؛:().!؟?()[\]{}«»"'،_\-…/\\|]+/)
   266→    .filter((t) => t.length > 1 || (t.length === 1 && /[0-9]/.test(t)));
   267→}

   269→/** بردار فراوانی واژه (Term Frequency) — الگوریتم عین termVector در base-agent؛ رقم‌های تکی حفظ می‌شوند («ولیعصر 2») */
   270→export function wordTermVector(text: string): Map<string, number> {
   271→  const vec = new Map<string, number>();
   272→  for (const tok of tokenizeWords(text)) vec.set(tok, (vec.get(tok) ?? 0) + 1);
   273→  return vec;
   274→}

   276→/** بردار نویسه‌نگرام ۱+۲+۳-گرامی از رشته بدون فاصله — مقاوم‌سازی املای محاوره‌ای */
   277→export function charGramVector(flat: string): Map<string, number> {
   278→  const vec = new Map<string, number>();
   279→  const bump = (k: string) => vec.set(k, (vec.get(k) ?? 0) + 1);
   280→  for (let i = 0; i < flat.length; i++) bump(flat[i]);
   281→  for (let i = 0; i + 1 < flat.length; i++) bump(flat.slice(i, i + 2));
   282→  for (let i = 0; i + 2 < flat.length; i++) bump(flat.slice(i, i + 3));
   283→  return vec;
   284→}

   286→/** شباهت کسینوسی دو بردار فراوانی — الگوریتم عین cosineSimilarity در base-agent */
   287→export function cosineSimilarity(a: Map<string, number>, b: Map<string, number>): number {
   288→  if (a.size === 0 || b.size === 0) return 0;
   289→  let dot = 0;
   290→  let normA = 0;
   291→  let normB = 0;
   292→  for (const v of a.values()) normA += v * v;
   293→  for (const v of b.values()) normB += v * v;
   294→  if (normA === 0 || normB === 0) return 0;
   295→  for (const [k, v] of a) {
   296→    const w = b.get(k);
   297→    if (w) dot += v * w;
   298→  }
   299→  return dot / (Math.sqrt(normA) * Math.sqrt(normB));
   300→}

   334→  tokens: Set<string>;
   335→}
   336→
   337→/** رجیستری جستجو — نمایه‌ها یک‌بار ساخته می‌شوند (lazy + cache در location-registry.ts) */
   338→export interface LocationRegistry {
   339→  entries: readonly LocationEntry[];
   340→  /** جستجو با کلید کامل یا هسته (بدون پرانتز) */
   341→  findByKey(key: string): readonly LocationEntry[];
   342→  /** توکن → شناسه مدخل‌های شامل آن (لایه توکن مشترک) */
   343→  findIdsByToken(token: string): readonly string[];
   344→  /** مدخل با شناسه */
   345→  byId(id: string): LocationEntry | undefined;
   346→}
   347→
   348→/** حذف محتوای پرانتز — «دادگستری (ایل گلی)» → «دادگستری» */
   349→function stripParens(s: string): string {
   350→  return s.replace(/[(（][^()]*[)）]/g, " ").replace(/\s+/g, " ").trim();
   351→}
   352→
   353→/** واژه‌های عمومی مکان — در لایه برداری از دو طرف کنار گذاشته می‌شوند */
   354→export const GENERIC_LOCATION_WORDS: ReadonlySet<string> = new Set([
   355→  "خیابان", "کوچه", "بلوار", "بازار", "پارک", "پل", "فلکه", "میدان",
   356→  "شهرک", "کوی", "محله", "منطقه", "جاده", "بن‌بست", "مجتمع",
   357→]);
   358→
   359→/** حذف واژه‌های عمومی از هسته — «شهرک ولیعصر» → «ولیعصر» (اگر خالی شد، عین ورودی) */
   360→function stripGenericWords(s: string): string {
   361→  const toks = tokenizeWords(s).filter((t) => !GENERIC_LOCATION_WORDS.has(t));
   362→  return toks.length > 0 ? toks.join(" ") : s;
   363→}
   364→
   365→/** کلید تطبیق دقیق: ارقام→لاتین، حروف یکسان، حذف فاصله/ZWNJ/نقطه‌گذاری، کوچک */
   366→export function locationKey(s: string): string {
   367→  let out = latinDigits(unifyPersianLetters(s))
   368→    .replace(CTRL_RE, "")
   369→    .replace(/\u200C/g, "")
   370→    .replace(/[\s«»"'()\[\]{}،؛:,.!?؟…\-_ـ*#+@^~`|/\\· ٬٬]+/g, "");
   371→  return out.toLowerCase();
   372→}
   373→
   374→/** نرمال‌سازی سبک برای توکن‌سازی لایه سوم */
   375→function lightNormalize(s: string): string {
   376→  return latinDigits(unifyPersianLetters(s)).replace(CTRL_RE, "").replace(/[\u200C]+/g, " ").trim();
   377→}
   378→
   379→/** ساخت رجیستری نمایه‌دار از مدخل‌های خام — خالص و deterministic؛
   380→ *  نمایه‌ها: کلید کامل + کلید هسته (بدون پرانتز) + توکن‌ها؛ بردارهای لایه دوم تنبل ساخته می‌شوند */
   381→export function buildLocationRegistry(rawEntries: readonly LocationEntry[]): LocationRegistry {
   382→  const indexed: IndexedEntry[] = rawEntries.map((e) => ({
   383→    ...e,
   384→    fullKey: locationKey(e.name),
   385→    coreKey: locationKey(stripParens(e.name) || e.name),
   386→    tokens: new Set(tokenizeWords(lightNormalize(e.name)).filter((t) => t.length >= 3)),
   387→  }));
   388→
   389→  const byKey = new Map<string, LocationEntry[]>();
   390→  const byIdMap = new Map<string, LocationEntry>();
   391→  const tokenIndex = new Map<string, Set<string>>();
   392→
   393→  const pushKey = (key: string, e: LocationEntry) => {
   394→    if (!key) return;
   395→    const arr = byKey.get(key);
   396→    if (arr) arr.push(e);
   397→    else byKey.set(key, [e]);
   398→  };
   399→  for (const e of indexed) {
   400→    pushKey(e.fullKey, e);
   401→    if (e.coreKey !== e.fullKey) pushKey(e.coreKey, e);
   402→    byIdMap.set(e.id, e);
   403→    for (const t of e.tokens) {
   404→      const set = tokenIndex.get(t);
   405→      if (set) set.add(e.id);
   406→      else tokenIndex.set(t, new Set([e.id]));
   407→    }
   408→  }
   409→
   410→  return {
   411→    entries: rawEntries,
   412→    findByKey: (key: string): readonly LocationEntry[] => byKey.get(key) ?? [],
   413→    findIdsByToken: (token: string): readonly string[] => [...(tokenIndex.get(token) ?? [])],
   414→    byId: (id: string): LocationEntry | undefined => byIdMap.get(id),
   415→  };
   416→}
   417→
   418→// بردارهای لایه دوم — ساخته‌شده تنبل (یک‌بار برای هر رجیستری، کش WeakMap)
   419→interface EntryVector {
   420→  words: Map<string, number>;
   421→  grams: Map<string, number>;
   422→  core: string;
   423→  kind: "area" | "line";
   424→}
   425→const REGISTRY_VECS = new WeakMap<LocationRegistry, Map<string, EntryVector>>();
   426→
   427→/** بردارهای مدخل‌های رجیستری — تنبل + کش‌شده؛ برای رجیستری‌های ساخته‌شده با buildLocationRegistry و رجیستری‌های تزریقی تست */
   428→function registryVectors(registry: LocationRegistry): Map<string, EntryVector> {
   429→  let m = REGISTRY_VECS.get(registry);
   430→  if (m) return m;
   431→  m = new Map<string, EntryVector>();
   432→  for (const e of registry.entries) {
   433→    const core = stripGenericWords(lightNormalize(stripParens(e.name) || e.name));
   434→    m.set(e.id, { words: wordTermVector(core), grams: charGramVector(locationKey(core)), core, kind: e.kind });
   435→  }
   436→  REGISTRY_VECS.set(registry, m);
   437→  return m;
   438→}

   440→// ==========================================================================
   441→// ۹) findLocationInRegistry — تطبیق سه‌لایه (عین B.1-۲ دستور)
   442→// ==========================================================================
   443→
   444→/** آستانه لایه برداری — عین دستور: score ≥ ۰٫۷ */
   445→export const AREA_MATCH_THRESHOLD = 0.7;
   446→
   447→function toMatch(e: LocationEntry, score: number, method: LocationMatch["method"]): LocationMatch {
   448→  return {
   449→    matchedId: e.id,
   450→    matchedName: e.name,
   451→    kind: e.kind,
   452→    areaId: e.areaId,
   453→    areaName: e.areaName,
   454→    score,
   455→    method,
   456→  };
   457→}
   458→
   459→function pickUniqueByKey(cands: readonly LocationEntry[]): LocationEntry | null {
   460→  const areas = cands.filter((c) => c.kind === "area");
   461→  if (areas.length === 1) return areas[0];
   462→  const lines = cands.filter((c) => c.kind === "line");
   463→  if (areas.length === 0 && lines.length === 1) return lines[0];
   464→  return null; // چند ناحیه یا چند خط هم‌نام → مبهم → عدم تطبیق (fail-closed)
   465→}
   466→
   467→/**
   468→ * تطبیق نام محله/منطقه در رجیستری — سه لایه عین دستور:
   469→ *  ۱) تطبیق دقیق (کلید کامل، سپس کلید هسته بدون پرانتز) — برخورد هم‌نام: ترجیح ناحیه
   470→ *  ۲) تطبیق برداری: max(کسینوس واژه‌ای، کسینوس نویسه‌نگرام) روی هسته بدون واژه عمومی — آستانه ۰٫۷
   471→ *     تساوی هم‌نوع (ناحیه-ناحیه / خط-خط) → مبهم؛ تساوی ناحیه-خط → ترجیح ناحیه
   472→ *  ۳) توکن متمایز مشترک (posting یکتا یا دوگانه که به یک مدخل می‌رسد)
   473→ * بدون تطبیق → null (صفر تطبیق کاذب — نمونه‌های ناموجود راستی‌آزمایی شده)
   474→ */
   475→export function findLocationInRegistry(raw: string, registry: LocationRegistry): LocationMatch | null {
   476→  if (typeof raw !== "string" || raw.trim().length === 0) return null;
   477→  const fullKey = locationKey(raw);
   478→  if (!fullKey) return null;
   479→
   480→  // ---- لایه ۱: تطبیق دقیق ----
   481→  const coreKey = locationKey(stripParens(raw) || raw);
   482→  for (const key of fullKey === coreKey ? [fullKey] : [fullKey, coreKey]) {
   483→    const cands = registry.findByKey(key);
   484→    if (cands.length === 1) return toMatch(cands[0], 1, "exact");
   485→    if (cands.length > 1) {
   486→      const picked = pickUniqueByKey(cands);
   487→      if (picked) return toMatch(picked, 1, "exact+pref");
   488→    }
   489→  }
   490→
   491→  // ---- لایه ۲: تطبیق برداری ----
   492→  const inputCore = stripGenericWords(lightNormalize(stripParens(raw) || raw));
   493→  const wv = wordTermVector(inputCore);
   494→  const gv = charGramVector(locationKey(inputCore));
   495→  const vecs = registryVectors(registry);
   496→  let bestEntry: LocationEntry | null = null;
   497→  let bestScore = 0;
   498→  let bestKind: "area" | "line" | null = null;
   499→  let secondScore = -1;
   500→  let secondKind: "area" | "line" | null = null;
   501→  for (const e of registry.entries) {
   502→    const v = vecs.get(e.id);
   503→    if (!v) continue;
   504→    const score = Math.max(cosineSimilarity(wv, v.words), cosineSimilarity(gv, v.grams));
   505→    if (bestEntry === null) {
   506→      bestEntry = e;
   507→      bestScore = score;
   508→      bestKind = v.kind;
   509→      continue;
   510→    }
   511→    if (score > bestScore + 1e-9) {
   512→      // برترین جدید — برترین قبلی به دومین تبدیل می‌شود
   513→      secondScore = bestScore;
   514→      secondKind = bestKind;
   515→      bestEntry = e;
   516→      bestScore = score;
   517→      bestKind = v.kind;
   518→    } else if (Math.abs(score - bestScore) <= 1e-9) {
   519→      if (v.kind === "area" && bestKind === "line") {
   520→        // تساوی ناحیه-خط → ناحیه جایگزین؛ خط به دومین تبدیل می‌شود
   521→        secondScore = bestScore;
   522→        secondKind = bestKind;
   523→        bestEntry = e;
   524→        bestKind = v.kind;
   525→      } else {
   526→        // هم‌امتیاز هم‌نوع (یا خط هم‌امتیاز با ناحیه برتر) → کاندید دوم
   527→        secondScore = score;
   528→        secondKind = v.kind;
   529→      }
   530→    } else if (score > secondScore) {
   531→      secondScore = score;
   532→      secondKind = v.kind;
   533→    }
   534→  }
   535→  if (bestEntry && bestKind && bestScore >= AREA_MATCH_THRESHOLD) {
   536→    // تساوی دقیق هم‌نوع (ناحیه-ناحیه / خط-خط) → مبهم (fail-closed)؛ ناحیه-خط هم‌امتیاز → ناحیه (بالا اعمال شد)
   537→    if (secondKind !== null && secondKind === bestKind && Math.abs(secondScore - bestScore) <= 1e-9) {
   538→      return null;
   539→    }
   540→    return toMatch(bestEntry, bestScore, "vector");
   541→  }
   542→
   543→  // ---- لایه ۳: توکن متمایز مشترک ----
   544→  const inputTokens = tokenizeWords(lightNormalize(raw)).filter((t) => t.length >= 3);
   545→  const candIds = new Set<string>();
   546→  for (const t of inputTokens) {
   547→    const posting = registry.findIdsByToken(t);
   548→    if (posting.length >= 1 && posting.length <= 2) {
   549→      for (const id of posting) candIds.add(id);
   550→    }
   551→  }
   552→  if (candIds.size === 1) {
   553→    const e = registry.byId([...candIds][0]);
   554→    if (e) return toMatch(e, 0.5, "token");
   555→  }
   556→  return null;
   557→}

   563→/** خروجی normalizeAreaName — normalized همیشه نام رسمی رجیستری در صورت تطبیق (§۳.۳) */
   564→export interface NormalizedAreaName {
   565→  /** نام نرمال‌شده (در صورت تطبیق: نام رسمی رجیستری؛ وگرنه نام پاک‌شده ورودی) */
   566→  normalized: string;

   570→  method?: "exact" | "exact+pref" | "vector" | "token" | "none";

   579→/**
   580→ * نرمال‌سازی نام محله/منطقه — طبق قواعد normalizeName، سپس تطبیق با رجیستری ۹۲۶ محله.
   581→ * اگر تطبیق یافت نشد: matchedId = undefined (لاگ/شمارش در گزارش خط لوله).
   582→ */
   583→export function normalizeAreaName(raw: string, registry: LocationRegistry): NormalizedAreaName {
   584→  const normalized = normalizeName(raw);
   585→  const match = findLocationInRegistry(raw, registry);
   586→  if (!match) {
   587→    return { normalized, method: "none", score: 0 };
   588→  }
   589→  return {
   590→    normalized: match.matchedName,
   591→    matchedId: match.matchedId,
   592→    matchedName: match.matchedName,
   593→    method: match.method,
   594→    score: match.score,
   595→    matchedKind: match.kind,
   596→    ...(match.kind === "area"
   597→      ? { areaId: match.areaId, areaName: match.areaName }
   598→      : { areaId: match.areaId, areaName: match.areaName, lineId: match.matchedId, lineName: match.matchedName }),
   599→  };
   600→}

   603→// ۱۱) dedupKey — کلید یکتای مفهومی هر موجودیت (عین B.1-۷ + §۲.۱/§۳.۷ طراحی)
   604→// ==========================================================================

   606→function str(v: unknown): string {
   607→  return typeof v === "string" ? v : v === null || v === undefined ? "" : String(v);
   608→}
   609→
   610→function isoDateOrNull(v: unknown): string {
   611→  if (v instanceof Date) return v.toISOString();
   612→  const s = str(v);
   613→  if (!s) return "";
   614→  const d = new Date(s);
   615→  return Number.isNaN(d.getTime()) ? "" : d.toISOString();
   616→}

   618→/**
   619→ * کلید یکتای رکورد — عین قواعد B.1-۷ دستور:
   620→ *  transaction: contractHolo اگر موجود، وگرنه propertyCode + contractDate
   621→ *  expert: personnelCode · listing: propertyCode · request: requestCode
   622→ *  visit: visitCode · contract: contractHolo
   623→ *  interaction: requestCode + requestRealtorName + propertyRealtorName
   624→ *
   625→ * نقشه صادقانه (عین کشف §۱.۱۱ P6.5-03): «visitCode» در داده واقعی ARM وجود ندارد —
   626→ * معادل پایدار آن شناسه ردیف (id) است؛ «requestCode» تعاملات هم در ساختار واقعی
   627→ * realtor-interactions نیست — کلید واقعی جفت نام مشاوران است (requestCode در صورت
   628→ * وجود در رکورد، پیشوند می‌شود). این کلید «مفهومی» فقط برای گزارش/شمارش/تست است؛
   629→ * خود upsert روی کلید فیزیکی Prisma عمل می‌کند (§۳.۷ طراحی).
   630→ */
   631→export function dedupKey(entity: BulkEntity, record: Record<string, unknown>): string {
   632→  switch (entity) {
   633→    case "transaction": {
   634→      const holo = normalizeDigitsOnly(str(record.contractHolo));
   635→      if (holo) return `transaction:${holo}`;
   636→      const code = normalizePropertyCode(str(record.propertyCode));
   637→      const date = isoDateOrNull(record.contractDate);
   638→      return `transaction:${code}:${date}`;
   639→    }
   640→    case "expert":
   641→      return `expert:${normalizeDigitsOnly(str(record.personnelCode))}`;
   642→    case "listing":
   643→      return `listing:${normalizePropertyCode(str(record.propertyCode))}`;
   644→    case "request":
   645→      return `request:${normalizeRequestCode(str(record.requestCode))}`;
   646→    case "visit":
   647→      return `visit:${normalizeDigitsOnly(str(record.visitCode ?? record.id))}`;
   648→    case "contract": {
   649→      const holo = normalizeDigitsOnly(str(record.contractHolo));
   650→      return holo ? `contract:${holo}` : `contract:${normalizeDigitsOnly(str(record.id))}`;
   651→    }
   652→    case "interaction": {
   653→      const rc = normalizeRequestCode(str(record.requestCode));
   654→      const a = normalizeName(str(record.requestRealtorName));
   655→      const b = normalizeName(str(record.propertyRealtorName));
   656→      return `interaction:${rc}:${a}:${b}`;
   657→    }
   658→  }
   659→}

   662→// ۱۲) mergeRecords — merge با حفظ non-null (عین B.1 دستور + §۲.۴ طراحی)
   663→// ==========================================================================
   664→
   665→/**
   666→ * merge رکورد جدید با رکورد موجود — فیلدهای غیر-null جدید جایگزین می‌شوند؛
   667→ * فیلدهای null جدید، مقادیر قدیمی را پاک نمی‌کنند («جدیدترین داده غیرتهی برنده»).
   668→ * هم‌ارز خالصِ mergeNonNullUpdate در لایه Repository (منبع مرجع تست‌ها و خط لوله اعتبارسنجی).
   669→ */
   670→export function mergeRecords<T extends Record<string, unknown>>(existing: T, incoming: Partial<T>): T {
   671→  const out: Record<string, unknown> = { ...existing };
   672→  for (const [k, v] of Object.entries(incoming)) {
   673→    if (v !== null && v !== undefined) out[k] = v;
   674→  }
   675→  return out as T;
   676→}

   751→
   752→/** آمار نرمال‌سازی یک ردیف */
   753→export interface RowNormStats {
   754→  /** فیلدهایی که نرمال‌سازی مقدارشان را عوض کرد */
   755→  changedFields: string[];
   756→  /** فیلدهایی که ورودی معنادار داشت اما خروجی نامعتبر شد (رد صادقانه) */
   757→  invalidFields: string[];
   758→  /** تطبیق محله موفق (فقط listing) */
   759→  areaMatched: boolean;
   760→  /** نام محله/خط موجود ولی بدون تطبیق (فقط listing) */
   761→  areaUnmatched: boolean;
   762→  /** کلید یکتای مفهومی */
   763→  key: string;
   764→}
   765→
   766→/**
   767→ * نرمال‌سازی یک ردیف نگاشت‌شده خط لوله — خالص و idempotent
   768→ * (نرمال‌سازی مجدد ردیف نرمال‌شده = همان ردیف؛ اثبات‌شده در تست deterministic).
   769→ * فیلدهای متناسب هر موجودیت + تطبیق رجیستری محله برای listing (areaName→areaId، lineName→lineId).
   770→ */
   771→export function normalizeBulkRow<T extends Record<string, unknown>>(
   772→  entity: BulkEntity,
   773→  row: T,
   774→  registry: LocationRegistry
   775→): { row: T; stats: RowNormStats } {
   776→  const out: Record<string, unknown> = { ...row };
   777→  const changedFields: string[] = [];
   778→  const invalidFields: string[] = [];
   779→  let areaMatched = false;
   780→  let areaUnmatched = false;
   781→
   782→  for (const rule of FIELD_RULES[entity]) {
   783→    const raw = out[rule.field];
   784→    if (typeof raw !== "string" || raw.length === 0) continue; // فقط رشته‌های غیرخالی
   785→    let normalized: string;
   786→
   787→    switch (rule.kind) {
   788→      case "name":
   789→        normalized = normalizeName(raw);
   790→        break;
   791→      case "propertyCode":
   792→        normalized = normalizePropertyCode(raw);
   793→        break;
   794→      case "requestCode":
   795→        normalized = normalizeRequestCode(raw);
   796→        break;
   797→      case "mobile":
   798→        normalized = normalizeMobile(raw);
   799→        break;
   800→      case "holo":
   801→        normalized = normalizeDigitsOnly(raw);
   802→        break;
   803→      // ---- P6.5-04-e-v2: مشتق‌سازی لایه ۲ از لایه ۱ خام (سه‌لایه حقیقت) ----
   804→      case "amount": {
   805→        // rawPrice (+rawPriceUnit جدا) → price (ریال — §۳.۵ مرز واحد)
   806→        const unit = typeof out.rawPriceUnit === "string" && out.rawPriceUnit.trim() ? ` ${out.rawPriceUnit.trim()}` : "";
   807→        const price = normalizeAmount(raw + unit);
   808→        if (Number.isFinite(price)) {
   809→          if (out.price !== price) changedFields.push("price");
   810→          out.price = price;
   811→        } else if (raw.trim().length > 0) {
   812→          invalidFields.push("price");
   813→        }
   814→        continue; // خود rawPrice عین منبع می‌ماند (لایه ۱ هرگز تغییر نمی‌کند)
   815→      }
   816→      case "area": {
   817→        // rawArea → meterage (عدد ساده — بدون تفسیر واحد پول)
   818→        const meterage = parsePersianNumber(raw);
   819→        if (Number.isFinite(meterage)) {
   820→          if (out.meterage !== meterage) changedFields.push("meterage");
   821→          out.meterage = meterage;
   822→        } else if (raw.trim().length > 0) {
   823→          invalidFields.push("meterage");
   824→        }
   825→        continue; // خود rawArea عین منبع می‌ماند
   826→      }
   827→      case "jalaliDate": {
   828→        // rawDate → contractDate — فقط وقتی مقدار معتبر موجود نیست (رکورد قدیمی/ناقص)
   829→        const hasExisting =
   830→          out.contractDate instanceof Date ||
   831→          (typeof out.contractDate === "string" && out.contractDate.length > 0);
   832→        if (!hasExisting) {
   833→          const iso = normalizeJalaliDate(raw);
   834→          if (iso) {
   835→            out.contractDate = new Date(iso);
   836→            changedFields.push("contractDate");
   837→          } else if (raw.trim().length > 0) {
   838→            invalidFields.push("contractDate");
   839→          }
   840→        }
   841→        continue; // خود rawDate عین منبع می‌ماند
   842→      }
   843→      case "areaName":
   844→      case "lineName": {
   845→        const match = findLocationInRegistry(raw, registry);
   846→        if (match) {
   847→          areaMatched = true;
   848→          if (rule.kind === "areaName") {
   849→            // فیلد منطقه → هویت منطقه (چه مدخل خط باشد چه ناحیه، areaId/areaName والد معتبر است)
   850→            if (out.areaId !== match.areaId) changedFields.push("areaId");
   851→            out.areaId = match.areaId;
   852→            if (out.areaName !== match.areaName) changedFields.push("areaName");
   853→            out.areaName = match.areaName;
   854→            normalized = match.areaName;
   855→          } else if (match.kind === "line") {
   856→            // فیلد محله → هویت محله (خط)
   857→            if (out.lineId !== match.matchedId) changedFields.push("lineId");
   858→            out.lineId = match.matchedId;
   859→            normalized = match.matchedName;
   860→          } else {
   861→            // نام «منطقه» در فیلد محله → اطلاعات منطقه پر می‌شود؛ lineId نمی‌گیرد و نام فیلد پاک‌شده می‌ماند
   862→            if (!out.areaId) {
   863→              out.areaId = match.areaId;
   864→              out.areaName = match.areaName;
   865→              changedFields.push("areaId", "areaName");
   866→            }
   867→            normalized = normalizeName(raw);
   868→          }
   869→        } else {
   870→          if (raw.trim().length > 0) areaUnmatched = true;
   871→          normalized = normalizeName(raw);
   872→        }
   873→        if (out[rule.field] !== normalized) {
   874→          if (!changedFields.includes(rule.field)) changedFields.push(rule.field);
   875→          out[rule.field] = normalized;
   876→        }
   877→        if (normalized.length === 0 && raw.trim().length > 0) {
   878→          invalidFields.push(rule.field);
   879→        }
   880→        continue;
   881→      }
   882→    }
   883→
   884→    if (normalized !== raw) {
   885→      changedFields.push(rule.field);
   886→      out[rule.field] = normalized;
   887→    }
   888→    // رد صادقانه: ورودی غیرخالی → خروجی خالی (موبایل نامعتبر و…)
   889→    if (normalized.length === 0 && raw.trim().length > 0) {
   890→      invalidFields.push(rule.field);
   891→    }
   892→  }
   893→
   894→  return {
   895→    row: out as T,
   896→    stats: {
   897→      changedFields,
   898→      invalidFields,
   899→      areaMatched,
   900→      areaUnmatched,
   901→      key: dedupKey(entity, out),
   902→    },
   903→  };
   904→}
