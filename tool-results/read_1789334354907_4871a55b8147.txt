   108→/** نرمال‌سازی جست‌وجوی فارسی (ارقام فارسی/عربی، آ ← ا، حذف فاصله/ZWNJ) */
   109→function normFa(s: string): string {
   110→  return s
   111→    .replace(/[۰-۹]/g, (d) => String("۰۱۲۳۴۵۶۷۸۹".indexOf(d)))
   112→    .replace(/[٠-٩]/g, (d) => String("٠١٢٣٤٥٦٧٨٩".indexOf(d)))
   113→    .replace(/[آأإ]/g, "ا")
   114→    .replace(/[\u200c\s]/g, "")
   115→    .trim();
   116→}
