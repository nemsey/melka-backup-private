    35→    35→export function latinDigits(input: string): string {
    36→    36→  return input.replace(/[۰-۹٠-٩]/g, (d) => FA_TO_EN_DIGITS[d] ?? d);
    37→    37→}

    39→    39→/** یکسان‌سازی حروف: ي→ی، ك→ک، أ/إ/آ→ا، ة→ه + حذف اعراب/کشیده */
    40→    40→export function unifyPersianLetters(input: string): string {
    41→    41→  return input
    42→    42→    .replace(/[ي]/g, "ی")
    43→    43→    .replace(/[ك]/g, "ک")
    44→    44→    .replace(/[آأإ]/g, "ا")
    45→    45→    .replace(/[ة]/g, "ه")
    46→    46→    .replace(/[\u064B-\u0652\u0670\u0640]/g, ""); // اعراب + تنوین + کشیده ( Tateel )
    47→    47→}

    56→    56→/** نشان‌های نگارشی که از نام حذف می‌شوند (عین دستور: نقطه، ویرگول، پرانتز و…) */
    57→    57→const NAME_PUNCT_RE = /[.,()\[\]{}«»"'\u066A\u066B\u066C،؛:!?؟…\-_ـ*#+@^~`|/\\]/g;
    58→    58→
    59→    59→/**
    60→    60→ * نرمال‌سازی نام (اشخاص، مالکان) — عین قواعد B.1-۱ دستور:
    61→    61→ *  · حذف ZWNJ (U+200C)
    62→    62→ *  · ی/ک عربی → فارسی · أ/إ/آ → ا · ة → ه
    63→    63→ *  · حذف فاصله اضافی (→ یک فاصله) · حذف نشان‌های نگارشی و اعراب
    64→    64→ * نمونه شاهد دستور: «محمّد  احمدی» → «محمد احمدی»
    65→    65→ */
    66→    66→export function normalizeName(raw: string): string {
    67→    67→  if (typeof raw !== "string" || raw.length === 0) return "";
    68→    68→  let s = raw.replace(CTRL_RE, "");
    69→    69→  s = latinDigits(s);
    70→    70→  s = unifyPersianLetters(s);
    71→    71→  s = s.replace(/\u200C/g, ""); // ZWNJ حذف (عین دستور)
    72→    72→  s = s.replace(NAME_PUNCT_RE, " ");
    73→    73→  s = s.replace(/\s+/g, " ").trim();
    74→    74→  return s;
    75→    75→}

    82→    82→function normalizeArmCode(raw: string): string {
    83→    83→  if (typeof raw !== "string") return "";
    84→    84→  let s = latinDigits(raw).replace(CTRL_RE, "");
    85→    85→  s = s.replace(/[\s\u200C\u066C,،_\-]/g, ""); // حذف فاصله و جداکننده‌های رایج کد
    86→    86→  if (s.length === 0) return "";
    87→    87→  // حرف اول (P یا R یا هر پیشوند لاتین) → بزرگ؛ بقیه فقط حرف/رقم نگه داشته می‌شود
    88→    88→  const first = s[0];
    89→    89→  const isLatinLetter = /[a-zA-Z]/.test(first);
    90→    90→  if (isLatinLetter) s = first.toUpperCase() + s.slice(1);
    91→    91→  // حذف صفرهای ابتدای بخش عددی (پس از پیشوند حرفی)
    92→    92→  const body = isLatinLetter ? s.slice(1) : s;
    93→    93→  const bodyNoLeadingZeros = body.replace(/^0+(?=\d)/, "");
    94→    94→  const cleanedBody = bodyNoLeadingZeros.replace(/[^0-9A-Za-z]/g, "");
    95→    95→  if (!isLatinLetter && cleanedBody.length === 0) return "";
    96→    96→  if (isLatinLetter && cleanedBody.length === 0) return ""; // فقط حرف بدون بدنه → نامعتبر
    97→    97→  return isLatinLetter ? first.toUpperCase() + cleanedBody : cleanedBody;
    98→    98→}

   123→   123→ *  · حذف +۹۸/۰۰۹۸ → ۰ · افزودن ۰ به ۱۰رقمی‌های آغازشونده با ۹
   124→   124→ *  · اعتبارسنجی نهایی ^09\d{9}$ — نامعتبر → رشته خالی (شمارش در گزارش)
   125→   125→ */
   126→   126→export function normalizeMobile(raw: string): string {
   127→   127→  if (typeof raw !== "string") return "";
   128→   128→  let d = latinDigits(raw).replace(/[^0-9]/g, "");
   129→   129→  if (d.length === 0) return "";
   130→   130→  if (d.startsWith("0098")) d = "0" + d.slice(4);
   131→   131→  else if (d.startsWith("98") && d.length === 12) d = "0" + d.slice(2);
   132→   132→  if (d.length === 10 && d.startsWith("9")) d = "0" + d;
   133→   133→  return /^09\d{9}$/.test(d) ? d : "";
   134→   134→}

   137→   137→// ۵) normalizeAmount — مبلغ → ریال پایه (عین B.1-۵ و §۳.۵ طراحی)
   138→   138→// ==========================================================================
   139→   139→
   140→   140→/**
   141→   141→ * نرمال‌سازی مبلغ → عدد پایه به «ریال».
   142→   142→ *  · ارقام فارسی/عربی → لاتین · حذف جداکننده هزارگان (٬ ، , ' و فاصله)
   143→   143→ *  · مقیاس‌واژه‌ها: میلیون ×۱۰۰۰ · میلیارد ×۱۰⁶ · تریلیون ×۱۰⁹ (نسبت به مبنای تومان)
   144→   144→ *  · واحد: «ریال» ×۱ · «تومان» ×۱۰ · بدون واحد → پیش‌فرض تومان ×۱۰ (نمونه دستور: «۱٬۲۰۰٬۰۰۰» → ۱۲٬۰۰۰٬۰۰۰ ریال)
   145→   145→ *  · كسر با «/» یا «.» یا «٫» بین ارقام (۲/۵ میلیارد = ۲٫۵)
   146→   146→ *  · ورودی number → عیناً (مقدار پایه ساخت‌یافته — بدون تفسیر واحد متنی)
   147→   147→ *  · ورودی متنی غیرقابل‌تجزیه → NaN (پاسخ صادقانه؛ فراخواننده باید فیلد را کنار بگذارد)
   148→   148→ */
   149→   149→export function normalizeAmount(raw: string | number): number {
   150→   150→  if (typeof raw === "number") return Number.isFinite(raw) ? raw : NaN;
   151→   151→  if (typeof raw !== "string") return NaN;
   152→   152→  let s = latinDigits(raw).replace(CTRL_RE, "").trim();
   153→   153→  if (s.length === 0) return NaN;
   154→   154→
   155→   155→  // واحد پول
   156→   156→  const hasToman = s.includes("تومان");
   157→   157→  const hasRial = !hasToman && s.includes("ریال");
   158→   158→  const unitFactor = hasRial ? 1 : 10; // پیش‌فرض (بدون واحد) = تومان، عین نمونه دستور
   159→   159→
   160→   160→  // مقیاس‌واژه‌ها
   161→   161→  let scale = 1;
   162→   162→  if (s.includes("تریلیون")) scale = 1e12;
   163→   163→  else if (s.includes("میلیارد")) scale = 1e9;
   164→   164→  else if (s.includes("میلیون")) scale = 1e6;
   165→   165→
   166→   166→  // استخراج بخش عددی: نگه‌داشتن ارقام + جداکننده اعشار / و .
   167→   167→  s = s.replace(/[\u066C\u060C,'\u00A0\s]/g, ""); // جداکننده هزارگان
   168→   168→  s = s.replace(/\u066B/g, "."); // ممیز عربی اعشار
   169→   169→  const decimalMatch = s.match(/(\d+)([/.])(\d+)/);
   170→   170→  if (decimalMatch) {
   171→   171→    s = decimalMatch[1] + "." + decimalMatch[3];
   172→   172→  } else {
   173→   173→    s = s.replace(/[/.]/g, ""); // نقطه/ممیل بدون جفت رقم = نویسه زائد
   174→   174→  }
   175→   175→  const numeric = s.match(/^-?\d+(\.\d+)?/);
   176→   176→  if (!numeric) return NaN;
   177→   177→  const value = parseFloat(numeric[0]);
   178→   178→  if (!Number.isFinite(value)) return NaN;
   179→   179→  return value * scale * unitFactor;
   180→   180→}

   183→   183→// ۶) normalizeJalaliDate — تاریخ جلالی → ISO 8601 نیمه‌شب UTC (عین B.1-۶ و §۳.۴)
   184→   184→// ==========================================================================
   185→   185→
   186→   186→/**
   187→   187→ * نرمال‌سازی تاریخ جلالی → «YYYY-MM-DDT00:00:00.000Z» (تاریخ تجاری بدون ساعت).
   188→   188→ *  · فرم‌ها: 1400/01/15 · 1400-1-5 · ۱۴۰۰/۰۱/۱۵ (تبدیل با jalaliToGregorian موجود)
   189→   189→ *  · سال ۱۳۰۰–۱۶۰۰ = جلالی؛ سال ≥ ۱۶۰۱ = میلادی (تحمل ورودی ISO)
   190→   190→ *  · تاریخ نامعتبر (مثل 1400/13/01) → رشته خالی «» — هرگز مقدار ساختگی (§۳.۴ طراحی)
   191→   191→ */
   192→   192→export function normalizeJalaliDate(raw: string): string {
   193→   193→  if (typeof raw !== "string") return "";
   194→   194→  let s = latinDigits(raw).replace(CTRL_RE, "").trim();
   195→   195→  if (s.length === 0) return "";
   196→   196→
   197→   197→  // ورودی ISO با ساعت → فقط بخش تاریخ
   198→   198→  const iso = s.match(/^(\d{4})-(\d{2})-(\d{2})T/);
   199→   199→  if (iso) s = `${iso[1]}/${iso[2]}/${iso[3]}`;
   200→   200→
   201→   201→  s = s.replace(/[-.٫]/g, "/");
   202→   202→  const m = s.match(/^(\d{4})\/(\d{1,2})\/(\d{1,2})$/);
   203→   203→  if (!m) return "";
   204→   204→  const y = parseInt(m[1], 10);
   205→   205→  const mo = parseInt(m[2], 10);
   206→   206→  const d = parseInt(m[3], 10);
   207→   207→
   208→   208→  if (y >= 1300 && y <= 1600) {
   209→   209→    // جلالی
   210→   210→    if (mo < 1 || mo > 12) return "";
   211→   211→    if (d < 1 || d > jalaliMonthLength(y, mo)) return "";
   212→   212→    const [gy, gm, gd] = jalaliToGregorian(y, mo, d);
   213→   213→    return new Date(Date.UTC(gy, gm - 1, gd)).toISOString();
   214→   214→  }
   215→   215→  if (y > 1600) {
   216→   216→    // میلادی (تحمل ورودی)
   217→   217→    if (mo < 1 || mo > 12) return "";
   218→   218→    const dt = new Date(Date.UTC(y, mo - 1, d));
   219→   219→    if (dt.getUTCMonth() !== mo - 1 || dt.getUTCDate() !== d) return ""; // ۳۰ فوریه و…
   220→   220→    return dt.toISOString();
   221→   221→  }
   222→   222→  return ""; // سال خارج از بازه → قابل‌تبدیل نیست
   223→   223→}

   244→   244→export function charGramVector(flat: string): Map<string, number> {
   245→   245→  const vec = new Map<string, number>();
   246→   246→  const bump = (k: string) => vec.set(k, (vec.get(k) ?? 0) + 1);
   247→   247→  for (let i = 0; i < flat.length; i++) bump(flat[i]);
   248→   248→  for (let i = 0; i + 1 < flat.length; i++) bump(flat.slice(i, i + 2));
   249→   249→  for (let i = 0; i + 2 < flat.length; i++) bump(flat.slice(i, i + 3));
   250→   250→  return vec;
   251→   251→}

   253→   253→/** شباهت کسینوسی دو بردار فراوانی — الگوریتم عین cosineSimilarity در base-agent */
   254→   254→export function cosineSimilarity(a: Map<string, number>, b: Map<string, number>): number {
   255→   255→  if (a.size === 0 || b.size === 0) return 0;
   256→   256→  let dot = 0;
   257→   257→  let normA = 0;
   258→   258→  let normB = 0;
   259→   259→  for (const v of a.values()) normA += v * v;
   260→   260→  for (const v of b.values()) normB += v * v;
   261→   261→  if (normA === 0 || normB === 0) return 0;
   262→   262→  for (const [k, v] of a) {
   263→   263→    const w = b.get(k);
   264→   264→    if (w) dot += v * w;
   265→   265→  }
   266→   266→  return dot / (Math.sqrt(normA) * Math.sqrt(normB));
   267→   267→}

   341→   341→/** نرمال‌سازی سبک برای توکن‌سازی لایه سوم */
   342→   342→function lightNormalize(s: string): string {
   343→   343→  return latinDigits(unifyPersianLetters(s)).replace(CTRL_RE, "").replace(/[\u200C]+/g, " ").trim();
   344→   344→}
   345→   345→
   346→   346→/** ساخت رجیستری نمایه‌دار از مدخل‌های خام — خالص و deterministic؛
   347→   347→ *  نمایه‌ها: کلید کامل + کلید هسته (بدون پرانتز) + توکن‌ها؛ بردارهای لایه دوم تنبل ساخته می‌شوند */
   348→   348→export function buildLocationRegistry(rawEntries: readonly LocationEntry[]): LocationRegistry {
   349→   349→  const indexed: IndexedEntry[] = rawEntries.map((e) => ({
   350→   350→    ...e,
   351→   351→    fullKey: locationKey(e.name),
   352→   352→    coreKey: locationKey(stripParens(e.name) || e.name),
   353→   353→    tokens: new Set(tokenizeWords(lightNormalize(e.name)).filter((t) => t.length >= 3)),
   354→   354→  }));
   355→   355→
   356→   356→  const byKey = new Map<string, LocationEntry[]>();
   357→   357→  const byIdMap = new Map<string, LocationEntry>();
   358→   358→  const tokenIndex = new Map<string, Set<string>>();
   359→   359→
   360→   360→  const pushKey = (key: string, e: LocationEntry) => {
   361→   361→    if (!key) return;
   362→   362→    const arr = byKey.get(key);
   363→   363→    if (arr) arr.push(e);
   364→   364→    else byKey.set(key, [e]);
   365→   365→  };
   366→   366→  for (const e of indexed) {
   367→   367→    pushKey(e.fullKey, e);
   368→   368→    if (e.coreKey !== e.fullKey) pushKey(e.coreKey, e);
   369→   369→    byIdMap.set(e.id, e);
   370→   370→    for (const t of e.tokens) {
   371→   371→      const set = tokenIndex.get(t);
   372→   372→      if (set) set.add(e.id);
   373→   373→      else tokenIndex.set(t, new Set([e.id]));
   374→   374→    }
   375→   375→  }
   376→   376→
   377→   377→  return {
   378→   378→    entries: rawEntries,
   379→   379→    findByKey: (key: string): readonly LocationEntry[] => byKey.get(key) ?? [],
   380→   380→    findIdsByToken: (token: string): readonly string[] => [...(tokenIndex.get(token) ?? [])],
   381→   381→    byId: (id: string): LocationEntry | undefined => byIdMap.get(id),
   382→   382→  };
   383→   383→}
   384→   384→
   385→   385→// بردارهای لایه دوم — ساخته‌شده تنبل (یک‌بار برای هر رجیستری، کش WeakMap)
   386→   386→interface EntryVector {
   387→   387→  words: Map<string, number>;
   388→   388→  grams: Map<string, number>;
   389→   389→  core: string;
   390→   390→  kind: "area" | "line";
   391→   391→}
   392→   392→const REGISTRY_VECS = new WeakMap<LocationRegistry, Map<string, EntryVector>>();
   393→   393→
   394→   394→/** بردارهای مدخل‌های رجیستری — تنبل + کش‌شده؛ برای رجیستری‌های ساخته‌شده با buildLocationRegistry و رجیستری‌های تزریقی تست */
   395→   395→function registryVectors(registry: LocationRegistry): Map<string, EntryVector> {
   396→   396→  let m = REGISTRY_VECS.get(registry);
   397→   397→  if (m) return m;
   398→   398→  m = new Map<string, EntryVector>();
   399→   399→  for (const e of registry.entries) {
   400→   400→    const core = stripGenericWords(lightNormalize(stripParens(e.name) || e.name));
   401→   401→    m.set(e.id, { words: wordTermVector(core), grams: charGramVector(locationKey(core)), core, kind: e.kind });
   402→   402→  }
   403→   403→  REGISTRY_VECS.set(registry, m);
   404→   404→  return m;
   405→   405→}

   408→   408→// ۹) findLocationInRegistry — تطبیق سه‌لایه (عین B.1-۲ دستور)
   409→   409→// ==========================================================================
   410→   410→
   411→   411→/** آستانه لایه برداری — عین دستور: score ≥ ۰٫۷ */
   412→   412→export const AREA_MATCH_THRESHOLD = 0.7;
   413→   413→
   414→   414→function toMatch(e: LocationEntry, score: number, method: LocationMatch["method"]): LocationMatch {
   415→   415→  return {
   416→   416→    matchedId: e.id,
   417→   417→    matchedName: e.name,
   418→   418→    kind: e.kind,
   419→   419→    areaId: e.areaId,
   420→   420→    areaName: e.areaName,
   421→   421→    score,
   422→   422→    method,
   423→   423→  };
   424→   424→}
   425→   425→
   426→   426→function pickUniqueByKey(cands: readonly LocationEntry[]): LocationEntry | null {
   427→   427→  const areas = cands.filter((c) => c.kind === "area");
   428→   428→  if (areas.length === 1) return areas[0];
   429→   429→  const lines = cands.filter((c) => c.kind === "line");
   430→   430→  if (areas.length === 0 && lines.length === 1) return lines[0];
   431→   431→  return null; // چند ناحیه یا چند خط هم‌نام → مبهم → عدم تطبیق (fail-closed)
   432→   432→}
   433→   433→
   434→   434→/**
   435→   435→ * تطبیق نام محله/منطقه در رجیستری — سه لایه عین دستور:
   436→   436→ *  ۱) تطبیق دقیق (کلید کامل، سپس کلید هسته بدون پرانتز) — برخورد هم‌نام: ترجیح ناحیه
   437→   437→ *  ۲) تطبیق برداری: max(کسینوس واژه‌ای، کسینوس نویسه‌نگرام) روی هسته بدون واژه عمومی — آستانه ۰٫۷
   438→   438→ *     تساوی هم‌نوع (ناحیه-ناحیه / خط-خط) → مبهم؛ تساوی ناحیه-خط → ترجیح ناحیه
   439→   439→ *  ۳) توکن متمایز مشترک (posting یکتا یا دوگانه که به یک مدخل می‌رسد)
   440→   440→ * بدون تطبیق → null (صفر تطبیق کاذب — نمونه‌های ناموجود راستی‌آزمایی شده)
   441→   441→ */
   442→   442→export function findLocationInRegistry(raw: string, registry: LocationRegistry): LocationMatch | null {
   443→   443→  if (typeof raw !== "string" || raw.trim().length === 0) return null;
   444→   444→  const fullKey = locationKey(raw);
   445→   445→  if (!fullKey) return null;
   446→   446→
   447→   447→  // ---- لایه ۱: تطبیق دقیق ----
   448→   448→  const coreKey = locationKey(stripParens(raw) || raw);
   449→   449→  for (const key of fullKey === coreKey ? [fullKey] : [fullKey, coreKey]) {
   450→   450→    const cands = registry.findByKey(key);
   451→   451→    if (cands.length === 1) return toMatch(cands[0], 1, "exact");
   452→   452→    if (cands.length > 1) {
   453→   453→      const picked = pickUniqueByKey(cands);
   454→   454→      if (picked) return toMatch(picked, 1, "exact+pref");
   455→   455→    }
   456→   456→  }
   457→   457→
   458→   458→  // ---- لایه ۲: تطبیق برداری ----
   459→   459→  const inputCore = stripGenericWords(lightNormalize(stripParens(raw) || raw));
   460→   460→  const wv = wordTermVector(inputCore);
   461→   461→  const gv = charGramVector(locationKey(inputCore));
   462→   462→  const vecs = registryVectors(registry);
   463→   463→  let bestEntry: LocationEntry | null = null;
   464→   464→  let bestScore = 0;
   465→   465→  let bestKind: "area" | "line" | null = null;
   466→   466→  let secondScore = -1;
   467→   467→  let secondKind: "area" | "line" | null = null;
   468→   468→  for (const e of registry.entries) {
   469→   469→    const v = vecs.get(e.id);
   470→   470→    if (!v) continue;
   471→   471→    const score = Math.max(cosineSimilarity(wv, v.words), cosineSimilarity(gv, v.grams));
   472→   472→    if (bestEntry === null) {
   473→   473→      bestEntry = e;
   474→   474→      bestScore = score;
   475→   475→      bestKind = v.kind;
   476→   476→      continue;
   477→   477→    }
   478→   478→    if (score > bestScore + 1e-9) {
   479→   479→      // برترین جدید — برترین قبلی به دومین تبدیل می‌شود
   480→   480→      secondScore = bestScore;
   481→   481→      secondKind = bestKind;
   482→   482→      bestEntry = e;
   483→   483→      bestScore = score;
   484→   484→      bestKind = v.kind;
   485→   485→    } else if (Math.abs(score - bestScore) <= 1e-9) {
   486→   486→      if (v.kind === "area" && bestKind === "line") {
   487→   487→        // تساوی ناحیه-خط → ناحیه جایگزین؛ خط به دومین تبدیل می‌شود
   488→   488→        secondScore = bestScore;
   489→   489→        secondKind = bestKind;
   490→   490→        bestEntry = e;
   491→   491→        bestKind = v.kind;
   492→   492→      } else {
   493→   493→        // هم‌امتیاز هم‌نوع (یا خط هم‌امتیاز با ناحیه برتر) → کاندید دوم
   494→   494→        secondScore = score;
   495→   495→        secondKind = v.kind;
   496→   496→      }
   497→   497→    } else if (score > secondScore) {
   498→   498→      secondScore = score;
   499→   499→      secondKind = v.kind;
   500→   500→    }
   501→   501→  }
   502→   502→  if (bestEntry && bestKind && bestScore >= AREA_MATCH_THRESHOLD) {
   503→   503→    // تساوی دقیق هم‌نوع (ناحیه-ناحیه / خط-خط) → مبهم (fail-closed)؛ ناحیه-خط هم‌امتیاز → ناحیه (بالا اعمال شد)
   504→   504→    if (secondKind !== null && secondKind === bestKind && Math.abs(secondScore - bestScore) <= 1e-9) {
   505→   505→      return null;
   506→   506→    }
   507→   507→    return toMatch(bestEntry, bestScore, "vector");
   508→   508→  }
   509→   509→
   510→   510→  // ---- لایه ۳: توکن متمایز مشترک ----
   511→   511→  const inputTokens = tokenizeWords(lightNormalize(raw)).filter((t) => t.length >= 3);
   512→   512→  const candIds = new Set<string>();
   513→   513→  for (const t of inputTokens) {
   514→   514→    const posting = registry.findIdsByToken(t);
   515→   515→    if (posting.length >= 1 && posting.length <= 2) {
   516→   516→      for (const id of posting) candIds.add(id);
   517→   517→    }
   518→   518→  }
   519→   519→  if (candIds.size === 1) {
   520→   520→    const e = registry.byId([...candIds][0]);
   521→   521→    if (e) return toMatch(e, 0.5, "token");
   522→   522→  }
   523→   523→  return null;
   524→   524→}

   531→   531→export interface NormalizedAreaName {

   548→   548→ * اگر تطبیق یافت نشد: matchedId = undefined (لاگ/شمارش در گزارش خط لوله).

   560→   560→    method: match.method,
   561→   561→    score: match.score,
   562→   562→    matchedKind: match.kind,
   563→   563→    ...(match.kind === "area"
   564→   564→      ? { areaId: match.areaId, areaName: match.areaName }
   565→   565→      : { areaId: match.areaId, areaName: match.areaName, lineId: match.matchedId, lineName: match.matchedName }),
   566→   566→  };
   567→   567→}

   570→   570→// ۱۱) dedupKey — کلید یکتای مفهومی هر موجودیت (عین B.1-۷ + §۲.۱/§۳.۷ طراحی)
   571→   571→// ==========================================================================
   572→   572→
   573→   573→function str(v: unknown): string {
   574→   574→  return typeof v === "string" ? v : v === null || v === undefined ? "" : String(v);
   575→   575→}
   576→   576→
   577→   577→function isoDateOrNull(v: unknown): string {
   578→   578→  if (v instanceof Date) return v.toISOString();
   579→   579→  const s = str(v);
   580→   580→  if (!s) return "";
   581→   581→  const d = new Date(s);
   582→   582→  return Number.isNaN(d.getTime()) ? "" : d.toISOString();
   583→   583→}
   584→   584→
   585→   585→/**
   586→   586→ * کلید یکتای رکورد — عین قواعد B.1-۷ دستور:
   587→   587→ *  transaction: contractHolo اگر موجود، وگرنه propertyCode + contractDate
   588→   588→ *  expert: personnelCode · listing: propertyCode · request: requestCode
   589→   589→ *  visit: visitCode · contract: contractHolo
   590→   590→ *  interaction: requestCode + requestRealtorName + propertyRealtorName
   591→   591→ *
   592→   592→ * نقشه صادقانه (عین کشف §۱.۱۱ P6.5-03): «visitCode» در داده واقعی ARM وجود ندارد —
   593→   593→ * معادل پایدار آن شناسه ردیف (id) است؛ «requestCode» تعاملات هم در ساختار واقعی
   594→   594→ * realtor-interactions نیست — کلید واقعی جفت نام مشاوران است (requestCode در صورت
   595→   595→ * وجود در رکورد، پیشوند می‌شود). این کلید «مفهومی» فقط برای گزارش/شمارش/تست است؛
   596→   596→ * خود upsert روی کلید فیزیکی Prisma عمل می‌کند (§۳.۷ طراحی).
   597→   597→ */
   598→   598→export function dedupKey(entity: BulkEntity, record: Record<string, unknown>): string {
   599→   599→  switch (entity) {
   600→   600→    case "transaction": {
   601→   601→      const holo = normalizeDigitsOnly(str(record.contractHolo));
   602→   602→      if (holo) return `transaction:${holo}`;
   603→   603→      const code = normalizePropertyCode(str(record.propertyCode));
   604→   604→      const date = isoDateOrNull(record.contractDate);
   605→   605→      return `transaction:${code}:${date}`;
   606→   606→    }
   607→   607→    case "expert":
   608→   608→      return `expert:${normalizeDigitsOnly(str(record.personnelCode))}`;
   609→   609→    case "listing":
   610→   610→      return `listing:${normalizePropertyCode(str(record.propertyCode))}`;
   611→   611→    case "request":
   612→   612→      return `request:${normalizeRequestCode(str(record.requestCode))}`;
   613→   613→    case "visit":
   614→   614→      return `visit:${normalizeDigitsOnly(str(record.visitCode ?? record.id))}`;
   615→   615→    case "contract": {
   616→   616→      const holo = normalizeDigitsOnly(str(record.contractHolo));
   617→   617→      return holo ? `contract:${holo}` : `contract:${normalizeDigitsOnly(str(record.id))}`;
   618→   618→    }
   619→   619→    case "interaction": {
   620→   620→      const rc = normalizeRequestCode(str(record.requestCode));
   621→   621→      const a = normalizeName(str(record.requestRealtorName));
   622→   622→      const b = normalizeName(str(record.propertyRealtorName));
   623→   623→      return `interaction:${rc}:${a}:${b}`;
   624→   624→    }
   625→   625→  }
   626→   626→}

   634→   634→ * فیلدهای null جدید، مقادیر قدیمی را پاک نمی‌کنند («جدیدترین داده غیرتهی برنده»).
   635→   635→ * هم‌ارز خالصِ mergeNonNullUpdate در لایه Repository (منبع مرجع تست‌ها و خط لوله اعتبارسنجی).
   636→   636→ */
   637→   637→export function mergeRecords<T extends Record<string, unknown>>(existing: T, incoming: Partial<T>): T {
   638→   638→  const out: Record<string, unknown> = { ...existing };
   639→   639→  for (const [k, v] of Object.entries(incoming)) {
   640→   640→    if (v !== null && v !== undefined) out[k] = v;
   641→   641→  }
   642→   642→  return out as T;
   643→   643→}

   706→   706→  /** فیلدهایی که نرمال‌سازی مقدارشان را عوض کرد */
   707→   707→  changedFields: string[];
   708→   708→  /** فیلدهایی که ورودی معنادار داشت اما خروجی نامعتبر شد (رد صادقانه) */
   709→   709→  invalidFields: string[];
   710→   710→  /** تطبیق محله موفق (فقط listing) */
   711→   711→  areaMatched: boolean;
   712→   712→  /** نام محله/خط موجود ولی بدون تطبیق (فقط listing) */
   713→   713→  areaUnmatched: boolean;
   714→   714→  /** کلید یکتای مفهومی */
   715→   715→  key: string;
   716→   716→}
   717→   717→
   718→   718→/**
   719→   719→ * نرمال‌سازی یک ردیف نگاشت‌شده خط لوله — خالص و idempotent
   720→   720→ * (نرمال‌سازی مجدد ردیف نرمال‌شده = همان ردیف؛ اثبات‌شده در تست deterministic).
   721→   721→ * فیلدهای متناسب هر موجودیت + تطبیق رجیستری محله برای listing (areaName→areaId، lineName→lineId).
   722→   722→ */
   723→   723→export function normalizeBulkRow<T extends Record<string, unknown>>(
   724→   724→  entity: BulkEntity,
   725→   725→  row: T,
   726→   726→  registry: LocationRegistry
   727→   727→): { row: T; stats: RowNormStats } {
   728→   728→  const out: Record<string, unknown> = { ...row };
   729→   729→  const changedFields: string[] = [];
   730→   730→  const invalidFields: string[] = [];
   731→   731→  let areaMatched = false;
   732→   732→  let areaUnmatched = false;
   733→   733→
   734→   734→  for (const rule of FIELD_RULES[entity]) {
   735→   735→    const raw = out[rule.field];
   736→   736→    if (typeof raw !== "string" || raw.length === 0) continue; // فقط رشته‌های غیرخالی
   737→   737→    let normalized: string;
   738→   738→
   739→   739→    switch (rule.kind) {
   740→   740→      case "name":
   741→   741→        normalized = normalizeName(raw);
   742→   742→        break;
   743→   743→      case "propertyCode":
   744→   744→        normalized = normalizePropertyCode(raw);
   745→   745→        break;
   746→   746→      case "requestCode":
   747→   747→        normalized = normalizeRequestCode(raw);
   748→   748→        break;
   749→   749→      case "mobile":
   750→   750→        normalized = normalizeMobile(raw);
   751→   751→        break;
   752→   752→      case "holo":
   753→   753→        normalized = normalizeDigitsOnly(raw);
   754→   754→        break;
   755→   755→      case "areaName":
   756→   756→      case "lineName": {
   757→   757→        const match = findLocationInRegistry(raw, registry);
   758→   758→        if (match) {
   759→   759→          areaMatched = true;
   760→   760→          if (rule.kind === "areaName") {
   761→   761→            // فیلد منطقه → هویت منطقه (چه مدخل خط باشد چه ناحیه، areaId/areaName والد معتبر است)
   762→   762→            if (out.areaId !== match.areaId) changedFields.push("areaId");
   763→   763→            out.areaId = match.areaId;
   764→   764→            if (out.areaName !== match.areaName) changedFields.push("areaName");
   765→   765→            out.areaName = match.areaName;
   766→   766→            normalized = match.areaName;
   767→   767→          } else if (match.kind === "line") {
   768→   768→            // فیلد محله → هویت محله (خط)
   769→   769→            if (out.lineId !== match.matchedId) changedFields.push("lineId");
   770→   770→            out.lineId = match.matchedId;
   771→   771→            normalized = match.matchedName;
   772→   772→          } else {
   773→   773→            // نام «منطقه» در فیلد محله → اطلاعات منطقه پر می‌شود؛ lineId نمی‌گیرد و نام فیلد پاک‌شده می‌ماند
   774→   774→            if (!out.areaId) {
   775→   775→              out.areaId = match.areaId;
   776→   776→              out.areaName = match.areaName;
   777→   777→              changedFields.push("areaId", "areaName");
   778→   778→            }
   779→   779→            normalized = normalizeName(raw);
   780→   780→          }
   781→   781→        } else {
   782→   782→          if (raw.trim().length > 0) areaUnmatched = true;
   783→   783→          normalized = normalizeName(raw);
   784→   784→        }
   785→   785→        if (out[rule.field] !== normalized) {
   786→   786→          if (!changedFields.includes(rule.field)) changedFields.push(rule.field);
   787→   787→          out[rule.field] = normalized;
   788→   788→        }
   789→   789→        if (normalized.length === 0 && raw.trim().length > 0) {
   790→   790→          invalidFields.push(rule.field);
   791→   791→        }
   792→   792→        continue;
   793→   793→      }
   794→   794→    }
   795→   795→
   796→   796→    if (normalized !== raw) {
   797→   797→      changedFields.push(rule.field);
   798→   798→      out[rule.field] = normalized;
   799→   799→    }
   800→   800→    // رد صادقانه: ورودی غیرخالی → خروجی خالی (موبایل نامعتبر و…)
   801→   801→    if (normalized.length === 0 && raw.trim().length > 0) {
   802→   802→      invalidFields.push(rule.field);
   803→   803→    }
   804→   804→  }
   805→   805→
   806→   806→  return {
   807→   807→    row: out as T,
   808→   808→    stats: {
   809→   809→      changedFields,
   810→   810→      invalidFields,
   811→   811→      areaMatched,
   812→   812→      areaUnmatched,
   813→   813→      key: dedupKey(entity, out),
   814→   814→    },
   815→   815→  };
   816→   816→}
