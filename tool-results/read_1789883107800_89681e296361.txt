     1→/**
     2→ * Bulk Import Pipeline — P6.5-03 (طراحی §۱ PHASE-6.5 + دستور شورا P6.5-03)
     3→ * ------------------------------------------------------
     4→ * خط لوله دریافت داده انبوه از ARM/PM با چهار ستون فنی:
     5→ *   ۱) Checkpoint/Resume — ادامه از آخرین صفحهٔ «کامل-ذخیره‌شده» (SyncState.bulkImportCheckpoint)
     6→ *   ۲) Idempotency — upsert بر کلید طبیعی + merge non-null (§۲.۴) ⇒ اجرای دوباره = بی‌ضرر
     7→ *   ۳) Rate Limiting — حداقل فاصله «پایان پاسخ قبلی تا شروع درخواست بعدی» (§۱.۸)
     8→ *   ۴) Retry — ۵ تلاش با backoff نمایی 2s→4s→8s→16s→30s روی 5xx/429/شبکه (§۱.۷)
     9→ *
    10→ * اصول حاکم (قواعد بخش C دستور):
    11→ *   · صفر LLM call · صفر cost · صفر fetch مستقیم (فقط melk-api/client) · صفر Prisma مستقیم (فقط Repository)
    12→ *   · صفر فشار به سامانه مبدأ — rate limiting اجباری (default 200ms)
    13→ *   · Bulk هرگز deleteMany صادر نمی‌کند (§۱.۹) و فیلدهای sync منظم را لمس نمی‌کند (§۱.۵)
    14→ *
    15→ * معماری پورت‌ها (تست‌پذیری کامل با موک — §۱.۱۰: توسعه با موک، صفر فشار به API مبدأ):
    16→ *   BulkPageFetcher  = سازگار منبع (ARM/PM → ردیف‌های نگاشت‌شده UpsertInput)
    17→ *   BulkRecordSink   = ماندگاری (Repositoryهای upsertBulk با merge non-null)
    18→ *   BulkCheckpointStore = نقطه ازسرگیری (SyncState از طریق IntegrationRepository)
    19→ *
    20→ * انحراف صادقانه مستند (۱): پس از اتمام retryهای یک صفحه، نتیجه FAILED بازگردانده می‌شود و
    21→ * صفحه‌های بعدی طی نمی‌شوند — بند «continue به صفحه بعد» دستور با بند ۳ همان دستور
    22→ * (ذخیره checkpoint + توقف) در تعارض است و ادامۀ صفحه‌ها حفره داده می‌سازد که checkpoint
    23→ * قادر به ترمیمش نیست (نقض صفر حذف داده). انتخاب: مسیر طراحی مصوب §۱.۴-۳.
    24→ * status=PARTIAL برای خطاهای «رکورد-منفرد» درون صفحات موفق رزرو شده است.
    25→ */

   204→/** عین دستور: 2s→4s→8s→16s→30s (سقف 30s برای تلاش‌های بیشتر) */
   205→export const DEFAULT_RETRY_DELAYS_MS: readonly number[] = [2000, 4000, 8000, 16000, 30000];

   207→/** تأخیر پیش از تلاش retry شماره n (1-based) — null یعنی «دیگر retry نکن» */
   208→export function retryDelayFor(retryNumber: number, maxRetries: number): number | null {
   209→  if (retryNumber < 1) return null;
   210→  if (retryNumber > maxRetries) return null;
   211→  const idx = Math.min(retryNumber - 1, DEFAULT_RETRY_DELAYS_MS.length - 1);
   212→  return DEFAULT_RETRY_DELAYS_MS[idx];
   213→}

   215→/** خطای شبکه‌ای/HTTP با حمایت Retry-After — قابل‌تزریق در تست */
   216→export class BulkFetchError extends Error {
   217→  readonly status?: number;
   218→  readonly retryable: boolean;
   219→  readonly retryAfterMs?: number;
   220→  constructor(message: string, opts: { status?: number; retryable: boolean; retryAfterMs?: number } = { retryable: false }) {
   221→    super(message);
   222→    this.name = "BulkFetchError";
   223→    this.status = opts.status;
   224→    this.retryable = opts.retryable;
   225→    this.retryAfterMs = opts.retryAfterMs;
   226→  }

   229→/** §۱.۷ — قابل‌retry: 5xx/429/شبکه (پرچم retryable)؛ 4xx غیر 429 = فوری؛ ناشناخته = فوری */
   230→export function classifyFetchError(err: unknown): { retryable: boolean; retryAfterMs: number | null } {
   231→  if (err instanceof BulkFetchError) {
   232→    return { retryable: err.retryable, retryAfterMs: err.retryAfterMs ?? null };
   233→  }
   234→  if (err instanceof MelkApiError) {
   235→    const retryable = err.retryable === true || err.httpStatus >= 500 || err.httpStatus === 429;
   236→    return { retryable, retryAfterMs: null };
   237→  }
   238→  return { retryable: false, retryAfterMs: null };
   239→}

   252→function defaultSleep(ms: number): Promise<void> {
   253→  return new Promise((resolve) => setTimeout(resolve, ms));

   256→// ==========================================================================
   257→// RateLimiter — §۱.۸: فاصله از «پایان پاسخ قبلی» تا «شروع درخواست بعدی»
   258→// ==========================================================================
   259→
   260→export class RateLimiter {
   261→  private lastCompletedAt = 0;
   262→
   263→  constructor(private readonly minIntervalMs: number) {}
   264→
   265→  /** پیش از هر درخواست — نخستین درخواست آزاد است */
   266→  async wait(): Promise<void> {
   267→    if (this.lastCompletedAt === 0) return;
   268→    const elapsed = Date.now() - this.lastCompletedAt;
   269→    const remaining = this.minIntervalMs - elapsed;
   270→    if (remaining > 0) await defaultSleep(remaining);
   271→  }
   272→
   273→  /** پس از پایان هر پاسخ (موفق یا ناموفق) */
   274→  complete(): void {
   275→    this.lastCompletedAt = Date.now();
   276→  }
   277→}

   283→/** سقف ایمنی صفحات (حلقه بی‌نهایت منبع خراب را نمی‌پذیرد) — additive option */
   284→const DEFAULT_MAX_PAGES = 10_000;
   285→
   286→class BulkImportRunner {
   287→  private readonly opts: BulkImportOptions;
   288→  private readonly pageSize: number;
   289→  private readonly batchSize: number;
   290→  private readonly maxRetries: number;
   291→  private readonly deps: BulkImportDeps;
   292→  private readonly limiter: RateLimiter;
   293→  private readonly sleep: (ms: number) => Promise<void>;
   294→  private registry: LocationRegistry | null = null;
   295→
   296→  constructor(options: BulkImportOptions, deps: BulkImportDeps) {
   297→    this.opts = options;
   298→    this.pageSize = clampPageSize(options.pageSize);
   299→    this.batchSize = clampBatchSize(options.batchSize);
   300→    this.maxRetries = options.maxRetries ?? 5;
   301→    this.deps = deps;
   302→    this.limiter = new RateLimiter(Math.max(1, options.rateLimitMs ?? 200));
   303→    this.sleep = deps.sleep ?? defaultSleep;
   304→  }
   305→
   306→  private now(): Date {
   307→    return this.deps.now ? this.deps.now() : new Date();
   308→  }
   309→
   310→  /** رجیستری محله — تزریقی تست یا پیش‌فرض ۹۲۶ (lazy + cache) */
   311→  private resolveRegistry(): LocationRegistry {
   312→    if (this.deps.registry) return this.deps.registry;
   313→    if (!this.registry) this.registry = getDefaultLocationRegistry();
   314→    return this.registry;
   315→  }
   316→
   317→  /**
   318→   * P6.5-04-e/e-v2 — نرمال‌سازی لایه ۲ + مهر provenance (سه‌لایه حقیقت):
   319→   * لایه ۱ (Raw) توسط نگاشت آداپتور عین منبع پر شده؛ اینجا لایه ۲ فیلدهای
   320→   * نرمال را از خام مشتق می‌کند (amount/area/jalaliDate + نام‌ها/کدها) و
   321→   * مهر extractedAt/extractionBatchId/normalizationVersion روی هر ردیف می‌نشیند.
   322→   * نرمال‌سازی هرگز مقادیر خام را بازنویسی نمی‌کند.
   323→   */
   324→  private normalizeAndStamp(
   325→    entity: BulkEntity,
   326→    rows: readonly unknown[],
   327→    batchRunId: string | undefined,
   328→    stats: NormalizationStats
   329→  ): unknown[] {
   330→    const registry = this.resolveRegistry();
   331→    const extractedAt = this.now();
   332→    const out: unknown[] = [];
   333→    for (const row of rows) {
   334→      const { row: normalized, stats: rowStats } = normalizeBulkRow(
   335→        entity,
   336→        (row ?? {}) as Record<string, unknown>,
   337→        registry
   338→      );
   339→      const stamped = normalized as Record<string, unknown>;
   340→      stamped.extractedAt = extractedAt;
   341→      stamped.extractionBatchId = batchRunId ?? null;
   342→      stamped.normalizationVersion = NORMALIZATION_VERSION;
   343→      // rawSourceVersion: ARM فعلاً نسخه داده منتشر نمی‌کند — تهی صادقانه (کشف در Reconnaissance §۶)
   344→      if (stamped.rawSourceVersion === undefined) stamped.rawSourceVersion = null;
   345→      stats.rowsSeen += 1;
   346→      if (rowStats.changedFields.length > 0) stats.rowsChanged += 1;
   347→      stats.fieldsChanged += rowStats.changedFields.length;
   348→      stats.invalidValues += rowStats.invalidFields.length;
   349→      if (rowStats.areaMatched) stats.areaMatched += 1;
   350→      if (rowStats.areaUnmatched) stats.areaUnmatched += 1;
   351→      out.push(stamped);
   352→    }
   353→    return out;
   354→  }
   355→
   356→  async run(): Promise<BulkImportResult> {
   357→    const t0 = this.now().getTime();
   358→    const entity = this.opts.entity;
   359→    const range =
   360→      this.opts.startDate || this.opts.endDate
   361→        ? {
   362→            from: (this.opts.startDate ?? this.opts.endDate)!.toISOString(),
   363→            to: (this.opts.endDate ?? this.opts.startDate)!.toISOString(),
   364→          }
   365→        : null;
   366→
   367→    // ---- ۱) بارگذاری checkpoint (تطبیق entity + بازه + نسخه) ----
   368→    const saved = await this.deps.checkpointStore.load(entity);
   369→    const resumable =
   370→      saved !== null &&
   371→      saved.version === BULK_CHECKPOINT_VERSION &&
   372→      sameDateRange(saved.dateRange ?? null, range);
   373→
   374→    const cp: BulkImportCheckpoint = resumable
   375→      ? { ...saved, runId: saved.runId ?? this.deps.newRunId?.() ?? randomUUID() }
   376→      : {
   377→          entity,
   378→          lastPage: 0,
   379→          lastRecordId: "",
   380→          totalProcessed: 0,
   381→          startedAt: this.now().toISOString(),
   382→          version: BULK_CHECKPOINT_VERSION,
   383→          runId: this.deps.newRunId?.() ?? randomUUID(),
   384→          updatedAt: this.now().toISOString(),
   385→          dateRange: range,
   386→          totals: { fetched: 0, saved: 0, duplicates: 0, errors: 0 },
   387→        };
   388→
   389→    const progress: BulkImportProgress = {
   390→      entity,
   391→      totalFetched: cp.totals?.fetched ?? 0,
   392→      totalSaved: cp.totals?.saved ?? 0,
   393→      totalDuplicates: cp.totals?.duplicates ?? 0,
   394→      totalErrors: cp.totals?.errors ?? 0,
   395→      currentPage: cp.lastPage,
   396→      currentBatch: 0,
   397→      checkpoint: cp,
   398→      startedAt: new Date(cp.startedAt),
   399→      updatedAt: this.now(),
   400→    };
   401→
   402→    let status: BulkImportResult["status"] = "COMPLETED";
   403→    let error: string | undefined;
   404→
   405→    // P6.5-04 — آمار نرمال‌سازی این اجرا (سه‌لایه حقیقت)
   406→    const normStats: NormalizationStats = {
   407→      rowsSeen: 0,
   408→      rowsChanged: 0,
   409→      fieldsChanged: 0,
   410→      invalidValues: 0,
   411→      areaMatched: 0,
   412→      areaUnmatched: 0,
   413→    };
   414→
   415→    // ---- ۲) حلقه صفحات ----
   416→    let page = cp.lastPage + 1; // نقطه ادامه = آخرین صفحه کامل + ۱
   417→    let pagesRan = 0;
   418→
   419→    while (true) {
   420→      if (pagesRan >= DEFAULT_MAX_PAGES) {
   421→        // سقف ایمنی — از سر رفتن حلقه منبع خراب؛ داده‌های تا اینجا سالم‌اند
   422→        status = "PARTIAL";
   423→        error = `سقف ایمنی صفحات (${DEFAULT_MAX_PAGES}) رسید؛ داده تا صفحه ${cp.lastPage} سالم ذخیره شد.`;
   424→        break;
   425→      }
   426→
   427→      let fetched: BulkFetchPage;
   428→      try {
   429→        fetched = await this.fetchPageWithRetry(page);
   430→      } catch (err) {
   431→        // §۱.۴-۳ — checkpoint (آخرین صفحه کامل) ذخیره + توقف با خطای صریح فارسی
   432→        await this.deps.checkpointStore.save(cp);
   433→        status = "FAILED";
   434→        error = fetchFailureMessage(err, entity, page, cp.lastRecordId);
   435→        break;
   436→      }
   437→
   438→      progress.totalFetched += fetched.rows.length;
   439→      pagesRan += 1;
   440→
   441→      if (fetched.rows.length === 0) break; // پایان داده
   442→
   443→      // ---- P6.5-04: نرمال‌سازی لایه ۲ + مهر provenance (پیش از ذخیره) ----
   444→      const normalizedRows = this.normalizeAndStamp(entity, fetched.rows, cp.runId, normStats);
   445→
   446→      // ---- ذخیره: برش‌های batchSize + فلاش اجباری پیش از پیشروی checkpoint ----
   447→      for (let offset = 0; offset < normalizedRows.length; offset += this.batchSize) {
   448→        const chunk = normalizedRows.slice(offset, offset + this.batchSize);
   449→        const stats = await this.deps.sink.saveBatch(chunk, { sourceRunId: cp.runId!, entity });
   450→        progress.totalSaved += stats.created + stats.updated;
   451→        progress.totalDuplicates += stats.duplicates;
   452→        progress.totalErrors += stats.errors;
   453→        progress.currentBatch += 1;
   454→      }
   455→
   456→      // نا-تکرارپذیری فلاش/چک‌پوینت: صفحه کامل شد → داده نوشته شد → حالا checkpoint جلو می‌رود
   457→      const lastRow = normalizedRows[normalizedRows.length - 1] as Record<string, unknown> | undefined;
   458→      cp.lastRecordId = String(lastRow?.id ?? lastRow?.propertyCode ?? lastRow?.requestCode ?? lastRow?.personnelCode ?? "");
   459→      cp.totalProcessed = progress.totalFetched;
   460→      cp.lastPage = page;
   461→      cp.updatedAt = this.now().toISOString();
   462→      cp.totals = {
   463→        fetched: progress.totalFetched,
   464→        saved: progress.totalSaved,
   465→        duplicates: progress.totalDuplicates,
   466→        errors: progress.totalErrors,
   467→      };
   468→      await this.deps.checkpointStore.save(cp);
   469→
   470→      progress.currentPage = page;
   471→      progress.checkpoint = { ...cp };
   472→      progress.updatedAt = this.now();
   473→      this.opts.onProgress?.({ ...progress, checkpoint: { ...cp, totals: cp.totals ? { ...cp.totals } : undefined } });
   474→
   475→      if (!fetched.hasMore) break;
   476→      page += 1;
   477→    }
   478→
   479→    // خطای رکورد-منفرد درون صفحات موفق ⇒ PARTIAL (نه FAILED)
   480→    if (status === "COMPLETED" && progress.totalErrors > 0) {
   481→      status = "PARTIAL";
   482→      error = `${progress.totalErrors} رکورد در حین ذخیره خطا خورد (صفحات طی شد؛ از اجرای مجدد idempotent استفاده کنید).`;
   483→    }
   484→
   485→    return {
   486→      entity,
   487→      status,
   488→      progress,
   489→      ...(error !== undefined ? { error } : {}),
   490→      durationMs: this.now().getTime() - t0,
   491→      normalization: normStats,
   492→    };
   493→  }
   494→
   495→  /** واکشی یک صفحه با retry طبقه‌بندی‌شده + احترام کامل به rate limit و Retry-After */
   496→  private async fetchPageWithRetry(page: number): Promise<BulkFetchPage> {
   497→    let retryNumber = 0;
   498→    while (true) {
   499→      await this.limiter.wait();
   500→      try {
   501→        const res = await this.deps.fetcher.fetchPage({
   502→          entity: this.opts.entity,
   503→          page,
   504→          pageSize: this.pageSize,
   505→          ...(this.opts.startDate ? { startDate: this.opts.startDate } : {}),
   506→          ...(this.opts.endDate ? { endDate: this.opts.endDate } : {}),
   507→        });
   508→        this.limiter.complete();
   509→        return res;
   510→      } catch (err) {
   511→        this.limiter.complete(); // درخواست (ناموفق) هم «تمام شد» — فاصله از پایان آن
   512→        const { retryable, retryAfterMs } = classifyFetchError(err);
   513→        const policyDelay = retryDelayFor(retryNumber + 1, this.maxRetries);
   514→        if (!retryable || policyDelay === null) throw err;
   515→        // 429: احترام به Retry-After مبدأ — هرگز کمتر از سیاست، هرچه بیشتر خواست شکیبایی
   516→        await this.sleep(Math.max(policyDelay, retryAfterMs ?? 0));
   517→        retryNumber += 1;
   518→      }
   519→    }
   520→  }
   521→}

   536→export async function bulkImport(
   537→  options: BulkImportOptions,
   538→  deps: Partial<BulkImportDeps> & { armToken?: string; repoClient?: PrismaClient } = {}
   539→): Promise<BulkImportResult> {
   540→  const engineOpts: BulkImportOptions = {
   541→    ...options,
   542→    pageSize: clampPageSize(options.pageSize),
   543→    batchSize: clampBatchSize(options.batchSize),
   544→    rateLimitMs: Math.max(1, options.rateLimitMs ?? 200),
   545→    maxRetries: options.maxRetries ?? 5,
   546→  };
   547→  const fetcher =
   548→    deps.fetcher ??
   549→    createArmBulkFetcher({ entity: options.entity, armToken: deps.armToken ?? "" });
   550→  const sink = deps.sink ?? createRepositorySink(options.entity, { client: deps.repoClient });
   551→  const checkpointStore = deps.checkpointStore ?? new SyncStateBulkCheckpointStore({ client: deps.repoClient });
   552→  return new BulkImportRunner(engineOpts, {
   553→    fetcher,
   554→    sink,
   555→    checkpointStore,
   556→    ...(deps.sleep ? { sleep: deps.sleep } : {}),
   557→    ...(deps.now ? { now: deps.now } : {}),
   558→    ...(deps.newRunId ? { newRunId: deps.newRunId } : {}),
   559→    ...(deps.registry ? { registry: deps.registry } : {}),
   560→  }).run();
   561→}

   567→/** نگاشت ردیف خام → ورودی مخزن برای هر موجودیت (listing/request = stub تا P6.5-04) */
   568→type RawRow = Record<string, unknown>;
   569→
   570→function pickRaw(row: RawRow, keys: string[]): unknown {
   571→  for (const k of keys) {
   572→    const v = row[k];
   573→    if (v !== undefined && v !== null && v !== "") return v;
   574→  }
   575→  return null;
   576→}
   577→
   578→function pickRawStr(row: RawRow, keys: string[]): string | null {
   579→  const v = pickRaw(row, keys);
   580→  if (typeof v === "string") return v;
   581→  if (v === undefined || v === null) return null;
   582→  if (typeof v === "object") return null; // شیء تو در تو ([object Object] نه!) — نگاشت تو در تو در آداپتور کامل
   583→  return String(v);
   584→}

   586→/**
   587→ * آگهی ARM → PropertyBulkInput — نگاشت با لایه ۱ حقیقت (P6.5-04-e-v2):
   588→ * مقادیر خام عین منبع (rawPrice/rawPriceUnit/rawArea) نگاشته می‌شوند و
   589→ * لایه ۲ (normalizeBulkRow با kind=amount/area) در خط لوله آن‌ها را به
   590→ * price (ریال) و meterage مشتق می‌کند — خود خام هرگز تغییر نمی‌کند.
   591→ * کلیدهای کاندید قیمت/متراژ: تطبیق نهایی با داده زنده در D0/Reconnaissance.
   592→ */
   593→function stubMapPropertyRow(row: RawRow): PropertyBulkInput | null {
   594→  const id = pickRawStr(row, ["id", "_id", "objectId"]);
   595→  const propertyCode = pickRawStr(row, ["code", "propertyCode"]);
   596→  if (!id && !propertyCode) return null;
   597→  return {
   598→    id: id ?? propertyCode!,
   599→    propertyCode: propertyCode ?? id!,
   600→    title: pickRawStr(row, ["title"]),
   601→    status: pickRawStr(row, ["status"]),
   602→    ownerName: pickRawStr(row, ["relatedRealtorName", "realtorName", "ownerName"]),
   603→    ownerMobile: pickRawStr(row, ["ownerMobile", "mobileNumber", "mobile"]),
   604→    // مکان — کلیدهای کاندید (تطبیق نهایی با داده زنده در D0/Reconnaissance)
   605→    areaName: pickRawStr(row, ["areaName", "areaTitle", "area"]),
   606→    lineName: pickRawStr(row, ["lineName", "lineTitle", "line"]),
   607→    // P6.5-04-e-v2 — لایه ۱ حقیقت (Raw) — عین منبع، بدون هیچ تبدیلی
   608→    // (کلیدهای متراژ عمداً جدا از کلیدهای نام محله — «area» نام محله است، نه متراژ)
   609→    rawPrice: pickRawStr(row, ["price", "priceLabel", "priceText", "totalPrice", "priceFa"]),
   610→    rawPriceUnit: pickRawStr(row, ["priceUnit", "unit", "currencyLabel"]),
   611→    rawArea: pickRawStr(row, ["meterage", "metrage", "meterageLabel", "areaMeterage", "propertyMeterage"]),
   612→    // B.3 (P6.5-04-v2) — لایه ۱ حقیقت: نام/موبایل مالک خام عین منبع
   613→    rawOwnerName: pickRawStr(row, ["relatedRealtorName", "realtorName", "ownerName"]),
   614→    rawOwnerMobile: pickRawStr(row, ["ownerMobile", "mobileNumber", "mobile"]),
   615→  };
   616→}

   618→/** STUB P6.5-04 — درخواست ARM → ArmRequestBulkRow (نگاشت حداقلی) */
   619→function stubMapRequestRow(row: RawRow): ArmRequestBulkRow | null {
   620→  const id = pickRawStr(row, ["id", "_id"]);
   621→  const requestCode = pickRawStr(row, ["code", "requestCode"]);

   623→  const customerName = pickRawStr(row, ["applicantName", "customerName", "fullName"]);
   624→  const customerMobile = pickRawStr(row, ["applicantMobile", "customerMobile", "mobileNumber"]);
   625→  return {
   626→    id: id ?? requestCode!,
   627→    requestCode: requestCode ?? id!,
   628→    customerName,
   629→    customerMobile,
   630→    // B.3 (P6.5-04-v2) — لایه ۱ حقیقت: نام/موبایل مشتری خام عین منبع
   631→    rawCustomerName: customerName,
   632→    rawCustomerMobile: customerMobile,
   633→  };
   634→}

   636→export function createArmBulkFetcher(init: { entity: BulkEntity; armToken: string }): BulkPageFetcher {
   637→  return {
   638→    fetchPage: async (req) => {
   639→      if (!init.armToken) {
   640→        throw new BulkFetchError("توکن ARM برای Bulk Import لازم است — نشست کاربر واردشده را تمدید کنید.", { status: 401, retryable: false });
   641→      }
   642→      const cfg = getMelkApiConfig();
   643→      const path = armEntityPath(req.entity);
   644→      const res = await armCall<unknown>({
   645→        path,
   646→        baseUrl: cfg.armApiBase,
   647→        token: init.armToken,
   648→        query: {
   649→          page: req.page,
   650→          limit: req.pageSize,
   651→          // B.4-استاتیک: نام پارامترهای بازه عین خود سامانه = startDate/endDate
   652→          // (صفحه‌های گزارش با toDate().toISOString() سریال می‌کنند — قالب ISO کامل)
   653→          ...(req.startDate ? { startDate: req.startDate.toISOString() } : {}),
   654→          ...(req.endDate ? { endDate: req.endDate.toISOString() } : {}),
   655→        },
   656→        timeoutMs: cfg.timeoutMs,
   657→        maxRetries: 0, // مدیریت retry با موتور خود خط لوله (نه client)
   658→      });
   659→
   660→      const mapped = extractPagedList(res.data, (row, i) =>
   661→        mapArmRowToInput(req.entity, (row ?? {}) as RawRow, i, req.page, req.pageSize)
   662→      );
   663→      const totalPages =
   664→        mapped.totalPages ?? (mapped.total > 0 ? Math.ceil(mapped.total / req.pageSize) : 1);
   665→      return {
   666→        rows: mapped.rows,
   667→        hasMore: mapped.rows.length > 0 && req.page < totalPages,
   668→      };
   669→    },
   670→  };
   671→}

   673→function armEntityPath(entity: BulkEntity): string {
   674→  switch (entity) {
   675→    // B.4-استاتیک: لیست معاملات از traded-count می‌آید — traded-count-total فقط
   676→    // «خلاصهٔ چهارگانه» می‌دهد (کشف ۲ شب ۱؛ عین رفع B.1 در مسیر sync)
   677→    case "transaction": return MELK_ENDPOINTS.arm.tradedCount;
   678→    case "expert": return MELK_ENDPOINTS.arm.realtorsList;
   679→    case "interaction": return MELK_ENDPOINTS.arm.realtorInteractions;
   680→    case "visit": return MELK_ENDPOINTS.arm.visitsList;
   681→    case "contract": return MELK_ENDPOINTS.arm.tradeOverviewsList;
   682→    case "listing": return MELK_ENDPOINTS.arm.propertiesList;
   683→    case "request": return MELK_ENDPOINTS.arm.requestsList;
   684→  }
   685→}

   687→/** نگاشت ردیف خام ARM → ورودی مخزن هر موجودیت (صادرشده برای تست مسیر واقعی mapping) */
   688→export function mapArmRowToInput(entity: BulkEntity, row: RawRow, index: number, page: number, pageSize: number): unknown | null {
   689→  switch (entity) {
   690→    case "transaction": {
   691→      const crm = mapArmTradedRow(row, index, page, pageSize);
   692→      return crm ? toTxUpsert(crm) : null;
   693→    }
   694→    case "expert": {
   695→      const crm = mapArmRealtorRow(row, index);
   696→      return crm ? toExUpsert(crm) : null;
   697→    }
   698→    case "interaction": {
   699→      const crm = mapArmInteractionRow(row, index);
   700→      return crm ? toInUpsert(crm) : null;
   701→    }
   702→    case "visit": {
   703→      const crm = mapArmVisitRow(row, index);
   704→      return crm ? toViUpsert(crm) : null;
   705→    }
   706→    case "contract": {
   707→      // فهرست صورت‌وضعیت‌ها — غنی‌سازی مرحله دوم (info/{id}) در P6.5-04
   708→      const crm = mapArmContractRow(row);
   709→      return crm ? toContractUpsert(crm) : null;
   710→    }
   711→    case "listing": return stubMapPropertyRow(row);
   712→    case "request": return stubMapRequestRow(row);
   713→  }
   714→}

   720→export function createRepositorySink(entity: BulkEntity, opts: { client?: PrismaClient } = {}): BulkRecordSink {
   721→  // مخازن یک‌بار ساخته می‌شوند — پیش‌فرض تولید (db از env) یا کلاینت تزریقی تست (E.3)
   722→  const txRepo = opts.client ? new PrismaTransactionRepository({ client: opts.client }) : getTransactionRepository();
   723→  const exRepo = opts.client ? new PrismaExpertRepository({ client: opts.client }) : getExpertRepository();
   724→  const viRepo = opts.client ? new PrismaVisitRepository({ client: opts.client }) : getVisitRepository();
   725→  const ctRepo = opts.client ? new PrismaContractRepository({ client: opts.client }) : getContractRepository();
   726→  const rqRepo = opts.client ? new PrismaRequestRepository({ client: opts.client }) : getRequestRepository();
   727→  const prRepo = opts.client ? new PrismaPropertyRecordRepository({ client: opts.client }) : getPropertyRecordRepository();
   728→
   729→  return {
   730→    saveBatch: async (rows, meta) => {
   731→      const up: BulkUpsertOptions = { sourceRunId: meta.sourceRunId };
   732→      switch (entity) {
   733→        case "transaction":
   734→          return txRepo.upsertBulk(rows as TransactionUpsertInput[], up);
   735→        case "expert":
   736→          return exRepo.upsertBulk(rows as ExpertUpsertInput[], up);
   737→        case "interaction":
   738→          return exRepo.upsertInteractionsBulk(rows as InteractionUpsertInput[], up);
   739→        case "visit":
   740→          return viRepo.upsertBulk(rows as VisitUpsertInput[], up);
   741→        case "contract":
   742→          return ctRepo.upsertBulk(rows as ContractUpsertInput[], up);
   743→        case "request":
   744→          return rqRepo.upsertArmBulk(rows as ArmRequestBulkRow[], up);
   745→        case "listing":
   746→          return prRepo.upsertBulk(rows as PropertyBulkInput[], up);
   747→      }
   748→    },
   749→  };
   750→}

   756→export class SyncStateBulkCheckpointStore implements BulkCheckpointStore {
   757→  private readonly repo;
   758→
   759→  constructor(deps: { client?: PrismaClient } = {}) {
   760→    // IntegrationRepository با کلاینت تزریقی (الگوی تست E.3)
   761→    this.repo = deps.client ? new PrismaIntegrationRepository({ client: deps.client }) : getIntegrationRepository();
   762→  }
   763→
   764→  async load(entity: BulkEntity): Promise<BulkImportCheckpoint | null> {
   765→    const raw = await this.repo.getBulkImportCheckpoint();
   766→    if (!raw) return null;
   767→    try {
   768→      const cp = JSON.parse(raw) as BulkImportCheckpoint;
   769→      if (cp && cp.entity === entity && cp.version === BULK_CHECKPOINT_VERSION) return cp;
   770→      return null; // موجودیت/نسخه متفاوت — نادیده (شروع تازه)
   771→    } catch {
   772→      return null; // JSON خراب — شروع تازه (fail-safe)
   773→    }
   774→  }
   775→
   776→  async save(checkpoint: BulkImportCheckpoint): Promise<void> {
   777→    await this.repo.setBulkImportCheckpoint(JSON.stringify(checkpoint));
   778→  }
