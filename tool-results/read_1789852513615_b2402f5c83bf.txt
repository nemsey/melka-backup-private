   126→/** همگام‌سازی با توکن ARM — با قفل همزمانی: اجرای موازی = پیوستن به اجرای جاری */
   127→export function syncFromArm(armToken: string): Promise<SyncResult> {
   128→  if (__armSyncInFlight) return __armSyncInFlight; // در جریان است → صبر تا اتمام (نه شروع مجدد)
   129→  __armSyncExecutionCount += 1;
   130→  __armSyncInFlight = doSyncFromArm(armToken).finally(() => {
   131→    __armSyncInFlight = null;
   132→  });
   133→  return __armSyncInFlight;
   134→}

   136→async function readSyncState() {
   137→  let state = await db.syncState.findUnique({ where: { id: SYNC_STATE_ID } });
   138→  if (!state) {
   139→    state = await db.syncState.create({ data: { id: SYNC_STATE_ID } });
   140→  }
   141→  return state;
   142→}

   144→function parseSummary(json: string | null | undefined): TradedSummary | null {
   145→  if (!json) return null;
   146→  try {
   147→    const s = JSON.parse(json) as TradedSummary;
   148→    if (typeof s.totalCount === "number") return s;
   149→    return null;
   150→  } catch {
   151→    return null;
   152→  }
   153→}

   155→/** خواندن وضعیت همگام‌سازی برای نمایش (بدون خطا پرتاب نمی‌شود) */
   156→export async function getSyncState(): Promise<SyncStateInfo> {
   157→  const state = await readSyncState();
   158→  const [txCount, exCount, inCount, viCount, sdCount] = await Promise.all([
   159→    db.transactionRecord.count(),
   160→    db.expertRecord.count(),
   161→    db.interactionRecord.count(),
   162→    db.visitRecord.count(),
   163→    db.supplyDemandRecord.count(),
   164→  ]);
   165→  return {
   166→    lastSyncedAt: state.lastSyncedAt?.toISOString() ?? null,
   167→    lastSuccessAt: state.lastSuccessAt?.toISOString() ?? null,
   168→    lastError: state.lastError,
   169→    lastErrorCode: state.lastErrorCode,
   170→    transactionCount: txCount,
   171→    expertCount: exCount,
   172→    interactionCount: inCount,
   173→    visitCount: viCount,
   174→    supplyDemandCount: sdCount,
   175→    summary: parseSummary(state.summaryJson),
   176→    attempts: state.attempts,
   177→  };
   178→}

   208→// ریشهٔ خالی‌بودن همهٔ جدول‌های production (SyncState.lastError با attempts=7):
   209→//  (۱) گزارش‌های realtor-reports/* بدون startDate+endDate → HTTP 400

   211→//  (۲) traded-count-total فقط «خلاصهٔ چهارگانه» می‌دهد (rowKeys=[] در همهٔ ۲۰۰های
   212→//      شب ۱) — لیست ردیف‌های معامله از traded-count می‌آید (count/list/page/totalPages).

   215→/**
   216→ * بازهٔ پیش‌فرض گزارش‌های realtor-reports — عین خود سامانه (کشف B.4-استاتیک):
   217→ * صفحه‌های TradedStatistics و realtor-interactions هر دو تاریخ انتخابی را با
   218→ * toDate().toISOString() سریال می‌کنند و لایهٔ درخواست (Qn) پارامترها را بدون
   219→ * هیچ تبدیلی به axios می‌دهد ⇒ قالب «ISO کامل» است، نه جلالی.
   220→ * (بازتفسیر شب ۱: 200 با totalCount=0 روی قالب جلالی — احتمال ریشهٔ قالب به‌همراه scope)
   221→ */
   222→export function buildReportDateRange(now: Date = new Date()): { startDate: string; endDate: string } {
   223→  // مبدأ مستند دستور: 2018-01-01 میلادی (= 1396/10/11 جلالی)
   224→  const startDate = "2018-01-01T00:00:00.000Z";
   225→  // مقصد = نیمه‌شب امروز تهران به ISO (عین انتخاب تاریخ «امروز» در datepicker خود سامانه)
   226→  const tehranToday = new Intl.DateTimeFormat("en-CA", {
   227→    timeZone: "Asia/Tehran",
   228→    year: "numeric",
   229→    month: "2-digit",
   230→    day: "2-digit",
   231→  }).format(now);
   232→  const endDate = new Date(`${tehranToday}T00:00:00+03:30`).toISOString();
   233→  return { startDate, endDate };
   234→}

   236→/** خلاصهٔ چهارگانهٔ گزارش تطابق از traded-count-total (تک‌تماس، با بازه) */
   237→async function fetchTradedSummary(
   238→  armToken: string,
   239→  range: { startDate: string; endDate: string }
   240→): Promise<TradedSummary | null> {
   241→  try {
   242→    const cfg = getMelkApiConfig();
   243→    const res = await armCall<unknown>({
   244→      path: MELK_ENDPOINTS.arm.tradedTotal,
   245→      baseUrl: cfg.armApiBase,
   246→      token: armToken,
   247→      query: { ...range },
   248→      timeoutMs: cfg.timeoutMs,
   249→      maxRetries: 0,
   250→    });
   251→    return mapArmTradedSummary(res.data ?? null);
   252→  } catch {
   253→    return null; // خلاصهٔ غیربحرانی است — در نبودش از خود ردیف‌ها محاسبه می‌شود
   254→  }
   255→}
   256→
   257→/**
   258→ * خواندن یک گزارش صفحه‌بندی‌شده تا انتها (با سقف ایمن صفحات).
   259→ * baseQuery = پارامترهای الزامی گزارش (مثل بازهٔ تاریخی) که در همهٔ صفحات تکرار می‌شوند.
   260→ */
   261→async function fetchAllPages<T>(
   262→  path: string,
   263→  armToken: string,
   264→  maxPages: number,
   265→  pageLimit: number,
   266→  mapRow: (row: Record<string, unknown>, index: number, page: number, limit: number) => T | null,
   267→  baseQuery: Record<string, string | number> = {}
   268→): Promise<{ rows: T[] }> {
   269→  const cfg = getMelkApiConfig();
   270→  const rows: T[] = [];
   271→  for (let page = 1; page <= maxPages; page++) {
   272→    const res = await armCall<unknown>({
   273→      path,
   274→      baseUrl: cfg.armApiBase,
   275→      token: armToken,
   276→      query: { ...baseQuery, page, limit: pageLimit },
   277→      timeoutMs: cfg.timeoutMs,
   278→      maxRetries: 0, // در حلقه صفحه‌بندی تلاش مجدد نمی‌کنیم (صفحه بعدی خودش تازه می‌شود)
   279→    });
   280→    const { rows: pageRows, total, totalPages } = extractPagedList(res.data, (row, i) => mapRow(row, i, page, pageLimit));
   281→    rows.push(...pageRows);
   282→    const pages = totalPages ?? (total > 0 ? Math.ceil(total / pageLimit) : 1);
   283→    if (page >= pages || pageRows.length === 0) break;
   284→    // احتیاط توصیه‌شدهٔ ملکا-شورا-۰۰۷ — فاصلهٔ کوتاه بین صفحات: rate آزاد، ولی
   285→    // بدون burst (پیک صبح شنبهٔ شب ۱ نشان داد لبهٔ مبدأ در فشار لحظه‌ای سیاه‌چاله می‌شود)
   286→    await new Promise((r) => setTimeout(r, 400));
   287→  }
   288→  return { rows };
   289→}

   291→/**
   292→ * استخراج کامل معاملات، کارشناسان، تعاملات، بازدیدها و عرضه/تقاضا از سامانه ARM
   293→ * با توکن ARM. صفحه‌بندی به‌طور خودکار طی می‌شود (تا سقف ایمن صفحات).
   294→ * (پیاده‌سازی داخلی — همیشه از syncFromArm استفاده کنید تا قفل همزمانی اعمال شود)
   295→ */
   296→async function doSyncFromArm(armToken: string): Promise<SyncResult> {
   297→  const now = new Date();
   298→  const incremental = syncMode() === "incremental";
   299→  const runId = randomUUID();
   300→  const syncStartedAt = Date.now();
   301→  await writeSyncState({
   302→    lastSyncedAt: now,
   303→    // شناسه اجرا فقط در حالت افزایشی معنا دارد (legacy به آن دست نمی‌زند)
   304→    ...(incremental ? { syncRunId: runId } : {}),
   305→  });
   306→  // P2-05 — رویداد الزامی همگام‌سازی: آغاز
   307→  const rangeForLog = buildReportDateRange();
   308→  logSync("sync-run", { status: "start", mode: incremental ? "incremental" : "legacy", detail: { runId, reportRange: rangeForLog } });
   309→
   310→  try {
   311→    // ---------------- ۱) گزارش تطابق معاملات + خلاصه ----------------
   312→    // رفع B.1: بازهٔ الزامی ارسال می‌شود و «لیست» از traded-count می‌آید
   313→    // (traded-count-total فقط خلاصهٔ چهارگانه است — کشف شب ۱).
   314→    const range = buildReportDateRange();
   315→    const { rows: transactions } = await fetchAllPages(
   316→      MELK_ENDPOINTS.arm.tradedCount,
   317→      armToken,
   318→      50,
   319→      100,
   320→      mapArmTradedRow,
   321→      range
   322→    );
   323→    const endpointSummary = await fetchTradedSummary(armToken, range);
   324→    const summary = resolveTradedSummary(endpointSummary, transactions);
   325→
   326→    // ---------------- ۲) فهرست کاربران/مشاوران ----------------
   327→    const { rows: experts } = await fetchAllPages(MELK_ENDPOINTS.arm.realtorsList, armToken, 30, 100, mapArmRealtorRow);
   328→
   329→    // ---------------- ۳) تعاملات مشاوران با یکدیگر ----------------
   330→    let interactions: CrmInteraction[] = [];
   331→    let interactionsFetched = false;
   332→    try {
   333→      // B.4-استاتیک: صفحهٔ خود سامانه کوئری را فقط با (startDate && endDate) || realtor
   334→      // فعال می‌کند ⇒ بازهٔ گزارش مانند خانوادهٔ realtor-reports ارسال می‌شود
   335→      const { rows: inRows } = await fetchAllPages(MELK_ENDPOINTS.arm.realtorInteractions, armToken, 20, 200, mapArmInteractionRow, range);
   336→      interactions = inRows;
   337→      interactionsFetched = true;
   338→    } catch {
   339→      // تعاملات گزارش مکمل است — نبودش نباید کل همگام‌سازی را شکست بدهد
   340→      interactions = [];
   341→    }
   342→
   343→    // ---------------- ۴) بازدیدها و پیگیری‌ها (غیربحرانی) ----------------
   344→    let visits: CrmVisit[] = [];
   345→    let visitsFetched = false;
   346→    try {
   347→      const { rows: viRows } = await fetchAllPages(MELK_ENDPOINTS.arm.visitsList, armToken, 20, 100, mapArmVisitRow);
   348→      visits = viRows;
   349→      visitsFetched = true;
   350→    } catch {
   351→      visits = [];
   352→    }
   353→
   354→    // ---------------- ۵) گزارش عرضه و تقاضا (غیربحرانی) ----------------
   355→    let supplyDemand: CrmSupplyDemand[] = [];
   356→    let supplyDemandFetched = false;
   357→    try {
   358→      // supply-demand هم‌خانوادهٔ realtor-reports است ⇒ بازهٔ الزامی می‌گیرد
   359→      // (حدس مستندشدهٔ B.1 — اگر سرور پارامتر اضافی را نادیده بگیرد بی‌ضرر است؛
   360→      // بدون بازه این خانواده 400 می‌دهد. شکست همچنان ساکت-fail غیربحرانی است.)
   361→      const { rows: sdRows } = await fetchAllPages(MELK_ENDPOINTS.armReports.supplyDemand, armToken, 10, 100, mapArmSupplyDemandRow, range);
   362→      supplyDemand = sdRows;
   363→      supplyDemandFetched = true;
   364→    } catch {
   365→      supplyDemand = [];
   366→    }
   367→
   368→    // ---------------- ۶) قراردادها و اسناد (صورت وضعیت‌ها) (غیربحرانی) ----------------
   369→    // لیست + جزئیات هر قرارداد (خریداران، فروشندگان، حق‌السهم مشاوران)
   370→    let contracts: CrmContract[] = [];
   371→    let contractsFetched = false;
   372→    try {
   373→      contracts = await fetchContracts(armToken);
   374→      contractsFetched = true;
   375→    } catch {
   376→      contracts = [];
   377→    }
   378→
   379→    // ---------------- ذخیره در پایگاه محلی ----------------
   380→    if (incremental) {
   381→      // P2-04 — مسیر افزایشی: upsert بر کلید طبیعی + نقطه ازسرگیری + نامه مرده
   382→      await applyIncrementalSync(runId, {
   383→        transactions, experts, interactions, visits, supplyDemand, contracts,
   384→        interactionsFetched, visitsFetched, supplyDemandFetched, contractsFetched,
   385→      });
   386→    } else {
   387→      // مسیر legacy — عین کد قبلی: پاک‌سازی کامل + درج اتمیک
   388→      await db.$transaction([
   389→        db.transactionRecord.deleteMany({}),
   390→        db.expertRecord.deleteMany({}),
   391→        db.interactionRecord.deleteMany({}),
   392→        db.visitRecord.deleteMany({}),
   393→        db.supplyDemandRecord.deleteMany({}),
   394→        db.contractRecord.deleteMany({}),
   395→        ...toTxCreates(transactions),
   396→        ...toExCreates(experts),
   397→        ...toInCreates(interactions),
   398→        ...toViCreates(visits),
   399→        ...toSdCreates(supplyDemand, runId),
   400→        ...toContractCreates(contracts),
   401→      ]);
   402→    }
   403→
   404→    const successAt = new Date();
   405→    await writeSyncState({
   406→      lastSuccessAt: successAt,
   407→      lastError: null,
   408→      lastErrorCode: null,
   409→      attempts: 0,
   410→      summaryJson: summary ? JSON.stringify(summary) : null,
   411→      ...(incremental ? { lastWatermark: successAt } : {}),
   412→    });
   413→    // P2-05 — رویداد الزامی همگام‌سازی: پایان موفق
   414→    logSync("sync-run", {
   415→      status: "ok",
   416→      mode: incremental ? "incremental" : "legacy",
   417→      duration: Date.now() - syncStartedAt,
   418→      detail: {
   419→        runId,
   420→        transactions: transactions.length,
   421→        experts: experts.length,
   422→        interactions: interactions.length,
   423→        visits: visits.length,
   424→        supplyDemand: supplyDemand.length,
   425→        contracts: contracts.length,
   426→      },
   427→    });
   428→
   429→    return {
   430→      ok: true,
   431→      transactions: transactions.length,
   432→      experts: experts.length,
   433→      interactions: interactions.length,
   434→      visits: visits.length,
   435→      supplyDemand: supplyDemand.length,
   436→      contracts: contracts.length,
   437→      syncedAt: successAt.toISOString(),
   438→    };
   439→  } catch (err) {
   440→    const e = err instanceof MelkApiError ? err : new MelkApiError("UNKNOWN", (err as Error)?.message ?? "خطای نامشخص", 500, false);
   441→    const state = await readSyncState();
   442→    await writeSyncState({
   443→      lastError: e.message,
   444→      lastErrorCode: e.code,
   445→      attempts: state.attempts + 1,
   446→    });
   447→    // P2-05 — رویداد الزامی همگام‌سازی: شکست
   448→    logSync("sync-run", {
   449→      status: "fail",
   450→      mode: incremental ? "incremental" : "legacy",
   451→      duration: Date.now() - syncStartedAt,
   452→      errorCode: e.code,
   453→      detail: { runId },
   454→    });
   455→    return {
   456→      ok: false,
   457→      transactions: 0,
   458→      experts: 0,
   459→      interactions: 0,
   460→      visits: 0,
   461→      supplyDemand: 0,
   462→      contracts: 0,
   463→      syncedAt: now.toISOString(),
   464→      error: e.message,
   465→      errorCode: e.code,
   466→    };
   467→  }
   468→}

   470→/** اگر سامانه خلاصه چهارگانه را نفرستاد، از ردیف‌ها محاسبه می‌شود */
   471→function computeSummaryFromRows(rows: CrmTransaction[]): TradedSummary {
   472→  return {
   473→    totalCount: rows.length,
   474→    hasRequestEvent: rows.filter((r) => r.eventsCount > 0).length,
   475→    hasSuggestedProperty: rows.filter((r) => r.suggestedPropertyCount > 0).length,
   476→    hasBoth: rows.filter((r) => r.eventsCount > 0 && r.suggestedPropertyCount > 0).length,
   477→  };
   478→}

   480→/**
   481→ * انتخاب خلاصهٔ چهارگانهٔ معاملات — شاهد شب ۲ (recon-night2، ۱۴۰۵/۰۶/۲۹):
   482→ * خلاصهٔ endpoint (traded-count-total) برای حساب realtor حتی با بازهٔ ISO درست هم
   483→ * totalCount=0 برمی‌گرداند (خلاصهٔ scope-محور خود آن مسیر)، در حالی که فهرست واقعی
   484→ * (traded-count) همان لحظه ۷۲۴+ ردیف داده می‌دهد. اصل «دادهٔ صادقانه»:
   485→ *  - خلاصهٔ endpoint معتبر و غیرصفر → همان (عین رفتار قبلی)
   486→ *  - خلاصهٔ صفر ولی ردیف واقعی موجود → محاسبه از خود ردیف‌ها (رفع نمایش غلط صفر)
   487→ *  - خلاصهٔ صفر و ردیف هم صفر → همان صفرِ صادقانهٔ endpoint
   488→ *  - خطای endpoint (null) → محاسبه از ردیف‌ها (عین رفتار قبلی)
   489→ */
   490→export function resolveTradedSummary(
   491→  endpointSummary: TradedSummary | null,
   492→  transactions: CrmTransaction[]
   493→): TradedSummary {
   494→  if (endpointSummary === null) return computeSummaryFromRows(transactions);
   495→  if (endpointSummary.totalCount === 0 && transactions.length > 0) {
   496→    return computeSummaryFromRows(transactions);
   497→  }
   498→  return endpointSummary;
   499→}

   644→/** مشخصات یک خانواده داده برای مسیر افزایشی */
   645→interface IncrementalFamily<T> {
   646→  entity: "transactions" | "experts" | "interactions" | "visits" | "supplyDemand" | "contracts";
   647→  rows: T[];
   648→  /** واکشی این خانواده موفق بوده؟ (شکست واکشی ⇒ خانواده دست‌نخورده می‌ماند — تفاوت عمدی با legacy) */
   649→  fetched: boolean;
   650→  keyOf: (row: T) => string;
   651→  upsert: (rows: T[]) => Promise<unknown>;
   652→  /** کلیدهای موجود در پایگاه (برای جاروی سنگ‌قبرها) */
   653→  listExistingKeys: () => Promise<string[]>;
   654→  deleteByKeys: (keys: string[]) => Promise<unknown>;
   655→}
   656→
   657→/**
   658→ * اعمال افزایشی: به‌جای پاک‌سازی کامل، upsert بر کلید طبیعی + جاروی
   659→ * ردیف‌های محو‌شده (با محافظت نامه‌های مرده باز) + نقطه ازسرگیری پس از
   660→ * هر خانواده. شکست ردیف منفرد کل اجرا را نمی‌شکند (نامه مرده می‌شود).
   661→ */
   662→/** صادرات برای تست (tests/phase2-sync.test.ts) — در کد محصول از syncFromArm استفاده کنید */
   663→export async function applyIncrementalSync(
   664→  runId: string,
   665→  data: {
   666→    transactions: CrmTransaction[];
   667→    experts: CrmExpert[];
   668→    interactions: CrmInteraction[];
   669→    visits: CrmVisit[];
   670→    supplyDemand: CrmSupplyDemand[];
   671→    contracts: CrmContract[];
   672→    interactionsFetched: boolean;
   673→    visitsFetched: boolean;
   674→    supplyDemandFetched: boolean;
   675→    contractsFetched: boolean;
   676→  }
   677→): Promise<void> {
   678→  const txRepo = getTransactionRepository();
   679→  const exRepo = getExpertRepository();
   680→  const mkRepo = getMarketRepository();
   681→  const ctRepo = getContractRepository();
   682→  const viRepo = getVisitRepository();
   683→
   684→  const families: IncrementalFamily<never>[] = [
   685→    {
   686→      entity: "transactions",
   687→      rows: data.transactions.map(toTxUpsert) as never[],
   688→      fetched: true,
   689→      keyOf: (t) => (t as unknown as TransactionUpsertInput).id,
   690→      upsert: (rows) => txRepo.upsertBatch(rows as unknown as TransactionUpsertInput[]),
   691→      listExistingKeys: async () => (await db.transactionRecord.findMany({ where: { sourceRunId: null }, select: { id: true } })).map((r) => r.id),
   692→      deleteByKeys: (keys) => db.transactionRecord.deleteMany({ where: { id: { in: keys } } }),
   693→    },
   694→    {
   695→      entity: "experts",
   696→      rows: data.experts.map(toExUpsert) as never[],
   697→      fetched: true,
   698→      keyOf: (e) => (e as unknown as ExpertUpsertInput).personnelCode,
   699→      upsert: (rows) => exRepo.upsertBatch(rows as unknown as ExpertUpsertInput[]),
   700→      listExistingKeys: async () => (await db.expertRecord.findMany({ where: { sourceRunId: null }, select: { personnelCode: true } })).map((r) => r.personnelCode),
   701→      deleteByKeys: (keys) => db.expertRecord.deleteMany({ where: { personnelCode: { in: keys } } }),
   702→    },
   703→    {
   704→      entity: "interactions",
   705→      rows: data.interactions.map(toInUpsert) as never[],
   706→      fetched: data.interactionsFetched,
   707→      keyOf: (i) => (i as unknown as InteractionUpsertInput).id,
   708→      upsert: (rows) => exRepo.upsertInteractionsBatch(rows as unknown as InteractionUpsertInput[]),
   709→      listExistingKeys: async () => (await db.interactionRecord.findMany({ where: { sourceRunId: null }, select: { id: true } })).map((r) => r.id),
   710→      deleteByKeys: (keys) => db.interactionRecord.deleteMany({ where: { id: { in: keys } } }),
   711→    },
   712→    {
   713→      entity: "visits",
   714→      rows: data.visits.map(toViUpsert) as never[],
   715→      fetched: data.visitsFetched,
   716→      keyOf: (v) => (v as unknown as VisitUpsertInput).id,
   717→      upsert: (rows) => viRepo.upsertBatch(rows as unknown as VisitUpsertInput[]),
   718→      listExistingKeys: async () => (await db.visitRecord.findMany({ where: { sourceRunId: null }, select: { id: true } })).map((r) => r.id),
   719→      deleteByKeys: (keys) => db.visitRecord.deleteMany({ where: { id: { in: keys } } }),
   720→    },
   721→    {
   722→      entity: "supplyDemand",
   723→      rows: data.supplyDemand.map((s) => toSdUpsert(s, runId)) as never[],
   724→      fetched: data.supplyDemandFetched,
   725→      keyOf: (s) => (s as unknown as SupplyDemandUpsertInput).id,
   726→      upsert: (rows) => mkRepo.upsertSupplyDemandBatch(rows as unknown as SupplyDemandUpsertInput[]),
   727→      listExistingKeys: async () => (await db.supplyDemandRecord.findMany({ select: { id: true } })).map((r) => r.id),
   728→      deleteByKeys: (keys) => db.supplyDemandRecord.deleteMany({ where: { id: { in: keys } } }),
   729→    },
   730→    {
   731→      entity: "contracts",
   732→      rows: data.contracts.map(toContractUpsert) as never[],
   733→      fetched: data.contractsFetched,
   734→      keyOf: (c) => (c as unknown as ContractUpsertInput).id,
   735→      upsert: (rows) => ctRepo.upsertBatch(rows as unknown as ContractUpsertInput[]),
   736→      listExistingKeys: async () => (await db.contractRecord.findMany({ where: { sourceRunId: null }, select: { id: true } })).map((r) => r.id),
   737→      deleteByKeys: (keys) => db.contractRecord.deleteMany({ where: { id: { in: keys } } }),
   738→    },
   739→  ];
   740→
   741→  const done: Array<{ entity: string; count: number; failed: number }> = [];
   742→
   743→  // شناسه اجرا از همان ابتدا ثبت می‌شود (خودکفا بودن applier برای تست/فراخوانی مستقیم)
   744→  await writeSyncState({ syncRunId: runId });
   745→
   746→  for (const family of families) {
   747→    // واکشی ناموفق ⇒ این خانواده در این اجرا دست‌نخورده می‌ماند (از دست رفتن داده نداریم)
   748→    if (!family.fetched) {
   749→      done.push({ entity: family.entity, count: 0, failed: -1 }); // -1 = skipped
   750→      continue;
   751→    }
   752→
   753→    const keys = family.rows.map(family.keyOf);
   754→    let failedKeys: string[] = [];
   755→
   756→    try {
   757→      // تلاش دسته‌ای (مسیر سریع — اکثر اجراها موفق‌اند)
   758→      await family.upsert(family.rows);
   759→    } catch {
   760→      // شکست دسته‌ای ⇒ تک‌تک برای ایزوله کردن ردیف‌های خراب (upsertها idempotentند)
   761→      for (let i = 0; i < family.rows.length; i++) {
   762→        try {
   763→          await family.upsert([family.rows[i]]);
   764→        } catch (err) {
   765→          failedKeys.push(keys[i]);
   766→          await recordDeadLetter(runId, family.entity, keys[i], err);
   767→        }
   768→      }
   769→    }
   770→
   771→    // ردیف‌های موفق این خانواده که قبلاً نامه مرده بودند → حل‌شده
   772→    await resolveDeadLetters(family.entity, keys.filter((k) => !failedKeys.includes(k)));
   773→
   774→    // جاروی سنگ‌قبرها: ردیف‌هایی که دیگر در سامانه نیستند (به‌جز نامه‌های مرده باز)
   775→    // P6.5-03 — رکوردهای Bulk (sourceRunId غیرتهی) از جارو محافظت می‌شوند:
   776→    // داده تاریخیِ خارج از پوشش واکشی sync است؛ حذفش = حفره داده غیرقابل ترمیم (کشف §۱.۹).
   777→    try {
   778→      const existing = await family.listExistingKeys();
   779→      const fetchedSet = new Set(keys);
   780→      const openDead = await openDeadLetterKeys(family.entity);
   781→      const stale = existing.filter((k) => !fetchedSet.has(k) && !openDead.has(k));
   782→      if (stale.length > 0) await family.deleteByKeys(stale);
   783→    } catch {
   784→      // جارو شکست خورد ⇒ اجرا همچنان موفق است؛ دفع بعد تمیز می‌کند
   785→    }
   786→
   787→    // نقطه ازسرگیری — پس از هر خانواده به‌روزرسانی می‌شود
   788→    done.push({ entity: family.entity, count: family.rows.length, failed: failedKeys.length });
   789→    await writeSyncState({
   790→      lastSyncCursor: JSON.stringify({ runId, done: done.map((d) => d.entity), detail: done }),
   791→    });
   792→  }
   793→}

   795→/**
   796→ * همگام‌سازی با توکن ARM نشست جاری — اگر توکن منقضی شده باشد
   797→ * ابتدا با refreshToken تمدید می‌شود.
   798→ */
   799→export async function syncWithRefresh(
   800→  armToken: string,
   801→  armRefreshToken: string
   802→): Promise<{ sync: SyncResult; newArmToken?: string }> {
   803→  const provider = getMelkProvider();
   804→  let token = armToken;
   805→  let newArmToken: string | undefined;
   806→
   807→  // تلاش اول؛ در صورت انقضا، تمدید و تلاش مجدد
   808→  let sync = await syncFromArm(token);
   809→  if (!sync.ok && (sync.errorCode === "UNAUTHORIZED" || sync.errorCode === "OTP_INVALID") && armRefreshToken) {
   810→    try {
   811→      const refreshed = await provider.refreshArmToken(armRefreshToken);
   812→      token = refreshed.accessToken;
   813→      newArmToken = refreshed.accessToken;
   814→      sync = await syncFromArm(token);
   815→    } catch (refreshErr) {
   816→      const e = refreshErr instanceof MelkApiError ? refreshErr : new MelkApiError("UNKNOWN", "تمدید توکن ناموفق", 401, false);
   817→      sync = { ok: false, transactions: 0, experts: 0, interactions: 0, visits: 0, supplyDemand: 0, contracts: 0, syncedAt: new Date().toISOString(), error: e.message, errorCode: e.code };
   818→    }
   819→  }
   820→  return { sync, newArmToken };
   821→}

  1038→async function readRefCache<T>(key: string): Promise<T | null> {
  1039→  const row = await db.refListCache.findUnique({ where: { key } });
  1040→  if (!row) return null;
  1041→  const age = Date.now() - row.fetchedAt.getTime();
  1042→  if (age > REF_TTL_MS) return null;
  1043→  try {
  1044→    return JSON.parse(row.dataJson) as T;
  1045→  } catch {
  1046→    return null;
  1047→  }
  1048→}
