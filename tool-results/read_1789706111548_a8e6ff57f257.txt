     1→/**
     2→ * Bulk Import Pipeline — P6.5-03 (طراحی §۱ PHASE-6.5 + دستور شورا P6.5-03)
     3→ * ------------------------------------------------------
     4→ * خط لوله دریافت داده انبوه از ARM/PM با چهار ستون فنی:
     5→ *   ۱) Checkpoint/Resume — ادامه از آخرین صفحهٔ «کامل-ذخیره‌شده» (SyncState.bulkImportCheckpoint)
     6→ *   ۲) Idempotency — upsert بر کلید طبیعی + merge non-null (§۲.۴) ⇒ اجرای دوباره = بی‌ضرر
     7→ *   ۳) Rate Limiting — حداقل فاصله «پایان پاسخ قبلی تا شروع درخواست بعدی» (§۱.۸)
     8→ *   ۴) Retry — ۵ تلاش با backoff نمایی 2s→4s→8s→16s→30s روی 5xx/429/شبکه (§۱.۷)
     9→ *
    10→ * اصول حاکم (قواعد بخش C دستور):
    11→ *   · صفر LLM call · صفر cost · صفر fetch مستقیم (فقط melk-api/client) · صفر Prisma مستقیم (فقط Repository)
    12→ *   · صفر فشار به سامانه مبدأ — rate limiting اجباری (default 200ms)
    13→ *   · Bulk هرگز deleteMany صادر نمی‌کند (§۱.۹) و فیلدهای sync منظم را لمس نمی‌کند (§۱.۵)
    14→ *
    15→ * معماری پورت‌ها (تست‌پذیری کامل با موک — §۱.۱۰: توسعه با موک، صفر فشار به API مبدأ):
    16→ *   BulkPageFetcher  = سازگار منبع (ARM/PM → ردیف‌های نگاشت‌شده UpsertInput)
    17→ *   BulkRecordSink   = ماندگاری (Repositoryهای upsertBulk با merge non-null)
    18→ *   BulkCheckpointStore = نقطه ازسرگیری (SyncState از طریق IntegrationRepository)
    19→ *
    20→ * انحراف صادقانه مستند (۱): پس از اتمام retryهای یک صفحه، نتیجه FAILED بازگردانده می‌شود و
    21→ * صفحه‌های بعدی طی نمی‌شوند — بند «continue به صفحه بعد» دستور با بند ۳ همان دستور
    22→ * (ذخیره checkpoint + توقف) در تعارض است و ادامۀ صفحه‌ها حفره داده می‌سازد که checkpoint
    23→ * قادر به ترمیمش نیست (نقض صفر حذف داده). انتخاب: مسیر طراحی مصوب §۱.۴-۳.
    24→ * status=PARTIAL برای خطاهای «رکورد-منفرد» درون صفحات موفق رزرو شده است.
    25→ */

   180→/** نسخهٔ قرارداد checkpoint (سازگاری آینده — تغییر = نادیده‌گرفتن checkpoint قدیمی) */
   181→export const BULK_CHECKPOINT_VERSION = "p65-03-v1";

   183→/** عین دستور: 2s→4s→8s→16s→30s (سقف 30s برای تلاش‌های بیشتر) */
   184→export const DEFAULT_RETRY_DELAYS_MS: readonly number[] = [2000, 4000, 8000, 16000, 30000];
   185→
   186→/** تأخیر پیش از تلاش retry شماره n (1-based) — null یعنی «دیگر retry نکن» */
   187→export function retryDelayFor(retryNumber: number, maxRetries: number): number | null {
   188→  if (retryNumber < 1) return null;
   189→  if (retryNumber > maxRetries) return null;
   190→  const idx = Math.min(retryNumber - 1, DEFAULT_RETRY_DELAYS_MS.length - 1);
   191→  return DEFAULT_RETRY_DELAYS_MS[idx];
   192→}

   194→/** خطای شبکه‌ای/HTTP با حمایت Retry-After — قابل‌تزریق در تست */
   195→export class BulkFetchError extends Error {
   196→  readonly status?: number;
   197→  readonly retryable: boolean;
   198→  readonly retryAfterMs?: number;
   199→  constructor(message: string, opts: { status?: number; retryable: boolean; retryAfterMs?: number } = { retryable: false }) {
   200→    super(message);
   201→    this.name = "BulkFetchError";
   202→    this.status = opts.status;
   203→    this.retryable = opts.retryable;
   204→    this.retryAfterMs = opts.retryAfterMs;
   205→  }
   206→}

   208→/** §۱.۷ — قابل‌retry: 5xx/429/شبکه (پرچم retryable)؛ 4xx غیر 429 = فوری؛ ناشناخته = فوری */
   209→export function classifyFetchError(err: unknown): { retryable: boolean; retryAfterMs: number | null } {
   210→  if (err instanceof BulkFetchError) {
   211→    return { retryable: err.retryable, retryAfterMs: err.retryAfterMs ?? null };
   212→  }
   213→  if (err instanceof MelkApiError) {
   214→    const retryable = err.retryable === true || err.httpStatus >= 500 || err.httpStatus === 429;
   215→    return { retryable, retryAfterMs: null };
   216→  }
   217→  return { retryable: false, retryAfterMs: null };
   218→}

   220→/** §۱.۳ — clamp بازه‌های مجاز (fail-fast معکوس: مقدار نامعتبر → default امن) */
   221→export function clampPageSize(raw?: number): number {
   222→  const n = Number.isFinite(raw) ? Math.floor(raw as number) : 100;
   223→  return Math.min(500, Math.max(100, n));
   224→}
   225→
   226→export function clampBatchSize(raw?: number): number {
   227→  const n = Number.isFinite(raw) ? Math.floor(raw as number) : 1000;
   228→  return Math.min(2000, Math.max(100, n));
   229→}

   231→function defaultSleep(ms: number): Promise<void> {
   232→  return new Promise((resolve) => setTimeout(resolve, ms));
   233→}

   235→// ==========================================================================
   236→// RateLimiter — §۱.۸: فاصله از «پایان پاسخ قبلی» تا «شروع درخواست بعدی»
   237→// ==========================================================================
   238→
   239→export class RateLimiter {
   240→  private lastCompletedAt = 0;
   241→
   242→  constructor(private readonly minIntervalMs: number) {}
   243→
   244→  /** پیش از هر درخواست — نخستین درخواست آزاد است */
   245→  async wait(): Promise<void> {
   246→    if (this.lastCompletedAt === 0) return;
   247→    const elapsed = Date.now() - this.lastCompletedAt;
   248→    const remaining = this.minIntervalMs - elapsed;
   249→    if (remaining > 0) await defaultSleep(remaining);
   250→  }
   251→
   252→  /** پس از پایان هر پاسخ (موفق یا ناموفق) */
   253→  complete(): void {
   254→    this.lastCompletedAt = Date.now();
   255→  }
   256→}

   262→/** سقف ایمنی صفحات (حلقه بی‌نهایت منبع خراب را نمی‌پذیرد) — additive option */
   263→const DEFAULT_MAX_PAGES = 10_000;
   264→
   265→class BulkImportRunner {
   266→  private readonly opts: BulkImportOptions;
   267→  private readonly pageSize: number;
   268→  private readonly batchSize: number;
   269→  private readonly maxRetries: number;
   270→  private readonly deps: BulkImportDeps;
   271→  private readonly limiter: RateLimiter;
   272→  private readonly sleep: (ms: number) => Promise<void>;
   273→
   274→  constructor(options: BulkImportOptions, deps: BulkImportDeps) {
   275→    this.opts = options;
   276→    this.pageSize = clampPageSize(options.pageSize);
   277→    this.batchSize = clampBatchSize(options.batchSize);
   278→    this.maxRetries = options.maxRetries ?? 5;
   279→    this.deps = deps;
   280→    this.limiter = new RateLimiter(Math.max(1, options.rateLimitMs ?? 200));
   281→    this.sleep = deps.sleep ?? defaultSleep;
   282→  }
   283→
   284→  private now(): Date {
   285→    return this.deps.now ? this.deps.now() : new Date();
   286→  }
   287→
   288→  async run(): Promise<BulkImportResult> {
   289→    const t0 = this.now().getTime();
   290→    const entity = this.opts.entity;
   291→    const range =
   292→      this.opts.startDate || this.opts.endDate
   293→        ? {
   294→            from: (this.opts.startDate ?? this.opts.endDate)!.toISOString(),
   295→            to: (this.opts.endDate ?? this.opts.startDate)!.toISOString(),
   296→          }
   297→        : null;
   298→
   299→    // ---- ۱) بارگذاری checkpoint (تطبیق entity + بازه + نسخه) ----
   300→    const saved = await this.deps.checkpointStore.load(entity);
   301→    const resumable =
   302→      saved !== null &&
   303→      saved.version === BULK_CHECKPOINT_VERSION &&
   304→      sameDateRange(saved.dateRange ?? null, range);
   305→
   306→    const cp: BulkImportCheckpoint = resumable
   307→      ? { ...saved, runId: saved.runId ?? this.deps.newRunId?.() ?? randomUUID() }
   308→      : {
   309→          entity,
   310→          lastPage: 0,
   311→          lastRecordId: "",
   312→          totalProcessed: 0,
   313→          startedAt: this.now().toISOString(),
   314→          version: BULK_CHECKPOINT_VERSION,
   315→          runId: this.deps.newRunId?.() ?? randomUUID(),
   316→          updatedAt: this.now().toISOString(),
   317→          dateRange: range,
   318→          totals: { fetched: 0, saved: 0, duplicates: 0, errors: 0 },
   319→        };
   320→
   321→    const progress: BulkImportProgress = {
   322→      entity,
   323→      totalFetched: cp.totals?.fetched ?? 0,
   324→      totalSaved: cp.totals?.saved ?? 0,
   325→      totalDuplicates: cp.totals?.duplicates ?? 0,
   326→      totalErrors: cp.totals?.errors ?? 0,
   327→      currentPage: cp.lastPage,
   328→      currentBatch: 0,
   329→      checkpoint: cp,
   330→      startedAt: new Date(cp.startedAt),
   331→      updatedAt: this.now(),
   332→    };
   333→
   334→    let status: BulkImportResult["status"] = "COMPLETED";
   335→    let error: string | undefined;
   336→
   337→    // ---- ۲) حلقه صفحات ----
   338→    let page = cp.lastPage + 1; // نقطه ادامه = آخرین صفحه کامل + ۱
   339→    let pagesRan = 0;
   340→
   341→    while (true) {
   342→      if (pagesRan >= DEFAULT_MAX_PAGES) {
   343→        // سقف ایمنی — از سر رفتن حلقه منبع خراب؛ داده‌های تا اینجا سالم‌اند
   344→        status = "PARTIAL";
   345→        error = `سقف ایمنی صفحات (${DEFAULT_MAX_PAGES}) رسید؛ داده تا صفحه ${cp.lastPage} سالم ذخیره شد.`;
   346→        break;
   347→      }
   348→
   349→      let fetched: BulkFetchPage;
   350→      try {
   351→        fetched = await this.fetchPageWithRetry(page);
   352→      } catch (err) {
   353→        // §۱.۴-۳ — checkpoint (آخرین صفحه کامل) ذخیره + توقف با خطای صریح فارسی
   354→        await this.deps.checkpointStore.save(cp);
   355→        status = "FAILED";
   356→        error = fetchFailureMessage(err, entity, page, cp.lastRecordId);
   357→        break;
   358→      }
   359→
   360→      progress.totalFetched += fetched.rows.length;
   361→      pagesRan += 1;
   362→
   363→      if (fetched.rows.length === 0) break; // پایان داده
   364→
   365→      // ---- ذخیره: برش‌های batchSize + فلاش اجباری پیش از پیشروی checkpoint ----
   366→      for (let offset = 0; offset < fetched.rows.length; offset += this.batchSize) {
   367→        const chunk = fetched.rows.slice(offset, offset + this.batchSize);
   368→        const stats = await this.deps.sink.saveBatch(chunk, { sourceRunId: cp.runId!, entity });
   369→        progress.totalSaved += stats.created + stats.updated;
   370→        progress.totalDuplicates += stats.duplicates;
   371→        progress.totalErrors += stats.errors;
   372→        progress.currentBatch += 1;
   373→      }
   374→
   375→      // نا-تکرارپذیری فلاش/چک‌پوینت: صفحه کامل شد → داده نوشته شد → حالا checkpoint جلو می‌رود
   376→      const lastRow = fetched.rows[fetched.rows.length - 1] as Record<string, unknown> | undefined;
   377→      cp.lastRecordId = String(lastRow?.id ?? lastRow?.propertyCode ?? lastRow?.requestCode ?? lastRow?.personnelCode ?? "");
   378→      cp.totalProcessed = progress.totalFetched;
   379→      cp.lastPage = page;
   380→      cp.updatedAt = this.now().toISOString();
   381→      cp.totals = {
   382→        fetched: progress.totalFetched,
   383→        saved: progress.totalSaved,
   384→        duplicates: progress.totalDuplicates,
   385→        errors: progress.totalErrors,
   386→      };
   387→      await this.deps.checkpointStore.save(cp);
   388→
   389→      progress.currentPage = page;
   390→      progress.checkpoint = { ...cp };
   391→      progress.updatedAt = this.now();
   392→      this.opts.onProgress?.({ ...progress, checkpoint: { ...cp, totals: cp.totals ? { ...cp.totals } : undefined } });
   393→
   394→      if (!fetched.hasMore) break;
   395→      page += 1;
   396→    }
   397→
   398→    // خطای رکورد-منفرد درون صفحات موفق ⇒ PARTIAL (نه FAILED)
   399→    if (status === "COMPLETED" && progress.totalErrors > 0) {
   400→      status = "PARTIAL";
   401→      error = `${progress.totalErrors} رکورد در حین ذخیره خطا خورد (صفحات طی شد؛ از اجرای مجدد idempotent استفاده کنید).`;
   402→    }
   403→
   404→    return {
   405→      entity,
   406→      status,
   407→      progress,
   408→      ...(error !== undefined ? { error } : {}),
   409→      durationMs: this.now().getTime() - t0,
   410→    };
   411→  }
   412→
   413→  /** واکشی یک صفحه با retry طبقه‌بندی‌شده + احترام کامل به rate limit و Retry-After */
   414→  private async fetchPageWithRetry(page: number): Promise<BulkFetchPage> {
   415→    let retryNumber = 0;
   416→    while (true) {
   417→      await this.limiter.wait();
   418→      try {
   419→        const res = await this.deps.fetcher.fetchPage({
   420→          entity: this.opts.entity,
   421→          page,
   422→          pageSize: this.pageSize,
   423→          ...(this.opts.startDate ? { startDate: this.opts.startDate } : {}),
   424→          ...(this.opts.endDate ? { endDate: this.opts.endDate } : {}),
   425→        });
   426→        this.limiter.complete();
   427→        return res;
   428→      } catch (err) {
   429→        this.limiter.complete(); // درخواست (ناموفق) هم «تمام شد» — فاصله از پایان آن
   430→        const { retryable, retryAfterMs } = classifyFetchError(err);
   431→        const policyDelay = retryDelayFor(retryNumber + 1, this.maxRetries);
   432→        if (!retryable || policyDelay === null) throw err;
   433→        // 429: احترام به Retry-After مبدأ — هرگز کمتر از سیاست، هرچه بیشتر خواست شکیبایی
   434→        await this.sleep(Math.max(policyDelay, retryAfterMs ?? 0));
   435→        retryNumber += 1;
   436→      }
   437→    }
   438→  }
   439→}

   441→function sameDateRange(a: BulkImportCheckpoint["dateRange"], b: BulkImportCheckpoint["dateRange"]): boolean {
   442→  return (a?.from ?? null) === (b?.from ?? null) && (a?.to ?? null) === (b?.to ?? null);
   443→}

   445→function fetchFailureMessage(err: unknown, entity: BulkEntity, page: number, lastRecordId: string): string {
   446→  const reason = err instanceof Error ? err.message : String(err);
   447→  return `خطای قطعی در واکشی صفحه ${page} از «${entity}»: ${reason} — checkpoint در آخرین صفحه کامل ذخیره شد (آخرین رکورد: ${lastRecordId || "—"}).`;
   448→}

   453→
   454→export async function bulkImport(
   455→  options: BulkImportOptions,
   456→  deps: Partial<BulkImportDeps> & { armToken?: string; repoClient?: PrismaClient } = {}
   457→): Promise<BulkImportResult> {
   458→  const engineOpts: BulkImportOptions = {
   459→    ...options,
   460→    pageSize: clampPageSize(options.pageSize),
   461→    batchSize: clampBatchSize(options.batchSize),
   462→    rateLimitMs: Math.max(1, options.rateLimitMs ?? 200),
   463→    maxRetries: options.maxRetries ?? 5,
   464→  };
   465→  const fetcher =
   466→    deps.fetcher ??
   467→    createArmBulkFetcher({ entity: options.entity, armToken: deps.armToken ?? "" });
   468→  const sink = deps.sink ?? createRepositorySink(options.entity, { client: deps.repoClient });
   469→  const checkpointStore = deps.checkpointStore ?? new SyncStateBulkCheckpointStore({ client: deps.repoClient });
   470→  return new BulkImportRunner(engineOpts, {
   471→    fetcher,
   472→    sink,
   473→    checkpointStore,
   474→    ...(deps.sleep ? { sleep: deps.sleep } : {}),
   475→    ...(deps.now ? { now: deps.now } : {}),
   476→    ...(deps.newRunId ? { newRunId: deps.newRunId } : {}),
   477→  }).run();
   478→}

   481→// سازگار منبع ARM — تمام درخواست از client.ts موجود (صفر fetch مستقیم)

   484→/** نگاشت ردیف خام → ورودی مخزن برای هر موجودیت (listing/request = stub تا P6.5-04) */
   485→type RawRow = Record<string, unknown>;
   486→
   487→function pickRaw(row: RawRow, keys: string[]): unknown {
   488→  for (const k of keys) {
   489→    const v = row[k];
   490→    if (v !== undefined && v !== null && v !== "") return v;
   491→  }
   492→  return null;
   493→}
   494→
   495→function pickRawStr(row: RawRow, keys: string[]): string | null {
   496→  const v = pickRaw(row, keys);
   497→  return typeof v === "string" ? v : v === undefined || v === null ? null : String(v);
   498→}
   499→
   500→/** STUB P6.5-04 — آگهی ARM → PropertyBulkInput (نگاشت حداقلی؛ تکمیل در normalize.ts) */
   501→function stubMapPropertyRow(row: RawRow): PropertyBulkInput | null {
   502→  const id = pickRawStr(row, ["id", "_id", "objectId"]);
   503→  const propertyCode = pickRawStr(row, ["code", "propertyCode"]);
   504→  if (!id && !propertyCode) return null;
   505→  return {
   506→    id: id ?? propertyCode!,
   507→    propertyCode: propertyCode ?? id!,
   508→    title: pickRawStr(row, ["title"]),
   509→    status: pickRawStr(row, ["status"]),
   510→    ownerName: pickRawStr(row, ["relatedRealtorName", "realtorName"]),
   511→  };
   512→}
   513→
   514→/** STUB P6.5-04 — درخواست ARM → ArmRequestBulkRow (نگاشت حداقلی) */
   515→function stubMapRequestRow(row: RawRow): ArmRequestBulkRow | null {
   516→  const id = pickRawStr(row, ["id", "_id"]);
   517→  const requestCode = pickRawStr(row, ["code", "requestCode"]);
   518→  if (!id && !requestCode) return null;
   519→  return {
   520→    id: id ?? requestCode!,
   521→    requestCode: requestCode ?? id!,
   522→    customerName: pickRawStr(row, ["applicantName", "customerName", "fullName"]),
   523→    customerMobile: pickRawStr(row, ["applicantMobile", "customerMobile", "mobileNumber"]),
   524→  };
   525→}
   526→
   527→export function createArmBulkFetcher(init: { entity: BulkEntity; armToken: string }): BulkPageFetcher {
   528→  return {
   529→    fetchPage: async (req) => {
   530→      if (!init.armToken) {
   531→        throw new BulkFetchError("توکن ARM برای Bulk Import لازم است — نشست کاربر واردشده را تمدید کنید.", { status: 401, retryable: false });
   532→      }
   533→      const cfg = getMelkApiConfig();
   534→      const path = armEntityPath(req.entity);
   535→      const res = await armCall<unknown>({
   536→        path,
   537→        baseUrl: cfg.armApiBase,
   538→        token: init.armToken,
   539→        query: {
   540→          page: req.page,
   541→          limit: req.pageSize,
   542→          // نام پارامترهای بازه تاریخ در P6.5-04/D0 با تست زنده نهایی می‌شود (نگاشت صادقانه)
   543→          ...(req.startDate ? { from: req.startDate.toISOString().slice(0, 10) } : {}),
   544→          ...(req.endDate ? { to: req.endDate.toISOString().slice(0, 10) } : {}),
   545→        },
   546→        timeoutMs: cfg.timeoutMs,
   547→        maxRetries: 0, // مدیریت retry با موتور خود خط لوله (نه client)
   548→      });
   549→
   550→      const mapped = extractPagedList(res.data, (row, i) =>
   551→        mapArmRowToInput(req.entity, (row ?? {}) as RawRow, i, req.page, req.pageSize)
   552→      );
   553→      const totalPages =
   554→        mapped.totalPages ?? (mapped.total > 0 ? Math.ceil(mapped.total / req.pageSize) : 1);
   555→      return {
   556→        rows: mapped.rows,
   557→        hasMore: mapped.rows.length > 0 && req.page < totalPages,
   558→      };
   559→    },
   560→  };
   561→}
   562→
   563→function armEntityPath(entity: BulkEntity): string {
   564→  switch (entity) {
   565→    case "transaction": return MELK_ENDPOINTS.arm.tradedTotal;
   566→    case "expert": return MELK_ENDPOINTS.arm.realtorsList;
   567→    case "interaction": return MELK_ENDPOINTS.arm.realtorInteractions;
   568→    case "visit": return MELK_ENDPOINTS.arm.visitsList;
   569→    case "contract": return MELK_ENDPOINTS.arm.tradeOverviewsList;
   570→    case "listing": return MELK_ENDPOINTS.arm.propertiesList;
   571→    case "request": return MELK_ENDPOINTS.arm.requestsList;
   572→  }
   573→}
   574→
   575→function mapArmRowToInput(entity: BulkEntity, row: RawRow, index: number, page: number, pageSize: number): unknown | null {
   576→  switch (entity) {
   577→    case "transaction": {
   578→      const crm = mapArmTradedRow(row, index, page, pageSize);
   579→      return crm ? toTxUpsert(crm) : null;
   580→    }
   581→    case "expert": {
   582→      const crm = mapArmRealtorRow(row, index);
   583→      return crm ? toExUpsert(crm) : null;
   584→    }
   585→    case "interaction": {
   586→      const crm = mapArmInteractionRow(row, index);
   587→      return crm ? toInUpsert(crm) : null;
   588→    }
   589→    case "visit": {
   590→      const crm = mapArmVisitRow(row, index);
   591→      return crm ? toViUpsert(crm) : null;
   592→    }
   593→    case "contract": {
   594→      // فهرست صورت‌وضعیت‌ها — غنی‌سازی مرحله دوم (info/{id}) در P6.5-04
   595→      const crm = mapArmContractRow(row);
   596→      return crm ? toContractUpsert(crm) : null;
   597→    }
   598→    case "listing": return stubMapPropertyRow(row);
   599→    case "request": return stubMapRequestRow(row);
   600→  }
   601→}

   604→// سینکِ مخزن — هر موجودیت به Repository خود (صفر Prisma مستقیم)
   605→// ==========================================================================
   606→
   607→export function createRepositorySink(entity: BulkEntity, opts: { client?: PrismaClient } = {}): BulkRecordSink {
   608→  // مخازن یک‌بار ساخته می‌شوند — پیش‌فرض تولید (db از env) یا کلاینت تزریقی تست (E.3)
   609→  const txRepo = opts.client ? new PrismaTransactionRepository({ client: opts.client }) : getTransactionRepository();
   610→  const exRepo = opts.client ? new PrismaExpertRepository({ client: opts.client }) : getExpertRepository();
   611→  const viRepo = opts.client ? new PrismaVisitRepository({ client: opts.client }) : getVisitRepository();
   612→  const ctRepo = opts.client ? new PrismaContractRepository({ client: opts.client }) : getContractRepository();
   613→  const rqRepo = opts.client ? new PrismaRequestRepository({ client: opts.client }) : getRequestRepository();
   614→  const prRepo = opts.client ? new PrismaPropertyRecordRepository({ client: opts.client }) : getPropertyRecordRepository();
   615→
   616→  return {
   617→    saveBatch: async (rows, meta) => {
   618→      const up: BulkUpsertOptions = { sourceRunId: meta.sourceRunId };
   619→      switch (entity) {
   620→        case "transaction":
   621→          return txRepo.upsertBulk(rows as TransactionUpsertInput[], up);
   622→        case "expert":
   623→          return exRepo.upsertBulk(rows as ExpertUpsertInput[], up);
   624→        case "interaction":
   625→          return exRepo.upsertInteractionsBulk(rows as InteractionUpsertInput[], up);
   626→        case "visit":
   627→          return viRepo.upsertBulk(rows as VisitUpsertInput[], up);
   628→        case "contract":
   629→          return ctRepo.upsertBulk(rows as ContractUpsertInput[], up);
   630→        case "request":
   631→          return rqRepo.upsertArmBulk(rows as ArmRequestBulkRow[], up);
   632→        case "listing":
   633→          return prRepo.upsertBulk(rows as PropertyBulkInput[], up);
   634→      }
   635→    },
   636→  };
   637→}

   639→// ==========================================================================
   640→// انبار checkpoint — SyncState.singleton از طریق IntegrationRepository (§۱.۵)
   641→// ==========================================================================
   642→
   643→export class SyncStateBulkCheckpointStore implements BulkCheckpointStore {
   644→  private readonly repo;
   645→
   646→  constructor(deps: { client?: PrismaClient } = {}) {
   647→    // IntegrationRepository با کلاینت تزریقی (الگوی تست E.3)
   648→    this.repo = deps.client ? new PrismaIntegrationRepository({ client: deps.client }) : getIntegrationRepository();
   649→  }
   650→
   651→  async load(entity: BulkEntity): Promise<BulkImportCheckpoint | null> {
   652→    const raw = await this.repo.getBulkImportCheckpoint();
   653→    if (!raw) return null;
   654→    try {
   655→      const cp = JSON.parse(raw) as BulkImportCheckpoint;
   656→      if (cp && cp.entity === entity && cp.version === BULK_CHECKPOINT_VERSION) return cp;
   657→      return null; // موجودیت/نسخه متفاوت — نادیده (شروع تازه)
   658→    } catch {
   659→      return null; // JSON خراب — شروع تازه (fail-safe)
   660→    }
   661→  }
   662→
   663→  async save(checkpoint: BulkImportCheckpoint): Promise<void> {
   664→    await this.repo.setBulkImportCheckpoint(JSON.stringify(checkpoint));
   665→  }
   666→}
