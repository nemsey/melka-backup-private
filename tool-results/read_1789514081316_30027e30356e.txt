    37→interface ExRow {
    38→  id: string;
    39→  personnelCode: string;
    40→  fullName: string;
    41→  mobile: string;
    42→  role: string;
    43→  businessTitle: string;

    45→  active: boolean;

    47→  /** مجموع تعامل‌ها (نقش درخواست‌دهنده + نقش آگهی‌گذار) */

    51→interface InRow {
    52→  id: string;

    55→  count: number;

    58→interface Meta {
    59→  roles: string[];
    60→  businesses: string[];
    61→  creators: string[];
    62→}
    63→
    64→interface DataSourceInfo {
    65→  real: boolean;

    67→  lastSuccessAt: string | null;

    71→const EMPTY = { fullName: "", mobile: "", personnelCode: "", role: "all", businessTitle: "all", creator: "all", active: "all" };
    72→type Filters = typeof EMPTY;
    73→
    74→/** رتبه همکاری بر اساس مجموع تعامل‌ها */
    75→function cooperationLevel(total: number): { label: string; style: string } {
    76→  if (total >= 20) return { label: "محور همکاری", style: "bg-teal-100 text-teal-800 dark:bg-teal-950 dark:text-teal-300" };
    77→  if (total >= 5) return { label: "همکار پرتکرار", style: "bg-amber-100 text-amber-800 dark:bg-amber-950 dark:text-amber-300" };
    78→  if (total > 0) return { label: "کم‌تعامل", style: "bg-stone-200 text-stone-700 dark:bg-stone-800 dark:text-stone-300" };
    79→  return { label: "بدون تعامل", style: "bg-stone-200 text-stone-700 dark:bg-stone-800 dark:text-stone-300" };
    80→}
