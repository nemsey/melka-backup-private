   109→/** همگام‌سازی با توکن ARM — با قفل همزمانی: اجرای موازی = پیوستن به اجرای جاری */
   110→export function syncFromArm(armToken: string): Promise<SyncResult> {
   111→  if (__armSyncInFlight) return __armSyncInFlight; // در جریان است → صبر تا اتمام (نه شروع مجدد)
   112→  __armSyncExecutionCount += 1;
   113→  __armSyncInFlight = doSyncFromArm(armToken).finally(() => {
   114→    __armSyncInFlight = null;
   115→  });
   116→  return __armSyncInFlight;
   117→}

   119→async function readSyncState() {
   120→  let state = await db.syncState.findUnique({ where: { id: SYNC_STATE_ID } });
   121→  if (!state) {
   122→    state = await db.syncState.create({ data: { id: SYNC_STATE_ID } });
   123→  }
   124→  return state;
   125→}

   127→function parseSummary(json: string | null | undefined): TradedSummary | null {
   128→  if (!json) return null;
   129→  try {
   130→    const s = JSON.parse(json) as TradedSummary;
   131→    if (typeof s.totalCount === "number") return s;
   132→    return null;
   133→  } catch {
   134→    return null;
   135→  }
   136→}

   138→/** خواندن وضعیت همگام‌سازی برای نمایش (بدون خطا پرتاب نمی‌شود) */
   139→export async function getSyncState(): Promise<SyncStateInfo> {
   140→  const state = await readSyncState();
   141→  const [txCount, exCount, inCount, viCount, sdCount] = await Promise.all([
   142→    db.transactionRecord.count(),
   143→    db.expertRecord.count(),
   144→    db.interactionRecord.count(),
   145→    db.visitRecord.count(),
   146→    db.supplyDemandRecord.count(),
   147→  ]);
   148→  return {
   149→    lastSyncedAt: state.lastSyncedAt?.toISOString() ?? null,
   150→    lastSuccessAt: state.lastSuccessAt?.toISOString() ?? null,
   151→    lastError: state.lastError,
   152→    lastErrorCode: state.lastErrorCode,
   153→    transactionCount: txCount,
   154→    expertCount: exCount,
   155→    interactionCount: inCount,
   156→    visitCount: viCount,
   157→    supplyDemandCount: sdCount,
   158→    summary: parseSummary(state.summaryJson),
   159→    attempts: state.attempts,
   160→  };

   182→/** خواندن یک گزارش صفحه‌بندی‌شده تا انتها (با سقف ایمن صفحات) */
   183→async function fetchAllPages<T>(
   184→  path: string,
   185→  armToken: string,
   186→  maxPages: number,
   187→  pageLimit: number,
   188→  mapRow: (row: Record<string, unknown>, index: number, page: number, limit: number) => T | null
   189→): Promise<{ rows: T[]; rawSummary: unknown }> {
   190→  const cfg = getMelkApiConfig();
   191→  const rows: T[] = [];
   192→  let rawSummary: unknown = null;
   193→  for (let page = 1; page <= maxPages; page++) {
   194→    const res = await armCall<unknown>({
   195→      path,
   196→      baseUrl: cfg.armApiBase,
   197→      token: armToken,
   198→      query: { page, limit: pageLimit },
   199→      timeoutMs: cfg.timeoutMs,
   200→      maxRetries: 0, // در حلقه صفحه‌بندی تلاش مجدد نمی‌کنیم (صفحه بعدی خودش تازه می‌شود)
   201→    });
   202→    // خلاصه چهارگانه گزارش تطابق — خود پاسخ «data» است
   203→    if (path === MELK_ENDPOINTS.arm.tradedTotal && res.data && typeof res.data === "object") {
   204→      const d = res.data as Record<string, unknown>;
   205→      if (typeof d.totalCount === "number" || typeof d.hasBoth === "number") {
   206→        rawSummary = d;
   207→      }
   208→    }
   209→    const { rows: pageRows, total, totalPages } = extractPagedList(res.data, (row, i) => mapRow(row, i, page, pageLimit));
   210→    rows.push(...pageRows);
   211→    const pages = totalPages ?? (total > 0 ? Math.ceil(total / pageLimit) : 1);
   212→    if (page >= pages || pageRows.length === 0) break;
   213→  }
   214→  return { rows, rawSummary };
   215→}

   217→/**
   218→ * استخراج کامل معاملات، کارشناسان، تعاملات، بازدیدها و عرضه/تقاضا از سامانه ARM
   219→ * با توکن ARM. صفحه‌بندی به‌طور خودکار طی می‌شود (تا سقف ایمن صفحات).
   220→ * (پیاده‌سازی داخلی — همیشه از syncFromArm استفاده کنید تا قفل همزمانی اعمال شود)
   221→ */
   222→async function doSyncFromArm(armToken: string): Promise<SyncResult> {
   223→  const now = new Date();
   224→  await writeSyncState({ lastSyncedAt: now });
   225→
   226→  try {
   227→    // ---------------- ۱) گزارش تطابق معاملات + خلاصه ----------------
   228→    const { rows: transactions, rawSummary } = await fetchAllPages(
   229→      MELK_ENDPOINTS.arm.tradedTotal,
   230→      armToken,
   231→      50,
   232→      100,
   233→      mapArmTradedRow
   234→    );
   235→    const summary = mapArmTradedSummary(rawSummary) ?? computeSummaryFromRows(transactions);
   236→
   237→    // ---------------- ۲) فهرست کاربران/مشاوران ----------------
   238→    const { rows: experts } = await fetchAllPages(MELK_ENDPOINTS.arm.realtorsList, armToken, 30, 100, mapArmRealtorRow);
   239→
   240→    // ---------------- ۳) تعاملات مشاوران با یکدیگر ----------------
   241→    let interactions: CrmInteraction[] = [];
   242→    try {
   243→      const { rows: inRows } = await fetchAllPages(MELK_ENDPOINTS.arm.realtorInteractions, armToken, 20, 200, mapArmInteractionRow);
   244→      interactions = inRows;
   245→    } catch {
   246→      // تعاملات گزارش مکمل است — نبودش نباید کل همگام‌سازی را شکست بدهد
   247→      interactions = [];
   248→    }
   249→
   250→    // ---------------- ۴) بازدیدها و پیگیری‌ها (غیربحرانی) ----------------
   251→    let visits: CrmVisit[] = [];
   252→    try {
   253→      const { rows: viRows } = await fetchAllPages(MELK_ENDPOINTS.arm.visitsList, armToken, 20, 100, mapArmVisitRow);
   254→      visits = viRows;
   255→    } catch {
   256→      visits = [];
   257→    }
   258→
   259→    // ---------------- ۵) گزارش عرضه و تقاضا (غیربحرانی) ----------------
   260→    let supplyDemand: CrmSupplyDemand[] = [];
   261→    try {
   262→      const { rows: sdRows } = await fetchAllPages(MELK_ENDPOINTS.armReports.supplyDemand, armToken, 10, 100, mapArmSupplyDemandRow);
   263→      supplyDemand = sdRows;
   264→    } catch {
   265→      supplyDemand = [];
   266→    }
   267→
   268→    // ---------------- ۶) قراردادها و اسناد (صورت وضعیت‌ها) (غیربحرانی) ----------------
   269→    // لیست + جزئیات هر قرارداد (خریداران، فروشندگان، حق‌السهم مشاوران)
   270→    let contracts: CrmContract[] = [];
   271→    try {
   272→      contracts = await fetchContracts(armToken);
   273→    } catch {
   274→      contracts = [];
   275→    }
   276→
   277→    // ---------------- ذخیره اتمیک در پایگاه محلی ----------------
   278→    await db.$transaction([
   279→      db.transactionRecord.deleteMany({}),
   280→      db.expertRecord.deleteMany({}),
   281→      db.interactionRecord.deleteMany({}),
   282→      db.visitRecord.deleteMany({}),
   283→      db.supplyDemandRecord.deleteMany({}),
   284→      db.contractRecord.deleteMany({}),
   285→      ...toTxCreates(transactions),
   286→      ...toExCreates(experts),
   287→      ...toInCreates(interactions),
   288→      ...toViCreates(visits),
   289→      ...toSdCreates(supplyDemand),
   290→      ...toContractCreates(contracts),
   291→    ]);
   292→
   293→    const successAt = new Date();
   294→    await writeSyncState({
   295→      lastSuccessAt: successAt,
   296→      lastError: null,
   297→      lastErrorCode: null,
   298→      attempts: 0,
   299→      summaryJson: summary ? JSON.stringify(summary) : null,
   300→    });
   301→
   302→    return {
   303→      ok: true,
   304→      transactions: transactions.length,
   305→      experts: experts.length,
   306→      interactions: interactions.length,
   307→      visits: visits.length,
   308→      supplyDemand: supplyDemand.length,
   309→      contracts: contracts.length,
   310→      syncedAt: successAt.toISOString(),
   311→    };
   312→  } catch (err) {
   313→    const e = err instanceof MelkApiError ? err : new MelkApiError("UNKNOWN", (err as Error)?.message ?? "خطای نامشخص", 500, false);
   314→    const state = await readSyncState();
   315→    await writeSyncState({
   316→      lastError: e.message,
   317→      lastErrorCode: e.code,
   318→      attempts: state.attempts + 1,
   319→    });
   320→    return {
   321→      ok: false,
   322→      transactions: 0,
   323→      experts: 0,
   324→      interactions: 0,
   325→      visits: 0,
   326→      supplyDemand: 0,
   327→      contracts: 0,
   328→      syncedAt: now.toISOString(),
   329→      error: e.message,
   330→      errorCode: e.code,
   331→    };
   332→  }
   333→}

   335→/** اگر سامانه خلاصه چهارگانه را نفرستاد، از ردیف‌ها محاسبه می‌شود */
   336→function computeSummaryFromRows(rows: CrmTransaction[]): TradedSummary {
   337→  return {
   338→    totalCount: rows.length,
   339→    hasRequestEvent: rows.filter((r) => r.eventsCount > 0).length,
   340→    hasSuggestedProperty: rows.filter((r) => r.suggestedPropertyCount > 0).length,
   341→    hasBoth: rows.filter((r) => r.eventsCount > 0 && r.suggestedPropertyCount > 0).length,
   342→  };
   343→}

   345→/**
   346→ * همگام‌سازی با توکن ARM نشست جاری — اگر توکن منقضی شده باشد
   347→ * ابتدا با refreshToken تمدید می‌شود.
   348→ */
   349→export async function syncWithRefresh(
   350→  armToken: string,
   351→  armRefreshToken: string
   352→): Promise<{ sync: SyncResult; newArmToken?: string }> {
   353→  const provider = getMelkProvider();
   354→  let token = armToken;
   355→  let newArmToken: string | undefined;
   356→
   357→  // تلاش اول؛ در صورت انقضا، تمدید و تلاش مجدد
   358→  let sync = await syncFromArm(token);
   359→  if (!sync.ok && (sync.errorCode === "UNAUTHORIZED" || sync.errorCode === "OTP_INVALID") && armRefreshToken) {
   360→    try {
   361→      const refreshed = await provider.refreshArmToken(armRefreshToken);
   362→      token = refreshed.accessToken;
   363→      newArmToken = refreshed.accessToken;
   364→      sync = await syncFromArm(token);
   365→    } catch (refreshErr) {
   366→      const e = refreshErr instanceof MelkApiError ? refreshErr : new MelkApiError("UNKNOWN", "تمدید توکن ناموفق", 401, false);
   367→      sync = { ok: false, transactions: 0, experts: 0, interactions: 0, visits: 0, supplyDemand: 0, contracts: 0, syncedAt: new Date().toISOString(), error: e.message, errorCode: e.code };
   368→    }
   369→  }
   370→  return { sync, newArmToken };
   371→}

   373→/** در صورت خالی‌بودن کش، داده‌های واقعی را با توکن ARM بارگذاری می‌کند */
   374→export async function ensureSeeded(armToken?: string): Promise<{ transactions: number; experts: number; interactions: number; visits: number; supplyDemand: number; liveError?: string }> {
   375→  const [txCount, exCount, inCount, viCount, sdCount] = await Promise.all([
   376→    db.transactionRecord.count(),
   377→    db.expertRecord.count(),
   378→    db.interactionRecord.count(),
   379→    db.visitRecord.count(),
   380→    db.supplyDemandRecord.count(),
   381→  ]);
   382→
   383→  if ((txCount === 0 || exCount === 0) && armToken) {
   384→    const { sync } = await syncWithRefresh(armToken, "");
   385→    if (sync.ok) return { transactions: sync.transactions, experts: sync.experts, interactions: sync.interactions, visits: sync.visits, supplyDemand: sync.supplyDemand };
   386→    return { transactions: txCount, experts: exCount, interactions: inCount, visits: viCount, supplyDemand: sdCount, liveError: sync.error };
   387→  }
   388→  return { transactions: txCount, experts: exCount, interactions: inCount, visits: viCount, supplyDemand: sdCount };
   389→}

   399→  await db.interactionRecord.deleteMany({});
   400→  await db.visitRecord.deleteMany({});

   402→  await db.contractRecord.deleteMany({});
   403→}

   500→export async function fetchContracts(armToken: string): Promise<CrmContract[]> {

   504→  for (const c of rows.slice(0, 400)) {

   508→        baseUrl: cfg.armApiBase,
   509→        token: armToken,
   510→        timeoutMs: cfg.timeoutMs,
   511→        maxRetries: 0,
   512→      });
   513→      detailed.push(mergeArmContractInfo(c, info.data));

   516→      detailed.push(c);
   517→    }
   518→  }
   519→  return detailed;
   520→}

   552→/** کلیدهای کش فهرست‌های عمومی */
   553→export const REF_KEYS = {
   554→  characters: "user-characters",
   555→  jobs: "user-jobs",
   556→  roles: "user-roles",
   557→  seriousness: "user-seriousness",
   558→  priceIndices: "price-indices",
   559→  commissionConfig: "commission-config",
   560→} as const;
   561→
   562→/** عمر مفید کش فهرست‌های عمومی — ۶ ساعت */
   563→const REF_TTL_MS = 6 * 60 * 60 * 1000;
   564→
   565→/** خواندن کش فهرست عمومی — null اگر نیست یا کهنه است */
   566→async function readRefCache<T>(key: string): Promise<T | null> {
   567→  const row = await db.refListCache.findUnique({ where: { key } });
   568→  if (!row) return null;
   569→  const age = Date.now() - row.fetchedAt.getTime();
   570→  if (age > REF_TTL_MS) return null;
   571→  try {
   572→    return JSON.parse(row.dataJson) as T;
   573→  } catch {
   574→    return null;
   575→  }
   576→}
   577→
   578→/** نوشتن کش فهرست عمومی */
   579→async function writeRefCache(key: string, data: unknown): Promise<void> {
   580→  await db.refListCache.upsert({
   581→    where: { key },
   582→    update: { dataJson: JSON.stringify(data), fetchedAt: new Date() },
   583→    create: { key, dataJson: JSON.stringify(data) },
   584→  });
   585→}

   587→/**
   588→ * فهرست‌های مرجع باشگاه مشتریان — عمومی (بدون توکن) با کش ۶ ساعته:
   589→ * شخصیت‌ها، شغل‌ها، نقش‌ها و جدیت مشتریان.
   590→ */
   591→export async function getCustomerRefLists(forceRefresh = false): Promise<{
   592→  characters: RefListItem[];
   593→  jobs: RefListItem[];
   594→  roles: RefListItem[];
   595→  seriousness: RefListItem[];
   596→  liveError: string | null;
   597→}> {
   598→  const cfg = getMelkApiConfig();
   599→  if (!forceRefresh) {
   600→    const [characters, jobs, roles, seriousness] = await Promise.all([
   601→      readRefCache<RefListItem[]>(REF_KEYS.characters),
   602→      readRefCache<RefListItem[]>(REF_KEYS.jobs),
   603→      readRefCache<RefListItem[]>(REF_KEYS.roles),
   604→      readRefCache<RefListItem[]>(REF_KEYS.seriousness),
   605→    ]);
   606→    if (characters && jobs && roles && seriousness) {
   607→      return { characters, jobs, roles, seriousness, liveError: null };
   608→    }
   609→  }
   610→
   611→  try {
   612→    const [charRes, jobRes, roleRes, seriousRes] = await Promise.all(
   613→      [
   614→        MELK_ENDPOINTS.arm.userCharacters,
   615→        MELK_ENDPOINTS.arm.userJobs,
   616→        MELK_ENDPOINTS.arm.userRoles,
   617→        MELK_ENDPOINTS.arm.userSeriousness,
   618→      ].map(async (path) => {
   619→        const r = await armCall<unknown>({
   620→          path,
   621→          baseUrl: cfg.armApiBase,
   622→          timeoutMs: cfg.timeoutMs,
   623→          maxRetries: 1,
   624→        });
   625→        return r.data;
   626→      })
   627→    );
   628→    const characters = mapRefList(charRes);
   629→    const jobs = mapRefList(jobRes);
   630→    const roles = mapRefList(roleRes);
   631→    const seriousness = mapRefList(seriousRes);
   632→    // ذخیره کش حتی ناقص — هر فهرست جداگانه
   633→    await Promise.all([
   634→      writeRefCache(REF_KEYS.characters, characters),
   635→      writeRefCache(REF_KEYS.jobs, jobs),
   636→      writeRefCache(REF_KEYS.roles, roles),
   637→      writeRefCache(REF_KEYS.seriousness, seriousness),
   638→    ]);
   639→    return { characters, jobs, roles, seriousness, liveError: null };
   640→  } catch (err) {
   641→    // بازگشت کش کهنه (اگر هست) با پیام خطا
   642→    const [characters, jobs, roles, seriousness] = await Promise.all([
   643→      db.refListCache.findUnique({ where: { key: REF_KEYS.characters } }),
   644→      db.refListCache.findUnique({ where: { key: REF_KEYS.jobs } }),
   645→      db.refListCache.findUnique({ where: { key: REF_KEYS.roles } }),
   646→      db.refListCache.findUnique({ where: { key: REF_KEYS.seriousness } }),
   647→    ]);
   648→    const parse = (row: { dataJson: string } | null): RefListItem[] => {
   649→      if (!row) return [];
   650→      try { return JSON.parse(row.dataJson) as RefListItem[]; } catch { return []; }
   651→    };
   652→    return {
   653→      characters: parse(characters),
   654→      jobs: parse(jobs),
   655→      roles: parse(roles),
   656→      seriousness: parse(seriousness),
   657→      liveError: err instanceof MelkApiError ? err.message : "خطا در دریافت فهرست‌های مرجع مشتریان.",
   658→    };
   659→  }
   660→}

   665→export async function getCommissionConfig(forceRefresh = false): Promise<{

   676→    const res = await armCall<unknown>({

   694→/**
   695→ * شاخص‌های قیمت روز سامانه ARM — عمومی با کش ۶ ساعته:
   696→ * فهرست + مناطق هر شاخص + مشخصات فنی.
   697→ */
   698→export async function getPriceIndexConfigs(forceRefresh = false): Promise<{
   699→  indices: PriceIndexConfig[];
   700→  liveError: string | null;
   701→}> {
   702→  const cfg = getMelkApiConfig();
   703→  if (!forceRefresh) {
   704→    const cached = await readRefCache<PriceIndexConfig[]>(REF_KEYS.priceIndices);
   705→    if (cached && cached.length > 0) return { indices: cached, liveError: null };
   706→  }
   707→  try {
   708→    const listRes = await armCall<unknown>({
   709→      path: MELK_ENDPOINTS.arm.currentPricesList,
   710→      baseUrl: cfg.armApiBase,
   711→      query: { returnList: "true" },
   712→      timeoutMs: cfg.timeoutMs,
   713→      maxRetries: 1,
   714→    });
   715→    const { rows } = extractPagedList<Record<string, unknown>>(listRes.data, (row) => (row && typeof row === "object" ? (row as Record<string, unknown>) : null));
   716→    const indices: PriceIndexConfig[] = [];
   717→    for (const row of rows) {
   718→      const base = mapPriceIndexConfig(row);
   719→      if (!base) continue;
   720→      // مناطق و مشخصات فنی هر شاخص — عمومی
   721→      const [areasRes, detailsRes] = await Promise.all([
   722→        armCall<unknown>({
   723→          path: MELK_ENDPOINTS.arm.currentPriceAreas(base.id),
   724→          baseUrl: cfg.armApiBase,
   725→          timeoutMs: cfg.timeoutMs,
   726→          maxRetries: 0,
   727→        }).catch(() => null),
   728→        armCall<unknown>({
   729→          path: MELK_ENDPOINTS.arm.currentPriceDetails(base.id),
   730→          baseUrl: cfg.armApiBase,
   731→          timeoutMs: cfg.timeoutMs,
   732→          maxRetries: 0,
   733→        }).catch(() => null),
   734→      ]);
   735→      indices.push({
   736→        ...base,
   737→        areas: areasRes ? mapPriceIndexAreas(areasRes.data) : [],
   738→        details: detailsRes ? mapPriceIndexDetails(detailsRes.data) : [],
   739→      });
   740→    }
   741→    if (indices.length > 0) await writeRefCache(REF_KEYS.priceIndices, indices);
   742→    return { indices, liveError: null };
   743→  } catch (err) {
   744→    const cached = await readRefCache<PriceIndexConfig[]>(REF_KEYS.priceIndices);
   745→    return {
   746→      indices: cached ?? [],
   747→      liveError: err instanceof MelkApiError ? err.message : "خطا در دریافت شاخص‌های قیمت روز.",
   748→    };
   749→  }
   750→}
