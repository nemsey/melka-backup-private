#!/bin/bash

set -e

# root 控制面（MCP 服务）的 PATH：只含 root 拥有的系统目录，不含任何 z 可写目录。
# 理由见下面「Starting ZAI service」那段。
ROOT_CONTROL_PLANE_PATH=/usr/sbin:/usr/bin:/sbin:/bin

# Boot timeline observability: lets us attribute FC 412 (port 81 health check
# failed in 120s) to image-pull vs start.sh-step latency. /proc/uptime gives
# seconds since container start, so subtracting from current time isolates the
# FC platform overhead from anything start.sh itself does.
BOOT_TIMELINE=/tmp/boot-timeline.log
: > "$BOOT_TIMELINE"
# All helpers below guard /proc/uptime with `|| echo ?` so a missing file
# (non-Linux dev shells, broken procfs) never trips set -e and kills bootstrap.
boot_uptime() { awk '{print $1}' /proc/uptime 2>/dev/null || echo "?"; }
boot_ts()     { date -u +%FT%TZ; }
echo "[BOOT] start.sh entered at $(boot_ts) uptime=$(boot_uptime)s" | tee -a "$BOOT_TIMELINE"

# Dump the full timeline on any exit. FC's 412 error body only captures stdout
# tail; without this, all early-step durations get truncated and we lose the
# evidence we need to pick a real fix.
dump_boot_timeline() {
    local rc=$?
    echo "[BOOT] exit rc=$rc at $(boot_ts) uptime=$(boot_uptime)s" >> "$BOOT_TIMELINE"
    echo ""
    echo "==== BOOT TIMELINE (full) ===="
    cat "$BOOT_TIMELINE" 2>/dev/null || true
    echo "==== END BOOT TIMELINE ===="
}
trap dump_boot_timeline EXIT

# 添加耗时统计函数
log_step_start() {
    local step_name="$1"
    echo "=========================================="
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] Starting: $step_name"
    echo "=========================================="
    echo "[BOOT] $(boot_ts) start step=\"$step_name\" uptime=$(boot_uptime)s" >> "$BOOT_TIMELINE"
    export STEP_START_TIME=$(date +%s)
}

log_step_end() {
    local step_name="$1"
    if [ -z "$step_name" ]; then
        step_name="Unknown step"
    fi
    local end_time=$(date +%s)
    local duration=$((end_time - STEP_START_TIME))
    echo "=========================================="
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] Completed: $step_name"
    echo "[LOG] Step: $step_name | Duration: ${duration}s"
    echo "=========================================="
    echo ""
    echo "[BOOT] $(boot_ts) end   step=\"$step_name\" duration=${duration}s uptime=$(boot_uptime)s" >> "$BOOT_TIMELINE"
}

# 导出函数以便子shell可以使用
export -f log_step_start
export -f log_step_end

# 沙盒模式（见 app/util/chatctx.py）：user = 一个用户一个沙盒，wsmgr 建函数时写 env SANDBOX_MODE=user。
# 老函数没有这个 env，下面所有 legacy 段落照旧执行。
# user 沙盒启动时没有项目：每个 chat 的项目在 /home/z/chats/chat-{id}，第一次被请求时由
# 控制服务（app/util/chatinit.py）恢复，做的事对应下面「项目初始化 / 官方 skills / lockfile / git」四段。
USER_MODE=false
if [ "${SANDBOX_MODE:-}" = "user" ]; then
  USER_MODE=true
  log_step_start "User-mode sandbox dirs"
  mkdir -p /home/z/chats /home/z/.chat-state /home/z/.chat-tmp
  chown z:z /home/z/chats /home/z/.chat-tmp
  chmod 755 /home/z/chats /home/z/.chat-tmp
  # .chat-state 里有每个 chat 的 token，只给控制服务（root）读。
  chmod 700 /home/z/.chat-state
  log_step_end "User-mode sandbox dirs"
fi

if [ "$USER_MODE" != "true" ]; then
# 检查 /home/sync 是否存在 repo.tar，决定走"全新项目"还是"恢复"分支
log_step_start "Project initialization check"
SYNC_DIR=/home/sync
ARCHIVE="$SYNC_DIR/repo.tar"
if [ ! -f "$ARCHIVE" ]; then
  # 没有 repo.tar：要么是全新 chat，要么上次 pre-stop 写失败
  # zcode.tar.gz 是 agent 模式的会话状态（zcode-driver/boot.sh 管），和项目无关，不算「别的内容」
  if [ -n "$(ls -A $SYNC_DIR 2>/dev/null | grep -v '^\.' | grep -v '^upload$' | grep -v '^zcode\.tar\.gz')" ]; then
    echo "WARNING: $ARCHIVE missing but $SYNC_DIR has other content (likely failed pre-stop), treating as clean project"
  else
    echo "Whoa, wat a nice clean project"
  fi
  echo "DATABASE_URL=file:/home/z/my-project/db/custom.db" > /home/z/my-project/.env
  mkdir -p /home/z/my-project/download && echo "Here are all the generated files." > /home/z/my-project/download/README.md
  chown -R z:z /home/z/my-project/download
  chmod -R 755 /home/z/my-project/download
  # 确保 skills 目录存在并设置权限（只 chown skills，避免递归整个项目）
  mkdir -p /home/z/my-project/skills
  chown -R z:z /home/z/my-project/skills
  chmod -R 755 /home/z/my-project/skills
else
  echo "restoring $SYNC_DIR to /home/z/my-project (archive size: $(stat -c%s "$ARCHIVE" 2>/dev/null || echo unknown) bytes)"
  # 检查并处理可能存在的挂载点（如 upload 目录）
  if [ -d "/home/z/my-project" ]; then
    echo "Cleaning project directory (preserving mount points)..."
    # 删除所有文件和目录，但跳过挂载点
    find /home/z/my-project -mindepth 1 -maxdepth 1 ! -path "/home/z/my-project/upload" -exec rm -rf {} + 2>/dev/null || true
    # 如果 upload 不是挂载点，也删除它
    if ! mountpoint -q /home/z/my-project/upload 2>/dev/null; then
      rm -rf /home/z/my-project/upload
    else
      echo "Skipping /home/z/my-project/upload (mount point detected)"
    fi
  fi
  mkdir -p /home/z/my-project
  # 解压前先看放不放得进盘（放不进就裁掉最大的工作区文件，不把盘写满），失败不留半截：
  # 见 app/util/diskguard.py。rc=3 是放不下（已清空、会在项目里留说明），其它非 0 是异常，都继续用现有内容启动。
  if ! (cd /app && /app/.venv/bin/python -c "from app.util.diskguard import legacy_restore_main; raise SystemExit(legacy_restore_main())" "$ARCHIVE" /home/z/my-project); then
    echo "WARNING: guarded restore of $ARCHIVE did not complete cleanly (disk full or corrupted archive?), continuing with whatever was restored"
  fi
  echo "DATABASE_URL=file:/home/z/my-project/db/custom.db" > /home/z/my-project/.env

  # 修改所有权和权限，但排除挂载点
  if mountpoint -q /home/z/my-project/upload 2>/dev/null; then
    echo "Setting ownership (handling upload mount point specially)..."
    # 对根目录下的文件和目录分别处理（排除 upload）
    find /home/z/my-project -mindepth 1 -maxdepth 1 ! -path "/home/z/my-project/upload" -exec chown -R z:z {} + 2>/dev/null || true
    find /home/z/my-project -mindepth 1 -maxdepth 1 ! -path "/home/z/my-project/upload" -exec chmod -R 755 {} + 2>/dev/null || true
    # OSS 挂载点是只读的，跳过权限设置
    echo "Skipping permission changes on OSS mount point: /home/z/my-project/upload"
  else
    echo "Setting ownership (no mount points detected)..."
    chown -R z:z /home/z/my-project
    chmod -R 755 /home/z/my-project
  fi
fi
chown -R z:z /home/z/my-project/.env
log_step_end "Project initialization check"

# 解压 official skills 文件到项目目录
log_step_start "Extracting official skills to project"
if [ -d "/home/official_skills" ]; then
  echo "Extracting skills from /home/official_skills to /home/z/my-project/skills..."
  mkdir -p /home/z/my-project/skills

  # 从 stages.yaml 收集允许的 skill 名（所有 stage 的正向引用，去重），无配置则不限制
  allowed_skills=""
  if [ -f "/home/official_skills/stages.yaml" ]; then
    allowed_skills=$(grep -E '^\s+-\s+\S' /home/official_skills/stages.yaml | sed 's/^\s*-\s*//' | tr -d '"'"'" | sort -u)
    echo "Stages config found, allowed skills: $(echo "$allowed_skills" | tr '\n' ' ')"
  fi

  # 先按 allowed_skills 过滤，把要解压的 zip 写进临时列表，避免 unzip 阶段再判断
  todo_file=$(mktemp)
  zip_count=0
  for zip_file in /home/official_skills/*.zip; do
    [ -f "$zip_file" ] || continue
    skill_name=$(basename "$zip_file" .zip)
    if [ -n "$allowed_skills" ] && ! echo "$allowed_skills" | grep -qx "$skill_name"; then
      echo "Skipping $skill_name: not in stages config"
      continue
    fi
    echo "$zip_file" >> "$todo_file"
    zip_count=$((zip_count + 1))
  done

  # 并行解压。FC 实例通常 1-2 vCPU，但 unzip 在 decompress 间隙会让出 CPU，
  # -P 4 实测能把解压总耗时压到串行的 30-50%。-qq 比 -q 更安静，少了 -x 的
  # "caution: excluded filename" 噪音行，减少 FC stdout buffering 开销。
  #
  # 每个 worker 先把 zip 顺序 cp 到本地 /tmp 再 unzip：FC layer 挂载点对随机小块
  # 读延迟高（unzip 内部按小块读），改成"cp（大块顺序读）+ 本地 unzip"FC 团队
  # 反馈能显著快于直接对挂载点 unzip。
  if [ "$zip_count" -gt 0 ]; then
    stage_dir=$(mktemp -d /tmp/skills_zip_XXXXXX)
    xargs -a "$todo_file" -P 4 -I{} \
      sh -c '
        src="$1"; stage="$2"
        base=$(basename "$src")
        echo "Extracting $base..."
        cp -f "$src" "$stage/$base"
        unzip -qq -o "$stage/$base" -x "__MACOSX/*" -x "*.DS_Store" -x "._*" -d /home/z/my-project/skills/
        rm -f "$stage/$base"
      ' _ {} "$stage_dir"
    rmdir "$stage_dir" 2>/dev/null || true
  fi
  rm -f "$todo_file"

  # 修正所有权和权限。root 解压导致 owner 是 root，zip 内可能含受限权限位。
  # 按 skills 下的顶层条目并发执行（不加 -type d，兼顾极少数直接散在顶层的文件），
  # 比单线程 chown -R / chmod -R 在大量小文件上更快。
  chown z:z /home/z/my-project/skills 2>/dev/null || true
  find /home/z/my-project/skills -maxdepth 1 -mindepth 1 -print0 \
    | xargs -0 -P 4 -I{} sh -c 'chown -R z:z "$1" 2>/dev/null; chmod -R 755 "$1" 2>/dev/null' _ {} \
    || true

  if [ "$zip_count" -eq 0 ]; then
    echo "No zip files found in /home/official_skills"
  else
    echo "Successfully extracted $zip_count skill(s)"
  fi
else
  echo "Warning: /home/official_skills directory not found"
fi
log_step_end "Extracting official skills to project"


# 把 lockfile 里锁死的 registry 域名改成本 region 的镜像源。
#
# 为什么必须改 lockfile 而不是只设 NPM_CONFIG_REGISTRY：
# 脚手架自带的 bun.lock 把**每个包的下载地址**都写成了绝对 URL
# （937 条 https://registry.npmjs.com/...）。bun install 按 lockfile 里的 URL
# 直接取包，完全绕过 registry 配置——所以注入 NPM_CONFIG_REGISTRY 对它无效。
#
# 国内站实测（BJ 沙箱 → 两个源，同一个 next 15.1.6 包）：
#   registry.npmjs.com      metadata 30s（撞超时）  tarball 25MB/59.4s  423KB/s
#   registry.npmmirror.com  metadata  1.6s          tarball 25MB/ 0.89s  28MB/s
# 937 个包累积起来就是 bun install 跑 7 分钟（14:58:07 → 15:05:13 实测），
# 期间 dev server 起不来、3000 端口不监听、preview 恒 502，用户看到「起不来」。
# 香港侧访问 npmjs 只要 0.14s，所以这个缺陷只在国内站暴露。
#
# 放在这里（解包之后、任何 bun install 之前）而不是放进 .zscripts/*.sh：
#   - bun install 有三个入口（dev.sh 两处、build.sh、mini-services-install.sh），
#     逐个改会漏；
#   - 这里是所有路径的必经点，且同时覆盖「全新项目」和「从 repo.tar 恢复」
#     两个分支——后者解出来的是用户历史的 lockfile，同样锁着 npmjs。
#
# 幂等：改过的再改一次是 no-op。
# 香港不配 NPM_CONFIG_REGISTRY ⇒ 整段跳过，行为逐字节不变。
log_step_start "Rewriting lockfile registry"
# ★ 本文件第 3 行有 set -e：这一段里任何命令返回非 0 都会**终止整个 start.sh**，
#   沙箱直接起不来——那比 bun install 慢严重得多。所以整段显式关掉 errexit，
#   段末恢复。改写 registry 是优化，任何情况下都不该阻断沙箱启动。
set +e
if [ -n "${NPM_CONFIG_REGISTRY:-}" ]; then
  # 用 shell 参数展开取 host，不走 sed：提取和改写都依赖同一个外部命令时，
  # 那个命令一出问题就会「_reg_host 变空 → 整段静默跳过」，连告警都没有。
  _reg_host="${NPM_CONFIG_REGISTRY#*://}"
  _reg_host="${_reg_host%%/*}"
  _reg_host="${_reg_host%%:*}"
  if [ -z "$_reg_host" ]; then
    echo "ERROR: NPM_CONFIG_REGISTRY=${NPM_CONFIG_REGISTRY} 解析不出 host，lockfile 未改写"
  fi
  if [ -n "$_reg_host" ]; then
    # 只处理文本格式的 bun.lock。bun.lockb 是二进制、sed 改不了，
    # 当前脚手架用的是 JSON 文本格式（已确认无 .lockb），真出现时下面会报出来。
    if [ -f /home/z/my-project/bun.lockb ]; then
      echo "WARNING: bun.lockb (binary) found — registry rewrite does not apply to it"
    fi
    # ★ 改完必须回读校验，不能只看 sed 的返回码。
    # 2026-09-23 自测时在 BSD sed 上踩到：sed 报 "invalid command code" 失败了，
    # 而脚本照样打印 "files changed: 2"——「我调用了」不等于「它改成了」。
    # 沙箱是 GNU sed，但一个只在某些环境静默失效的改写，不如让它自己喊出来。
    _n=0; _fail=0
    for _lock in $(find /home/z/my-project -maxdepth 4 -name bun.lock -not -path '*/node_modules/*' 2>/dev/null); do
      grep -q 'registry\.npmjs\.com' "$_lock" 2>/dev/null || continue
      sed -i "s#registry\.npmjs\.com#${_reg_host}#g" "$_lock" 2>/dev/null
      if grep -q 'registry\.npmjs\.com' "$_lock" 2>/dev/null; then
        echo "ERROR: lockfile registry rewrite FAILED (still npmjs): $_lock"
        _fail=$((_fail+1))
      else
        _n=$((_n+1))
      fi
    done
    if [ "$_fail" -gt 0 ]; then
      echo "ERROR: ${_fail} lockfile(s) still point at registry.npmjs.com —"\
           " bun install will fetch cross-border and dev server may take minutes to start"
    fi
    echo "lockfile registry rewritten to ${_reg_host} (ok=${_n} failed=${_fail})"
  fi
else
  echo "NPM_CONFIG_REGISTRY not set, keep lockfile registry as-is"
fi
set -e
log_step_end "Rewriting lockfile registry"
fi  # USER_MODE != true（项目初始化 / 官方 skills / lockfile）


# 先给权限，防止产生一些用户配置
log_step_start "Setting permissions"
chown z:z /home/z
chmod 755 /home/z
if [ "$USER_MODE" != "true" ]; then
# 确保 /home/sync 目录存在并给 z 用户读写权限
mkdir -p /home/sync
# 检查 /home/sync 是否为挂载点，如果是则跳过权限设置

else

/bin/bash /app/zcode-driver/boot.sh restore || true
log_step_end "Restoring zcode state"

if [ -f /home/z/.bash_profile ]; then

# 初始化 git（仅首次）
log_step_start "Git setup"
if [ "$USER_MODE" = "true" ]; then
  # 每个 chat 的 safe.directory / .gitignore / init 由 chatinit 做。
  su z -c "
git config --global user.email 'z@container'

else
cd /home/z/my-project

su z -c "
git config --global --add safe.directory /home/z/my-project
git config --global user.email 'z@container'
git config --global user.name 'Z User'
"

# Write .gitignore BEFORE any git op so git can prune skills/ as it walks the
# working tree. Without this, `git add .` lstats every file under
# /home/z/my-project/skills/ (~10k+ files from the 58 freshly-extracted official
# skill zips), blows past the 60s boot budget, and trips the EXIT trap — FC
# treats the SIGTERM as an unhealthy instance and rebuilds, looping forever.
# Incident 2026-06-09: agent-claw-cb40fa5b boot-looped 36 times in 12.5h.
# The same .gitignore also keeps runtime git_commit (app/util/git.py, git add -A)
# from hashing the skill tree on every user commit.
#
# Scope is intentionally narrow: only skills/ and node_modules/. Both are already
# explicitly excluded in the /git/ls-tree handler (mcp.py:600
# `iglob("!node_modules").iglob("!skills")`), so adding them to .gitignore
# costs nothing visible. Wider entries (upload/, download/, db/, *.log, .venv/,
# __pycache__/, .next/, dist/, build/) used to live here too but `ripgrep --files`
# honors .gitignore by default, which silently hid those dirs from /git/ls-tree —
# the dir tree the agent uses to enumerate user-visible project content. Keep
# this list minimal; do not add anything that the ls-tree handler doesn't
# already exclude.
#
# Auto-heal for instances polluted by b407e6a's broad .gitignore: that commit
# wrote `upload/` `download/` `db/` etc. into the file, then `git add . && git
# commit` tracked the .gitignore, so the broad version is baked into repo.tar
# and self-perpetuates across cold starts. Detect via the unique triple
# upload/+download/+db/ (sandbox-internal dir names that would never coincide
# in a user-authored .gitignore) and rewrite to the narrow version. Runs once
# per affected instance; after the rewrite + the next runtime git_commit, the
# narrow version lands in tar and the trigger no longer fires (convergent).
needs_gitignore_write=false
if [ ! -f /home/z/my-project/.gitignore ]; then
  needs_gitignore_write=true
elif grep -qE '^upload/$' /home/z/my-project/.gitignore \
     && grep -qE '^download/$' /home/z/my-project/.gitignore \
     && grep -qE '^db/$' /home/z/my-project/.gitignore; then
  echo "[BOOT] auto-heal: detected broad .gitignore from b407e6a, rewriting to narrow"
  needs_gitignore_write=true
fi
if [ "$needs_gitignore_write" = "true" ]; then
  cat > /home/z/my-project/.gitignore <<'EOF'
skills/
node_modules/
EOF
  chown z:z /home/z/my-project/.gitignore
fi

if [ ! -d ".git" ]; then
  echo "Initializing git repository for the first time..."
  # Split init/add/commit into per-command timing so the next 412 tells us
  # *which* sub-step hangs. Each command is wrapped in `timeout` so a real
  # hang fails fast (~60s) instead of dragging out to FC's 120s deadline +
  # SIGKILL — failing via bash's normal exit path lets the EXIT trap fire
  # and dump /tmp/boot-timeline.log to FC's log buffer.
  echo "[BOOT] $(boot_ts) git: starting init" | tee -a "$BOOT_TIMELINE"
  su z -c "cd /home/z/my-project && git config --global init.defaultBranch main && timeout --kill-after=10s 60s git init"
  echo "[BOOT] $(boot_ts) git: init done" >> "$BOOT_TIMELINE"

  echo "[BOOT] $(boot_ts) git: starting add ." | tee -a "$BOOT_TIMELINE"
  su z -c "cd /home/z/my-project && timeout --kill-after=10s 60s git add ."
  echo "[BOOT] $(boot_ts) git: add done" >> "$BOOT_TIMELINE"

  echo "[BOOT] $(boot_ts) git: starting commit" | tee -a "$BOOT_TIMELINE"
  su z -c "cd /home/z/my-project && timeout --kill-after=10s 60s git commit -m 'Initial commit'"
  echo "[BOOT] $(boot_ts) git: commit done" >> "$BOOT_TIMELINE"
else
  echo "Existing git repository detected, preserving current git state."
fi
fi  # USER_MODE
log_step_end "Git setup"

log_step_start "Z.ai config setup"
# gateway 根地址可注入：与 oss_gateway.py / prestop.py 共用 AI_GATEWAY_URL 这个 env
# （wsmgr 建 FC 函数时下发；缺省保持香港现网值）。带不带 /v1 都接受，防双拼。
_gateway="${AI_GATEWAY_URL:-https://internal-api.z.ai}"
case "$_gateway" in */v1) _gateway="${_gateway%/v1}";; esac
echo "{\"baseUrl\": \"${_gateway}/v1\", \"apiKey\": \"Z.ai\"}" > /etc/.z-ai-config
chmod 444 /etc/.z-ai-config
if [ "$USER_MODE" = "true" ]; then
  # 每个 chat 的命令在自己的 namespace 里看到 /etc/zai-chat/{chat}.json（chatctx.ns_command）。
  # 这个 chat 还没收到过 update_zai_cfg 时用 _default.json：不带 token 的初始配置，
  # 而不是 /etc/.z-ai-config（那里是最后一个写入的 chat 的 token）。
  mkdir -p /etc/zai-chat
  chown root:root /etc/zai-chat
  chmod 755 /etc/zai-chat
  cp /etc/.z-ai-config /etc/zai-chat/_default.json
  chmod 444 /etc/zai-chat/_default.json
  # 宿主视图里的 /home/z/my-project 不属于任何 chat；chat 的命令在 namespace 里看到的是自己的目录。
  # 改成 z 不可写：namespace 没套上的进程往这里写会直接失败，不会悄悄写进一个所有 chat 共用的目录。
  chown -R root:root /home/z/my-project
  chmod -R a-w /home/z/my-project
fi

# 添加服务就绪检查函数
wait_for_service() {
    local host="$1"
    local port="$2"
    local service_name="$3"
    local max_attempts=60
    local attempt=1
    
    echo "Waiting for $service_name to be ready on $host:$port..."
    
    while [ $attempt -le $max_attempts ]; do
        if curl -s --connect-timeout 2 --max-time 5 "http://$host:$port" > /dev/null 2>&1; then
            echo "$service_name is ready!"
            return 0
        fi
        
        echo "Attempt $attempt/$max_attempts: $service_name not ready yet, waiting..."
        sleep 1
        attempt=$((attempt + 1))
    done
    
    echo "ERROR: $service_name failed to start within $((max_attempts * 2)) seconds"
    return 1
}
